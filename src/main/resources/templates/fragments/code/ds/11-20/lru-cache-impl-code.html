Python — 해시맵과 이중 연결 리스트로 직접 구현

class Node:
    def __init__(self, key=0, val=0):
        self.key, self.val = key, val
        self.prev = self.next = None

class LRUCache:
    def __init__(self, capacity):
        self.cap = capacity
        self.map = {}                           # key -> Node
        self.head, self.tail = Node(), Node()   # 더미 노드로 경계 처리를 없앤다
        self.head.next, self.tail.prev = self.tail, self.head

    def _remove(self, node):
        node.prev.next, node.next.prev = node.next, node.prev

    def _add_front(self, node):
        node.prev, node.next = self.head, self.head.next
        self.head.next.prev = node
        self.head.next = node

    def get(self, key):
        if key not in self.map:
            return None
        node = self.map[key]
        self._remove(node)
        self._add_front(node)                  # 사용했으니 맨 앞으로
        return node.val

    def put(self, key, val):
        if key in self.map:
            node = self.map[key]
            node.val = val
            self._remove(node)
            self._add_front(node)
            return
        if len(self.map) == self.cap:
            lru = self.tail.prev                # 맨 뒤 = 가장 오래 쓰지 않은 노드
            self._remove(lru)
            del self.map[lru.key]
        node = Node(key, val)
        self.map[key] = node
        self._add_front(node)

    def keys(self):                       # 최근 → 오래된 순서
        out, n = [], self.head.next
        while n is not self.tail:
            out.append(n.key); n = n.next
        return out

c = LRUCache(3)
c.put(1, 'A'); c.put(2, 'B'); c.put(3, 'C')
print(c.keys())                         # [3, 2, 1]
print(c.get(1), c.keys())               # A [1, 3, 2]
print(c.get(9))                          # None
c.put(4, 'D')
print(c.keys(), 2 in c.map)              # [4, 1, 3] False
c.put(3, 'E')
print(c.keys(), c.get(3))               # [3, 4, 1] E

map은 키로 노드를 O(1)에 찾게 해 주고, 이중 연결 리스트는 노드를 O(1)에 떼어 맨 앞에 붙이게 해 줍니다. head와 tail은 값이 없는 더미 노드라서 리스트가 비어 있거나 노드가 하나일 때도 앞뒤 포인터를 같은 식으로 고칠 수 있습니다. get은 노드를 찾아 맨 앞으로 옮기고 값을 돌려줍니다. put은 이미 있는 키면 값을 바꾸고 맨 앞으로 옮기고, 새 키인데 가득 찼으면 tail 바로 앞 노드(가장 오래 쓰지 않은 항목)를 리스트와 map에서 지운 뒤 새 노드를 맨 앞에 넣습니다. 시각화의 세 탭이 보여 주는 순서와 같습니다. 1, 2, 3을 넣고 1을 조회하면 순서가 1, 3, 2가 되고, 키 4를 넣을 때 맨 뒤인 2가 제거됩니다. 가장 먼저 들어온 1이 아니라 가장 오래 쓰지 않은 2가 제거되는 것이 FIFO와 다른 점입니다.

Python — OrderedDict와 functools.lru_cache

from collections import OrderedDict

class LRU2:
    def __init__(self, cap):
        self.cap, self.d = cap, OrderedDict()

    def get(self, k):
        if k not in self.d: return None
        self.d.move_to_end(k)               # 사용한 키를 맨 뒤(최근)로
        return self.d[k]

    def put(self, k, v):
        self.d[k] = v
        self.d.move_to_end(k)
        if len(self.d) > self.cap:
            self.d.popitem(last=False)      # 맨 앞(가장 오래된) 항목 제거

c2 = LRU2(3)
for k, v in [(1, 'A'), (2, 'B'), (3, 'C')]: c2.put(k, v)
c2.get(1)
c2.put(4, 'D')
print(list(c2.d))                      # [3, 1, 4] (오래된 → 최근)

from functools import lru_cache

@lru_cache(maxsize=2)
def sq(n): return n * n

sq(1); sq(2); sq(1); sq(3)
print(sq.cache_info())              # CacheInfo(hits=1, misses=3, maxsize=2, currsize=2)

파이썬의 OrderedDict는 키의 순서를 기억하고 move_to_end와 popitem(last=False)를 O(1)에 지원해서, 위 코드처럼 해시맵과 순서 리스트를 한 객체로 대신합니다. 여기서는 뒤쪽이 최근이고 앞쪽이 가장 오래된 쪽입니다. 시각화와 방향만 반대이고 같은 동작입니다. 표준 라이브러리의 functools.lru_cache는 함수 호출 결과를 같은 정책으로 캐시하는 데코레이터입니다. maxsize=2로 sq(1), sq(2), sq(1), sq(3)을 호출하면 두 번째 sq(1)만 적중(hit)하고 나머지 3번은 미스이며, 세 번째 서로 다른 값이 들어올 때 가장 오래 쓰지 않은 sq(2)가 밀려납니다.

Java — LinkedHashMap으로 LRU 캐시

import java.util.LinkedHashMap;
import java.util.Map;

class LRUCache<K, V> extends LinkedHashMap<K, V> {
    private final int capacity;

    LRUCache(int capacity) {
        super(16, 0.75f, true);          // true: 접근 순서(access-order)로 유지
        this.capacity = capacity;
    }

    @Override
    protected boolean removeEldestEntry(Map.Entry<K, V> eldest) {
        return size() > capacity;           // 넘치면 가장 오래된 항목을 제거
    }
}

LRUCache<Integer, String> cache = new LRUCache<>(3);
cache.put(1, "A"); cache.put(2, "B"); cache.put(3, "C");
cache.get(1);
cache.put(4, "D");
System.out.println(cache.keySet());          // [3, 1, 4]
System.out.println(cache.containsKey(2));    // false

자바의 LinkedHashMap은 해시 테이블과 이중 연결 리스트를 함께 가진 맵이고, 생성자의 세 번째 인자를 true로 주면 항목을 조회하거나 넣을 때마다 그 항목을 순서의 맨 끝으로 옮기는 접근 순서 모드가 됩니다. 앞의 직접 구현이 하는 일을 라이브러리가 대신하는 셈입니다. put이 새 항목을 넣은 직후 호출되는 removeEldestEntry가 true를 돌려주면 가장 오래된 항목이 제거되므로, size가 capacity를 넘는지만 검사하면 용량 제한이 완성됩니다. 1, 2, 3을 넣고 1을 조회한 뒤 4를 넣으면 2가 제거되어 키 순서가 3, 1, 4가 됩니다. 이 클래스는 스레드에 안전하지 않습니다. 여러 스레드가 함께 쓰려면 바깥에서 동기화하거나 Caffeine 같은 동시성 캐시 라이브러리를 씁니다.
