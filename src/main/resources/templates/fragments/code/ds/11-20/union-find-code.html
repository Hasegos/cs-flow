Python — 경로 압축과 크기 기준 합치기

class UnionFind:
    def __init__(self, n):
        self.parent = list(range(n))      # 처음에는 모두 자기가 루트
        self.size = [1] * n
        self.count = n                        # 집합의 개수

    def find(self, x):
        root = x
        while self.parent[root] != root:
            root = self.parent[root]
        while self.parent[x] != root:        # 경로 압축: 지나온 노드를 루트에 직접 연결
            self.parent[x], x = root, self.parent[x]
        return root

    def union(self, a, b):
        ra, rb = self.find(a), self.find(b)
        if ra == rb:
            return False                   # 이미 같은 집합
        if self.size[ra] < self.size[rb]:
            ra, rb = rb, ra                   # 큰 쪽이 루트가 되도록
        self.parent[rb] = ra
        self.size[ra] += self.size[rb]
        self.count -= 1
        return True

uf = UnionFind(6)
print(uf.union(0, 1), uf.union(1, 2), uf.union(0, 2))   # True True False
print(uf.find(2) == uf.find(0), uf.find(3) == uf.find(0), uf.count)   # True False 4

parent 배열이 숲을 표현합니다. parent[i] == i이면 i가 루트이고, 아니면 i의 부모가 parent[i]입니다. find는 루트까지 올라간 뒤, 한 번 더 올라가며 지나온 노드의 부모를 루트로 바꿔 경로 압축을 합니다. union은 두 원소의 루트를 찾아, 이미 같으면 아무것도 하지 않고 False를 돌려줍니다. 다르면 크기가 작은 쪽 루트를 큰 쪽 루트 밑에 붙이고 크기를 더합니다. 이 반환값이 크루스칼 알고리즘에서 사이클 판별에 그대로 쓰입니다. 0, 1, 2를 합치면 세 번째 union(0, 2)는 이미 같은 집합이라 False입니다. 원소 6개에서 두 번 합쳤으므로 집합은 4개가 남고, 3은 아직 0과 다른 집합입니다.

Python — 합치기 방식에 따른 트리 높이

def height(parent):
    def depth(x):
        d = 0
        while parent[x] != x:
            x = parent[x]; d += 1
        return d
    return max(depth(i) for i in range(len(parent)))

def build(n, by_size):
    parent = list(range(n)); size = [1] * n
    def root(x):                          # 경로 압축 없는 find
        while parent[x] != x: x = parent[x]
        return x
    for i in range(n - 1):                # union(0,1), union(1,2), ... 순서
        ra, rb = root(i), root(i + 1)
        if by_size:
            if size[ra] < size[rb]: ra, rb = rb, ra
            parent[rb] = ra; size[ra] += size[rb]
        else:
            parent[ra] = rb                # 첫 루트를 둘째 루트 밑에
    return height(parent)

print(build(6, False), build(6, True))         # 5 1
print(build(1000, False), build(1000, True))   # 999 1

같은 순서의 union을 두 방식으로 실행해 트리 높이를 비교합니다. 첫 루트를 둘째 루트 밑에 붙이는 기본 방식은 한 줄로 이어져 원소 n개에서 높이가 n - 1까지 자랍니다. 시각화의 기본 합치기 탭이 보여 주는 높이 5와 같습니다. 크기 기준은 이 순서에서 큰 트리에 작은 트리를 계속 붙이므로 높이가 1에 머뭅니다. 원소가 1000개여도 999와 1의 차이가 납니다. 이 순서는 기본 방식에 가장 불리한 입력이어서 차이가 큽니다. 크기 기준의 높이는 어떤 순서에서도 log2(n) 이하라는 점이 보장입니다.

Python — 크루스칼 최소 신장 트리

def kruskal(n, edges):
    uf = UnionFind(n)
    total, used = 0, []
    for w, a, b in sorted(edges):         # 가중치가 작은 간선부터
        if uf.union(a, b):             # 서로 다른 집합일 때만 채택
            total += w
            used.append((a, b, w))
    return total, used

# (가중치, 정점, 정점)
edges = [(4, 0, 1), (8, 0, 2), (2, 1, 2), (6, 1, 3), (3, 2, 3), (5, 2, 4), (7, 3, 4)]
print(kruskal(5, edges))   # (14, [(1, 2, 2), (2, 3, 3), (0, 1, 4), (2, 4, 5)])

크루스칼 알고리즘은 간선을 가중치 순으로 훑으며, 두 끝점이 서로 다른 집합일 때만 간선을 고르고 두 집합을 합칩니다. 같은 집합이면 이미 연결된 두 정점을 또 잇는 것이어서 사이클이 생기므로 건너뜁니다. 이 판별과 합치기를 union 한 번이 모두 해 줍니다. 예에서 가중치 2, 3, 4, 5인 간선 4개가 채택되어 합이 14입니다. 가중치 6, 7, 8인 간선은 끝점이 이미 같은 집합이라 버려집니다. 연결 그래프이면 정점 V개에서 간선 V - 1개를 채택하고 끝납니다. 간선 정렬이 O(E log E)이고 union은 거의 상수 시간이어서 전체 시간은 정렬이 좌우합니다. 최소 신장 트리 자체는 알고리즘 과목의 MST 토픽에서 자세히 다룹니다.
