Python — 그리디 색칠과 정점 순서

graph = {
    'A': ['B', 'C'],
    'B': ['A', 'C', 'D', 'F'],
    'C': ['A', 'B', 'E', 'F'],
    'D': ['B'],
    'E': ['C', 'F'],
    'F': ['B', 'C', 'E'],
}

def greedy(graph, order):
    color = {}
    for v in order:
        used = {color[u] for u in graph[v] if u in color}   # 이웃이 쓰는 색
        c = 1
        while c in used:
            c += 1
        color[v] = c                                          # 남은 가장 작은 번호
    return color

c1 = greedy(graph, 'ABCDEF')
print(c1, max(c1.values()))
# {'A': 1, 'B': 2, 'C': 3, 'D': 1, 'E': 1, 'F': 4} 4

order = sorted(graph, key=lambda v: -len(graph[v]))   # 이웃이 많은 정점부터
c2 = greedy(graph, order)
print(order, c2, max(c2.values()))
# ['B', 'C', 'F', 'A', 'E', 'D'] {'B': 1, 'C': 2, 'F': 3, 'A': 3, 'E': 1, 'D': 2} 3

그리디 색칠은 정점을 정해진 순서로 보면서, 이미 칠해진 이웃이 쓰는 색을 모은 뒤 그 집합에 없는 가장 작은 번호의 색을 칠합니다. 이웃 하나가 막는 색은 하나뿐이므로 정점의 이웃이 d개면 색 d + 1개 안에서 항상 칠할 수 있고, 그래서 그리디는 최대 차수 + 1개 이하의 색을 씁니다. 이 그래프의 최대 차수는 4(B와 C)라서 어떤 순서로도 색 5개를 넘지 않습니다. 같은 그래프라도 순서에 따라 결과가 달라집니다. 알파벳 순서 A, B, C, D, E, F로 칠하면 F의 이웃 B, C, E가 색 2, 3, 1을 쓰고 있어 색 4가 필요해 색 4개가 됩니다. 이웃이 많은 정점부터 B, C, F, A, E, D로 칠하면 색 3개로 끝납니다. A, B, C가 서로 이웃이라 어떤 방법으로도 색 3개 미만으로는 칠할 수 없으므로 이 결과가 최소입니다. 시각화의 그리디 색칠 탭과 정점 순서의 영향 탭이 보여 주는 결과와 같습니다.

Python — 백트래킹으로 m가지 색으로 칠할 수 있는지 판단하기

def can_color(graph, m, order):
    color = {}
    tries = 0

    def solve(i):
        nonlocal tries
        if i == len(order):
            return True
        v = order[i]
        for c in range(1, m + 1):
            tries += 1
            if all(color.get(u) != c for u in graph[v]):   # 이웃과 충돌하지 않으면
                color[v] = c
                if solve(i + 1):
                    return True
                del color[v]                              # 막히면 되돌아간다
        return False

    ok = solve(0)
    return ok, color, tries

print(can_color(graph, 2, 'ABCDEF'))   # (False, {}, 10)
print(can_color(graph, 3, 'ABCDEF'))   # (True, {'A': 1, 'B': 2, 'C': 3, 'D': 1, 'E': 2, 'F': 1}, 13)

백트래킹은 정점을 순서대로 보며 색 1부터 m까지 차례로 시도합니다. 칠해진 이웃과 충돌하지 않으면 그 색을 배정하고 다음 정점으로 넘어가고, 다음 정점에서 어떤 색도 쓸 수 없으면 현재 정점의 색을 지우고(del) 다음 색을 시도합니다. 이것이 되돌아가기입니다. 색 3개로는 A부터 F까지 13번 시도해서 칠할 수 있습니다. 중간에 E에 색 1을 칠하면 F가 쓸 색이 없어 E로 되돌아가 색 2로 바꾸는 한 번의 되돌아가기가 일어납니다. 시각화의 백트래킹 탭이 보여 주는 시도 횟수 13번, 되돌아간 횟수 1번과 같습니다. 색 2개로는 A, B, C가 서로 이웃인 삼각형이 있어 칠할 수 없고, 모든 경우를 따져 본 뒤 10번의 시도 끝에 False를 돌려줍니다. 최악에는 정점 수 V에 대해 m^V가지 배정을 모두 살펴봐야 하므로 정점이 많아지면 느려집니다.

Python — 시험 시간표 배정과 2색 판별

conflicts = {                                         # 같은 학생이 함께 듣는 과목
    '수학': ['물리', '화학'],
    '물리': ['수학', '화학', '영어'],
    '화학': ['수학', '물리'],
    '영어': ['물리', '국어'],
    '국어': ['영어'],
}
slots = greedy(conflicts, sorted(conflicts, key=lambda v: -len(conflicts[v])))
by = {}
for k, s in slots.items():
    by.setdefault(s, []).append(k)
print(by)                  # {1: ['물리', '국어'], 2: ['수학', '영어'], 3: ['화학']}

from collections import deque

def two_colorable(graph):
    color = {}
    for s in graph:
        if s in color:
            continue
        color[s] = 0
        q = deque([s])
        while q:
            v = q.popleft()
            for u in graph[v]:
                if u not in color:
                    color[u] = 1 - color[v]       # 이웃은 반대 색
                    q.append(u)
                elif color[u] == color[v]:
                    return False
    return True

square = {'a': ['b', 'd'], 'b': ['a', 'c'], 'c': ['b', 'd'], 'd': ['c', 'a']}
tri = {'a': ['b', 'c'], 'b': ['a', 'c'], 'c': ['a', 'b']}
print(two_colorable(square), two_colorable(tri), two_colorable(graph))   # True False False

시험 시간표는 그래프 색칠의 대표적인 응용입니다. 같은 학생이 함께 듣는 과목을 간선으로 이으면, 색은 시험 시간대가 되고 같은 색의 과목은 겹치는 학생이 없어 같은 시간대에 칠 수 있습니다. 다섯 과목은 그리디로 시간대 3개에 배정됩니다. 수학, 물리, 화학이 서로 겹쳐 시간대가 3개 미만일 수는 없으므로 3개가 최소입니다. 색을 두 개만 쓸 수 있는지 판별하는 문제는 너비 우선 탐색으로 선형 시간에 풀립니다. 시작 정점에 색 0을 주고 이웃에게 반대 색을 주면서 퍼져 나가다 이웃과 같은 색이 나오면 불가능입니다. 네 정점이 사각형으로 이어진 그래프는 True, 삼각형은 False이고, 위의 그래프도 삼각형을 포함해 False입니다. 색 3개 이상으로 칠할 수 있는지는 일반 그래프에서 NP-완전 문제라서 이런 빠른 방법이 알려져 있지 않습니다. 그래서 위의 그리디나 백트래킹처럼 실용적인 방법을 씁니다.
