Java — 버블 정렬 (Bubble Sort)

/* ── 버블 정렬: 인접한 두 원소를 비교해 큰 값을 뒤로 보냄 ─── */
public void bubbleSort(int[] arr) {
    int n = arr.length;
    for (int i = 0; i < n - 1; i++) {
        boolean swapped = false;
        for (int j = 0; j < n - 1 - i; j++) {
            if (arr[j] > arr[j + 1]) {
                int tmp = arr[j];
                arr[j] = arr[j + 1];
                arr[j + 1] = tmp;
                swapped = true;
            }
        }
        // 이번 패스에서 교환이 없었다면 이미 정렬 완료 → 조기 종료
        if (!swapped) break;
    }
}

매 패스마다 가장 큰 값이 "거품처럼" 맨 뒤로 떠오른다고 해서 버블 정렬이라 부릅니다.
swapped 플래그로 조기 종료를 구현하면 최선의 경우(이미 정렬된 배열) O(n)까지 빨라지지만, 평균·최악은 O(n²)입니다.
추가 배열 없이 원본 배열 안에서 교환만 하므로 공간복잡도는 O(1)입니다.

Java — 병합 정렬 (Merge Sort)

/* ── 병합 정렬: 절반으로 분할 후, 정렬된 두 부분을 병합 ──── */
public void mergeSort(int[] arr, int lo, int hi) {
    if (lo >= hi) return;              // 원소가 1개면 이미 정렬된 상태

    int mid = (lo + hi) / 2;
    mergeSort(arr, lo, mid);          // 왼쪽 절반 정렬
    mergeSort(arr, mid + 1, hi);     // 오른쪽 절반 정렬
    merge(arr, lo, mid, hi);         // 정렬된 두 부분을 병합
}

private void merge(int[] arr, int lo, int mid, int hi) {
    int[] left  = Arrays.copyOfRange(arr, lo, mid + 1);
    int[] right = Arrays.copyOfRange(arr, mid + 1, hi + 1);
    int i = 0, j = 0, k = lo;

    while (i < left.length && j < right.length) {
        arr[k++] = (left[i] <= right[j]) ? left[i++] : right[j++];
    }
    while (i < left.length)  arr[k++] = left[i++];   // 왼쪽 나머지 복사
    while (j < right.length) arr[k++] = right[j++];  // 오른쪽 나머지 복사
}

left[i] <= right[j] 조건(왼쪽을 우선)이 같은 값의 상대적 순서를 유지시켜 병합 정렬을 안정 정렬로 만듭니다.
분할은 항상 정확히 절반씩 이루어지므로 재귀 깊이가 O(log n)이고, 각 깊이에서 병합에 O(n)이 들어 전체 시간복잡도는 최선·평균·최악 모두 O(n log n)입니다.
단, left·right 임시 배열 때문에 공간복잡도는 O(n)입니다.

Java — 퀵 정렬 (Quick Sort, Lomuto 파티션)

/* ── 퀵 정렬: 피벗 기준으로 분할 후 각 부분을 재귀 정렬 ──── */
public void quickSort(int[] arr, int lo, int hi) {
    if (lo >= hi) return;

    int pivotIdx = partition(arr, lo, hi);
    quickSort(arr, lo, pivotIdx - 1);   // 피벗보다 작은 쪽
    quickSort(arr, pivotIdx + 1, hi);  // 피벗보다 큰 쪽
}

/* ── Lomuto 파티션: 마지막 원소를 피벗으로 사용 ──────────── */
private int partition(int[] arr, int lo, int hi) {
    int pivot = arr[hi];
    int i = lo - 1;               // "피벗보다 작은 영역"의 마지막 인덱스

    for (int j = lo; j < hi; j++) {
        if (arr[j] < pivot) {
            i++;
            int tmp = arr[i]; arr[i] = arr[j]; arr[j] = tmp;
        }
    }
    int tmp = arr[i + 1]; arr[i + 1] = arr[hi]; arr[hi] = tmp;  // 피벗을 경계로 이동
    return i + 1;                // 피벗의 최종 위치
}

파티션이 끝나면 피벗은 항상 최종 위치에 놓이고, 왼쪽은 모두 피벗보다 작은 값·오른쪽은 모두 큰 값이 됩니다 — 그래서 피벗은 이후 재귀 호출에서 제외됩니다.
평균적으로 피벗이 배열을 절반씩 나누면 O(n log n)이지만, 이미 정렬된 배열에서 마지막 원소를 피벗으로 고르면 매번 한쪽에 n-1개가 몰려 최악의 경우 O(n²)이 됩니다.
실전 라이브러리는 이를 피하기 위해 피벗을 무작위로 고르거나 세 값의 중앙값을 사용합니다.
