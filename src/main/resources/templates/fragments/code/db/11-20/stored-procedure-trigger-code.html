SQL — PostgreSQL의 프로시저와 트리거

CREATE PROCEDURE transfer(src int, dst int, amt int)
LANGUAGE plpgsql
AS $$
BEGIN
    IF (SELECT balance FROM account WHERE id = src) < amt THEN
        RAISE EXCEPTION '잔액 부족';
    END IF;
    UPDATE account SET balance = balance - amt WHERE id = src;
    UPDATE account SET balance = balance + amt WHERE id = dst;
END;
$$;

CALL transfer(1, 2, 100);                          -- 앱은 한 번만 호출

CREATE FUNCTION log_balance() RETURNS trigger
LANGUAGE plpgsql
AS $$
BEGIN
    INSERT INTO audit_log(account_id, old_balance, new_balance)
    VALUES (OLD.id, OLD.balance, NEW.balance);   -- OLD = 변경 전 행, NEW = 변경 후 행
    RETURN NEW;
END;
$$;

CREATE TRIGGER trg_balance_audit
AFTER UPDATE OF balance ON account
FOR EACH ROW EXECUTE FUNCTION log_balance();

PostgreSQL 11부터 CREATE PROCEDURE로 프로시저를 만들고 CALL로 호출합니다. transfer는 잔액을 확인하고 두 계좌를 수정하는 문장을 프로시저 안에 모아 두어, 앱은 CALL 한 번으로 이체를 실행합니다. 시각화의 저장 프로시저 탭이 보여 주는 동작입니다. 트리거는 트리거 함수를 만들고 CREATE TRIGGER로 테이블의 이벤트에 연결합니다. 여기서는 account의 UPDATE가 balance를 SET 목록에 포함하면(값이 같아도) 각 행마다 log_balance가 자동으로 실행되어 변경 전후 값을 audit_log에 남깁니다. 앱이 audit_log를 호출하는 코드는 없습니다. 프로시저 언어와 문법은 DB마다 다르며, 위 예는 PostgreSQL의 PL/pgSQL입니다. AFTER 행 단위 트리거의 반환 값은 무시되며 RETURN NEW는 관례입니다(BEFORE 트리거에서는 반환 값이 저장될 행을 정합니다). account와 audit_log 테이블이 이미 있다고 가정한 조각이고, transfer는 동시 이체를 막는 잠금을 넣지 않은 예시입니다.

Python — 왕복 횟수와 트리거 흉내

class Db:
    def __init__(self):
        self.bal = {1: 1000, 2: 500}
        self.audit = []
        self.trips = 0
        self.triggers = []

    def _update(self, acc, delta):
        old = self.bal[acc]
        self.bal[acc] += delta
        for t in self.triggers:                   # 이벤트가 나면 트리거가 자동 실행
            t(acc, old, self.bal[acc])

    def run(self, sql, *args):                    # 앱이 보내는 문장 하나 = 왕복 1번
        self.trips += 1
        if sql == 'select':
            return self.bal[args[0]]
        self._update(*args)

    def call_transfer(self, src, dst, amt):       # 프로시저 호출 = 왕복 1번
        self.trips += 1
        if self.bal[src] < amt:
            raise ValueError('잔액 부족')
        self._update(src, -amt)                   # 나머지는 DB 안에서 처리
        self._update(dst, amt)

db = Db()
db.triggers.append(lambda acc, old, new: db.audit.append((acc, old, new)))

if db.run('select', 1) >= 100:                    # 프로시저 없이: 문장마다 왕복
    db.run('update', 1, -100)
    db.run('update', 2, 100)
print('앱이 하나씩:', db.trips, db.bal)

db.trips = 0
db.call_transfer(1, 2, 100)                       # 프로시저: 한 번 호출
print('프로시저:', db.trips, db.bal)
print('트리거가 남긴 이력:', len(db.audit), db.audit[0])
# 앱이 하나씩: 3 {1: 900, 2: 600}
# 프로시저: 1 {1: 800, 2: 700}
# 트리거가 남긴 이력: 4 (1, 1000, 900)

앱이 문장을 하나씩 보내면 SELECT 1번과 UPDATE 2번으로 왕복이 3번이고, 프로시저를 호출하면 왕복이 1번입니다. 시각화의 저장 프로시저 탭과 같은 결과입니다. 두 번 모두 잔액이 100씩 옮겨져 최종 {1: 800, 2: 700}이 됩니다. _update는 값이 바뀔 때마다 등록된 트리거 함수를 자동으로 호출해 (계좌, 변경 전, 변경 후)를 audit에 남깁니다. 이체 두 번에 UPDATE가 모두 4번 일어났으므로 이력도 4건이고, 앱 코드 어디에도 audit에 쓰는 호출은 없습니다. 시각화의 트리거 탭과 같은 동작입니다. 장난감 모델이고 실제 DB의 트랜잭션과 오류 처리는 훨씬 복잡합니다.

Java — JDBC로 프로시저 호출

// 프로시저 없이: 문장마다 서버와 주고받는다
try (Connection conn = ds.getConnection()) {
    conn.setAutoCommit(false);
    int bal;
    try (PreparedStatement ps = conn.prepareStatement("SELECT balance FROM account WHERE id = ?")) {
        ps.setInt(1, 1);
        try (ResultSet rs = ps.executeQuery()) { rs.next(); bal = rs.getInt(1); }
    }
    if (bal >= 100) { /* UPDATE 두 번 */ }
    conn.commit();
}

// 프로시저: 한 번 호출 (PostgreSQL 11 이상의 CALL 문)
try (Connection conn = ds.getConnection();
     PreparedStatement ps = conn.prepareStatement("CALL transfer(?, ?, ?)")) {
    ps.setInt(1, 1);
    ps.setInt(2, 2);
    ps.setInt(3, 100);
    ps.execute();                                   // 잔액 부족이면 SQLException
}

위쪽은 앱이 SELECT와 UPDATE를 하나씩 보내며 잔액 판단을 자바 코드에서 합니다. 아래쪽은 CALL transfer 한 번으로 모든 처리를 DB에 맡기고, 프로시저가 RAISE EXCEPTION을 내면 JDBC에서는 SQLException으로 받습니다. 표준 JDBC에는 CallableStatement의 {call ...} 문법도 있지만 pgJDBC는 기본 설정에서 이를 함수 호출로 바꾸므로 프로시저에는 escapeSyntaxCallMode 설정이 필요합니다. 아래 prepareStatement 방식은 서버가 CALL 문을 받는다는 점에 기댄 예시이므로 사용하는 드라이버 문서를 확인하세요. 프로시저 안에 업무 규칙이 들어가면 그 규칙은 자바 코드가 아니라 DB에서 바뀌고 배포됩니다. 규칙을 한곳에 두는 장점이 있습니다.
