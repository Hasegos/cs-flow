SQL — PostgreSQL의 전문 검색

-- 텍스트를 tsvector로 바꾼다 (PostgreSQL 문서의 예)
SELECT to_tsvector('english', 'a fat cat sat on a mat - it ate a fat rats');
-- 'ate':9 'cat':3 'fat':2,11 'mat':7 'rat':12 'sat':4

-- tsquery로 검색: @@ 연산자 (결과 t)
SELECT to_tsvector('english', 'a fat cat sat on a mat - it ate a fat rats')
       @@ to_tsquery('english', 'fat & rat');

-- tsvector 생성 열과 GIN 인덱스
CREATE TABLE docs (
    id   int PRIMARY KEY,
    body text,
    tsv  tsvector GENERATED ALWAYS AS (to_tsvector('english', body)) STORED
);
CREATE INDEX docs_tsv_idx ON docs USING GIN (tsv);

SELECT id FROM docs WHERE tsv @@ to_tsquery('english', 'cat');

-- 관련도 순으로 정렬
SELECT id, ts_rank(tsv, query) AS rank
FROM docs, to_tsquery('english', 'fat & cat') AS query
WHERE tsv @@ query
ORDER BY rank DESC;

-- LIKE '%word%' 같은 부분 문자열 검색은 pg_trgm 확장의 인덱스
CREATE EXTENSION pg_trgm;
CREATE INDEX docs_body_trgm ON docs USING GIN (body gin_trgm_ops);

to_tsvector는 텍스트를 토큰으로 나누고 불용어를 빼고 어간을 추출해 tsvector로 바꿉니다. 첫 줄의 결과는 PostgreSQL 문서에 실린 예와 같은 형태이고, 위치 번호는 불용어를 뺀 뒤에도 원래 단어의 위치를 유지합니다. @@ 연산자는 tsvector가 tsquery와 일치하는지 판단하며, fat & rat은 두 단어가 모두 있어야 일치합니다. 시각화의 토큰과 정규화 탭이 보여 주는 순서입니다. tsvector를 생성 열에 저장하고 GIN 인덱스를 만들면 단어에서 문서를 빠르게 찾을 수 있습니다. GIN(Generalized Inverted Index)은 키마다 그 키가 든 행의 목록(posting list)을 저장하는 역색인 형태의 인덱스입니다. 생성 열은 PostgreSQL 12 이상에서 쓸 수 있습니다. ts_rank는 관련도 점수를 계산합니다. 부분 문자열 검색을 빠르게 하려면 pg_trgm 확장의 gin_trgm_ops 인덱스를 씁니다. 결과는 텍스트 검색 설정(여기서는 english)에 따라 달라집니다.

Python — 역색인 만들기와 LIKE 비교

docs = {1: 'fat cat sat', 2: 'fat rat ran', 3: 'cat ate rat', 4: 'big fat cat',
        5: 'dog ran fast', 6: 'big dog sat', 7: 'cat ran fast', 8: 'fast dog'}

index = {}
for doc_id, text in docs.items():
    for word in text.split():
        index.setdefault(word, []).append(doc_id)       # 단어 -> 문서 목록(posting list)

for word in sorted(index):
    print(word, index[word])

def search_and(*words):
    result = set(index.get(words[0], []))
    for w in words[1:]:
        result &= set(index.get(w, []))                  # 문서 목록의 교집합
    return sorted(result)

print('cat:', search_and('cat'))
print('fat & cat:', search_and('fat', 'cat'))

scanned = 0
like = []
for doc_id, text in docs.items():                        # LIKE '%cat%' : 문서를 하나씩 검사
    scanned += 1
    if 'cat' in text:
        like.append(doc_id)
print('LIKE:', like, '검사한 문서', scanned)
# ate [3]  big [4, 6]  cat [1, 3, 4, 7]  dog [5, 6, 8]  fast [5, 7, 8]
# fat [1, 2, 4]  ran [2, 5, 7]  rat [2, 3]  sat [1, 6]  (한 줄에 하나씩 출력)
# cat: [1, 3, 4, 7]
# fat & cat: [1, 4]
# LIKE: [1, 3, 4, 7] 검사한 문서 8

문서 8개로 만든 장난감 역색인입니다. 문서를 하나씩 읽으며 단어마다 문서 번호를 목록에 추가하고, 두 단어 검색은 두 목록의 교집합으로 계산합니다. cat은 D1, D3, D4, D7이고 fat & cat은 D1, D4입니다. 마지막 반복문은 별도 인덱스가 없는 LIKE처럼 문서를 하나씩 검사해 같은 결과를 얻지만 검사한 문서가 8개입니다. 시각화의 역색인 탭과 LIKE와 전문 검색 탭이 보여 주는 결과입니다. 실제 검색 엔진은 어간 추출, 단어 위치, 순위, 디스크 저장 형식까지 다루므로 훨씬 복잡합니다.

Java — TreeMap과 TreeSet으로 만든 역색인

Map<Integer, String> docs = new TreeMap<>();
docs.put(1, "fat cat sat"); docs.put(2, "fat rat ran");
docs.put(3, "cat ate rat"); docs.put(4, "big fat cat");
docs.put(5, "dog ran fast"); docs.put(6, "big dog sat");
docs.put(7, "cat ran fast"); docs.put(8, "fast dog");

Map<String, TreeSet<Integer>> index = new TreeMap<>();         // 단어는 정렬된 순서로 보관
for (Map.Entry<Integer, String> e : docs.entrySet())
    for (String w : e.getValue().split(" "))
        index.computeIfAbsent(w, k -> new TreeSet<>()).add(e.getKey());
System.out.println(index);

TreeSet<Integer> both = new TreeSet<>(index.get("fat"));
both.retainAll(index.get("cat"));                              // 문서 목록의 교집합
System.out.println("fat & cat: " + both);

int scanned = 0;
List<Integer> like = new ArrayList<>();
for (Map.Entry<Integer, String> e : docs.entrySet()) {                // LIKE '%cat%' : 문서를 하나씩 검사
    scanned++;
    if (e.getValue().contains("cat")) like.add(e.getKey());
}
System.out.println("LIKE: " + like + " 검사한 문서 " + scanned);
// {ate=[3], big=[4, 6], cat=[1, 3, 4, 7], dog=[5, 6, 8], fast=[5, 7, 8], fat=[1, 2, 4], ran=[2, 5, 7], rat=[2, 3], sat=[1, 6]}
// fat & cat: [1, 4]
// LIKE: [1, 3, 4, 7] 검사한 문서 8

같은 모델을 자바로 옮겼습니다. TreeMap은 단어를 정렬된 순서로 보관해 색인이 정렬된 단어 목록처럼 출력되고, TreeSet은 문서 번호를 정렬해 담아 retainAll로 교집합을 간단히 구합니다. fat & cat은 D1, D4입니다. 마지막 반복문은 LIKE처럼 모든 문서를 검사하는 방법으로, 검사한 문서가 8개입니다. 색인은 문서를 한 번 훑어 미리 만들어 두고 이후 검색은 색인에서 단어를 찾는 방식입니다. 개념 설명용 코드입니다.
