SQL — 인덱스 생성과 EXPLAIN으로 확인하기

-- email 컬럼에 인덱스 생성
CREATE INDEX idx_users_email ON users(email);

-- 인덱스 적용 전: type = ALL → 풀 스캔(이 시각화의 "풀 스캔" 탭과 동일)
EXPLAIN SELECT * FROM users WHERE email = 'minjun@mail.com';

-- 인덱스 적용 후: type = ref → 인덱스 탐색(이 시각화의 "인덱스 탐색" 탭과 동일)
EXPLAIN SELECT * FROM users WHERE email = 'minjun@mail.com';

EXPLAIN의 type 컬럼이 ALL이면 테이블 전체를 훑었다는 뜻이고, ref·range·const 등이면 인덱스를 탄 것입니다.
rows 컬럼(예상 스캔 행 수)도 함께 보면, 인덱스 적용 전후로 이 시각화의 "비교 횟수"에 해당하는 수치가 실제로 얼마나 줄어드는지 확인할 수 있습니다.

Java — JPA Entity에 인덱스 선언하기

@Entity
@Table(
    name = "users",
    indexes = {
        @Index(name = "idx_users_email", columnList = "email")
    }
)
public class User {

    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;              // PK는 기본적으로 이미 인덱스가 걸려 있음

    @Column(nullable = false, unique = true)
    private String email;         // 자주 WHERE에 쓰이므로 인덱스 후보
}

@Table(indexes = ...)는 엔티티 기준 DDL 자동 생성 시 인덱스를 함께 만들어 줍니다 — 이 프로젝트처럼 스키마를 직접 SQL로 관리한다면 위의 CREATE INDEX 문이 실제로 적용되는 부분입니다.
기본키(@Id)는 대부분의 DBMS에서 자동으로 인덱스가 생성되므로, 이 시각화의 B-Tree는 사실 id 같은 기본키 조회에도 이미 똑같이 적용되고 있는 구조입니다.

SQL — 복합 인덱스와 컬럼 순서(최左 접두 규칙)

-- (user_id, product) 순서로 복합 인덱스 생성
CREATE INDEX idx_orders_user_product ON orders(user_id, product);

-- OK: 인덱스를 탄다 (맨 앞 컬럼 user_id로 시작)
SELECT * FROM orders WHERE user_id = 1 AND product = '노트북';
SELECT * FROM orders WHERE user_id = 1;

-- 인덱스를 타지 못함: product만으로는 루트에서 방향을 정할 수 없음
SELECT * FROM orders WHERE product = '노트북';

이 시각화의 B-Tree 루트가 user_id 하나만 기준값으로 갖고 있듯, 복합 인덱스도 맨 앞 컬럼(user_id) 없이는 트리를 어느 쪽으로 내려가야 할지 정할 수 없습니다.
그래서 마지막 쿼리처럼 뒤쪽 컬럼(product)만으로 조회하면 이 인덱스를 전혀 활용하지 못하고 풀 스캔으로 돌아갑니다.
