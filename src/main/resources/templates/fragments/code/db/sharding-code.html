PostgreSQL — 선언적 파티셔닝(Hash / Range)

-- 해시 샤딩과 같은 원리: 키를 나머지 연산으로 파티션에 배정
CREATE TABLE users (
    user_id BIGINT,
    name TEXT
) PARTITION BY HASH (user_id);

CREATE TABLE users_p0 PARTITION OF users
    FOR VALUES WITH (MODULUS 3, REMAINDER 0);
CREATE TABLE users_p1 PARTITION OF users
    FOR VALUES WITH (MODULUS 3, REMAINDER 1);

-- 레인지 샤딩과 같은 원리: 값 구간으로 파티션에 배정
CREATE TABLE users_by_range (
    user_id BIGINT,
    name TEXT
) PARTITION BY RANGE (user_id);

CREATE TABLE users_range_a PARTITION OF users_by_range
    FOR VALUES FROM (0) TO (150);

PostgreSQL 선언적 파티셔닝은 이 시각화의 해시/레인지 샤딩과 정확히 같은 개념을 단일 서버 안의 여러 테이블(파티션)로 구현한 것입니다. 실제 다중 서버 샤딩은 여기에 "어느 서버에 어느 파티션이 있는가"를 라우팅하는 계층이 하나 더 필요합니다.

Java — 애플리케이션 레벨 샤드 라우팅(개념)

public class ShardRouter {

    private static final int SHARD_COUNT = 3;

    // 해시 샤딩: 키 % 샤드 개수
    public int resolveShard(long userId) {
        return (int) (userId % SHARD_COUNT);
    }

    // 이 시각화의 ROUTER가 하는 일이 바로 이 메서드입니다
    public DataSource dataSourceFor(long userId) {
        return shardDataSources[resolveShard(userId)];
    }
}

실제로는 DataSource 여러 개를 미리 만들어두고, 요청마다 샤드 키로 라우팅 대상을 계산해 해당 샤드의 커넥션을 사용합니다. 샤드 키가 없는 쿼리(예: 이메일로 사용자 찾기)는 모든 샤드를 다 조회해야 해서(scatter-gather) 느립니다 — 그래서 샤드 키 선택이 중요합니다.

SQL — 크로스 샤드 집계는 애플리케이션에서 합친다

-- 각 샤드에서 개별 실행 (DB가 자동으로 합쳐주지 않음)
SELECT COUNT(*) FROM users; -- Shard 1에서 실행 → 1200
SELECT COUNT(*) FROM users; -- Shard 2에서 실행 → 980
SELECT COUNT(*) FROM users; -- Shard 3에서 실행 → 1450

-- 애플리케이션에서 세 결과를 더해 총합을 계산: 1200 + 980 + 1450 = 3630

단일 서버였다면 COUNT(*) 한 번이면 끝나지만, 샤딩된 환경에서는 각 샤드에 같은 쿼리를 따로 보내고 결과를 애플리케이션에서 합쳐야 합니다. 이런 집계·조인이 잦다면 샤딩의 비용이 이점보다 커질 수 있습니다.
