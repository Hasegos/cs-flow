PostgreSQL — 스트리밍 복제 기본 설정

-- Primary: postgresql.conf
wal_level = replica
max_wal_senders = 10
synchronous_commit = on       -- 동기 복제 시 'remote_write' 이상으로 설정
synchronous_standby_names = ''   -- 동기 복제 사용 시 Replica 이름 지정

-- Replica: standby.signal 파일 생성 + primary_conninfo 설정
-- pg_basebackup으로 Primary 데이터를 복제한 뒤 Replica로 기동

synchronous_standby_names가 비어 있으면 비동기 복제입니다. 여기에 Replica 이름을 지정하면 Primary가 그 Replica의 ack을 기다리는 동기 복제로 바뀝니다 — 이 시각화의 두 모드 차이가 실제로는 설정 한 줄 차이입니다.

SQL — 복제 지연(Replication Lag) 확인하기

-- Primary에서 실행: 각 Replica가 얼마나 뒤처져 있는지 확인
SELECT client_addr, state,
       pg_wal_lsn_diff(pg_current_wal_lsn(), replay_lsn) AS lag_bytes
FROM pg_stat_replication;

-- lag_bytes가 계속 커진다면 Replica가 Primary를 못 따라가고 있다는 신호

pg_stat_replication은 PostgreSQL이 각 Replica의 복제 상태를 보여주는 시스템 뷰입니다. 이 시각화의 "뒤늦게 기록" 스텝이 실제 운영 환경에서는 이 lag_bytes 값으로 관측됩니다.

Java — 읽기 쿼리를 Replica로 분산하기(개념)

// 쓰기는 Primary, 읽기는 Replica로 — 데이터소스 라우팅 개념 예시
public class ReplicaRoutingDataSource extends AbstractRoutingDataSource {
    protected Object determineCurrentLookupKey() {
        return TransactionSynchronizationManager.isCurrentTransactionReadOnly()
                ? "replica" : "primary";
    }
}

// Service 쪽에서는 @Transactional(readOnly = true) 여부만 신경 쓰면 됨
@Transactional(readOnly = true)
public List<Topic> getPublishedTopicsPageable(...) { ... }  // → replica로 라우팅

readOnly = true인 트랜잭션은 Replica로, 그 외에는 Primary로 보내는 방식입니다. 다만 방금 쓴 데이터를 바로 읽어야 하는 화면(예: 글쓰기 직후 상세 페이지)은 복제 지연 때문에 안 보일 수 있어 Primary로 강제 라우팅하는 예외 처리가 흔히 필요합니다.
