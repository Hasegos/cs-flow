SQL — SELECT ... FOR UPDATE로 명시적 배타락 걸기

START TRANSACTION;

-- 이 행에 배타락(X)을 걸고 조회 — 다른 트랜잭션은 이 행을 잠금이 풀릴 때까지 못 건드림
SELECT * FROM accounts WHERE id = 1 FOR UPDATE;

UPDATE accounts SET balance = balance - 20000 WHERE id = 1;

COMMIT;

FOR UPDATE는 단순 조회가 아니라 "이 값을 곧 수정할 거니 아무도 건드리지 마라"는 의미입니다 — 이 시각화의 "락 대기" 탭에서 A가 김민준 행에 배타락을 거는 것과 같은 동작입니다.
이 구문이 없으면 SELECT는 보통 락을 걸지 않아서, 조회한 값을 기준으로 계산한 뒤 UPDATE하는 사이에 다른 트랜잭션이 값을 바꿔버릴 수 있습니다.

Java — JPA에서 비관적 락(Pessimistic Lock) 사용

public interface AccountRepository extends JpaRepository<Account, Long> {

    // FOR UPDATE와 동일한 배타락을 걸고 조회
    @Lock(LockModeType.PESSIMISTIC_WRITE)
    @Query("SELECT a FROM Account a WHERE a.id = :id")
    Optional<Account> findByIdForUpdate(@Param("id") Long id);
}

@Service
@RequiredArgsConstructor
public class TransferService {

    private final AccountRepository accountRepository;

    @Transactional
    public void withdraw(Long accountId, long amount) {
        Account account = accountRepository.findByIdForUpdate(accountId)
                .orElseThrow(() -> new NotFoundException("계좌 없음"));
        account.withdraw(amount);
    }
}

PESSIMISTIC_WRITE가 SQL의 FOR UPDATE로 변환됩니다. 메서드가 끝나(COMMIT) 락이 풀릴 때까지, 같은 행을 조회하는 다른 트랜잭션은 이 메서드가 끝날 때까지 대기합니다.

Java — 데드락 예외를 잡아 재시도하기

@Service
@RequiredArgsConstructor
public class TransferRetryService {

    private final TransferService transferService;

    public void transferWithRetry(Long fromId, Long toId, long amount) {
        int attempts = 0;
        while (true) {
            try {
                transferService.transfer(fromId, toId, amount);
                return;
            } catch (CannotAcquireLockException e) {
                attempts++;                                // DB가 데드락 희생자로 골라 롤백시킨 경우
                if (attempts >= 3) throw e;
                log.warn("데드락으로 롤백됨, 재시도 {}회", attempts);
            }
        }
    }
}

DB가 데드락의 희생자로 트랜잭션을 골라 강제 ROLLBACK시키면, Spring은 이를 CannotAcquireLockException으로 변환해 던집니다.
이 시각화의 "데드락" 탭에서 B가 희생자로 롤백되는 것처럼, 실제 애플리케이션에서도 한쪽 트랜잭션은 예외를 받게 되므로, 짧은 대기 후 재시도하는 로직을 흔히 함께 둡니다.
