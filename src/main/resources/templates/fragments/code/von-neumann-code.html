C — 명령어 패치·디코드·실행 시뮬레이션

// 폰 노이만 구조의 기본 동작 — Fetch → Decode → Execute

typedef struct {
int memory[256];   // 명령어 + 데이터 동일 메모리
int pc;           // Program Counter (다음 명령어 주소)
int acc;          // Accumulator (누산기)
} CPU;

void run(CPU *cpu) {
    while (1) {
        /* 1. FETCH — PC가 가리키는 메모리에서 명령어를 가져옴 */
        int instr = cpu->memory[cpu->pc++];

        /* 2. DECODE — 명령어 상위 비트로 연산 종류 판별 */
        int opcode  = (instr >> 8) & 0xFF;
        int operand = instr & 0xFF;

        /* 3. EXECUTE — 연산 수행 */
        if      (opcode == 0x01) cpu->acc  = cpu->memory[operand]; // LOAD
        else if (opcode == 0x02) cpu->memory[operand] = cpu->acc;  // STORE
        else if (opcode == 0x03) cpu->acc += cpu->memory[operand]; // ADD
        else if (opcode == 0xFF) break;                            // HALT
    }
}

PC(Program Counter)가 메모리 주소를 가리키고, 명령어를 꺼내(Fetch) → 해석하고(Decode) → 실행(Execute)하는 과정이 반복됩니다. 명령어와 데이터가 같은 memory[] 배열에 들어있는 것이 폰 노이만 구조의 핵심입니다.

Java — 폰 노이만 병목 현상 체감

// 메모리 접근 패턴에 따른 성능 차이 (캐시 미스 vs 히트)

int[] data = new int[1024 * 1024];

// 순차 접근 — 캐시 히트율 높음 (빠름)
for (int i = 0; i < data.length; i++) {
    data[i] *= 2;
}

// 랜덤 접근 — 캐시 미스 빈번 (느림, 폰 노이만 병목 체감)
Random rand = new Random();
for (int i = 0; i < data.length; i++) {
    data[rand.nextInt(data.length)] *= 2;
}

CPU는 연산 속도가 메모리 접근 속도보다 훨씬 빠릅니다. 순차 접근은 캐시가 미리 데이터를 가져오지만, 랜덤 접근은 매번 느린 메인 메모리까지 가야 합니다. 이것이 폰 노이만 병목의 실체입니다.
