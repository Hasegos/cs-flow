C — 가상 주소 변환 (페이지 테이블 조회)

// 페이지 크기: 4KB = 4096 bytes
#define PAGE_SIZE   4096
#define PAGE_BITS   12    // log2(4096) = 12
#define VPN_MASK    0xFFFFF000
#define OFFSET_MASK 0x00000FFF

typedef struct {
    unsigned int pfn   : 20;  // 물리 프레임 번호 (20비트)
    unsigned int valid : 1;   // 유효 비트 (물리 메모리에 있으면 1)
    unsigned int dirty : 1;   // 수정 여부 (Write-Back용)
    unsigned int ref   : 1;   // 참조 비트 (LRU 교체 판단용)
} PageEntry;

PageEntry page_table[1048576]; // 2^20개 엔트리

// 가상 주소 → 물리 주소 변환
long translate(unsigned int vaddr) {
    unsigned int vpn    = (vaddr & VPN_MASK)    >> PAGE_BITS;
    unsigned int offset =  vaddr & OFFSET_MASK;

    PageEntry *entry = &page_table[vpn];

    if (!entry->valid) {
        // Page Fault — OS가 처리
        handle_page_fault(vpn);
    }

    entry->ref = 1;  // 참조 비트 설정
    return (((long)entry->pfn) << PAGE_BITS) | offset;
}

// 물리 주소 = PFN << 12 | Offset
// 예: VPN=5, Offset=0x1A0 → PA = (5 << 12) | 0x1A0 = 0x501A0

가상 주소 32비트 중 상위 20비트가 VPN(페이지 번호), 하위 12비트가 Offset입니다. 페이지 테이블에서 VPN에 해당하는 엔트리를 찾아 PFN을 얻고, 물리 주소 = PFN × 4096 + Offset 으로 계산합니다.

Java — TLB 시뮬레이션 (LinkedHashMap 기반)

import java.util.LinkedHashMap;
import java.util.Map;

/**
 * TLB 시뮬레이터 — 최근 사용 기반(LRU) 교체
 * 용량: 16엔트리 (실제 TLB와 유사한 소용량)
 */
public class TLBSimulator {

    private static final int TLB_SIZE = 16;

    // accessOrder=true: get() 호출 시 해당 항목을 MRU로 이동 (LRU 자동 관리)
    private final Map<Integer, Integer> tlb = new LinkedHashMap<>(
        TLB_SIZE, 0.75f, true
    ) {
        protected boolean removeEldestEntry(Map.Entry<Integer,Integer> e) {
            return size() > TLB_SIZE; // 용량 초과 시 LRU 항목 자동 제거
        }
    };

    private int[] pageTable; // 페이지 테이블 (VPN → PFN)
    private int   tlbHits, tlbMisses;

    public TLBSimulator(int[] pageTable) {
        this.pageTable = pageTable;
    }

    /** 가상 주소 → 물리 주소 변환 */
    public int translate(int vaddr) {
        int vpn    = vaddr >> 12;
        int offset = vaddr & 0xFFF;

        // TLB 조회
        Integer pfn = tlb.get(vpn);

        if (pfn != null) {
            tlbHits++;
            System.out.printf("TLB HIT  VPN:%d → PFN:%d%n", vpn, pfn);
        } else {
            // TLB Miss — 페이지 테이블 조회 후 TLB 갱신
            tlbMisses++;
            pfn = pageTable[vpn];
            tlb.put(vpn, pfn); // TLB에 캐싱
            System.out.printf("TLB MISS VPN:%d → PFN:%d (PT 조회)%n", vpn, pfn);
        }

        return (pfn << 12) | offset; // 물리 주소
    }

    public void printStats() {
        int total = tlbHits + tlbMisses;
        System.out.printf("TLB 히트율: %.1f%%%n", tlbHits * 100.0 / total);
    }
}

TLB는 VPN → PFN 변환 결과를 캐싱해 페이지 테이블 조회 비용을 줄입니다. 히트 시 메모리 접근 1회, 미스 시 페이지 테이블(메모리) 접근 후 TLB 갱신으로 총 2회 접근합니다. LinkedHashMap의 accessOrder로 LRU 교체를 자동 구현합니다.
