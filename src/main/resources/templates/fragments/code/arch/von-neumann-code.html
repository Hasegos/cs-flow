C — 명령어 패치·디코드·실행 시뮬레이션

// 폰 노이만 구조의 기본 동작 — Fetch → Decode → Execute

typedef struct {
int memory[256];   // 명령어 + 데이터 동일 메모리
int pc;           // Program Counter (다음 명령어 주소)
int acc;          // Accumulator (누산기)
} CPU;

void run(CPU *cpu) {
    while (1) {
        /* 1. FETCH — PC가 가리키는 메모리에서 명령어를 가져옴 */
        int instr = cpu->memory[cpu->pc++];

        /* 2. DECODE — 명령어 상위 비트로 연산 종류 판별 */
        int opcode  = (instr >> 8) & 0xFF;
        int operand = instr & 0xFF;

        /* 3. EXECUTE — 연산 수행 */
        if      (opcode == 0x01) cpu->acc  = cpu->memory[operand]; // LOAD
        else if (opcode == 0x02) cpu->memory[operand] = cpu->acc;  // STORE
        else if (opcode == 0x03) cpu->acc += cpu->memory[operand]; // ADD
        else if (opcode == 0xFF) break;                            // HALT
    }
}

PC(Program Counter)가 메모리 주소를 가리키고, 명령어를 꺼내(Fetch) → 해석하고(Decode) → 실행(Execute)하는 과정이 반복됩니다. 명령어와 데이터가 같은 memory[] 배열에 들어있는 것이 폰 노이만 구조의 핵심입니다.

Java — CPU-메모리 대역폭 병목 (폰 노이만 병목) 체감

// 폰 노이만 병목(Von Neumann Bottleneck): CPU와 메모리를 잇는 단일 버스의
// 처리량(bandwidth)이 CPU 연산 속도를 따라가지 못하는 현상.
// — John Backus, 1978 ACM Turing Award 강연에서 정의
//
// 아래 코드는 메모리 계층(캐시) 효과를 통해 병목을 간접 체감합니다.
// 캐시 미스가 잦을수록 CPU가 메인 메모리를 기다리는 시간이 늘어나
// CPU-메모리 간 대역폭 병목이 전체 성능을 제한하는 상황이 재현됩니다.

int[] data = new int[1024 * 1024];

// 순차 접근 — 공간 지역성 높음 → 캐시 히트율 높음 → 버스 대역폭 부담 적음 (빠름)
for (int i = 0; i < data.length; i++) {
    data[i] *= 2;
}

// 랜덤 접근 — 공간 지역성 없음 → 캐시 미스 빈발 → 매 연산마다 메인 메모리 접근
// → CPU가 버스를 통해 메모리를 기다리는 상태 반복 = 폰 노이만 병목 간접 체감
Random rand = new Random();
for (int i = 0; i < data.length; i++) {
    data[rand.nextInt(data.length)] *= 2;
}

폰 노이만 병목은 Backus(1978 ACM Turing Award)가 정의한 개념으로, CPU와 메모리를 잇는 단일 버스의 처리량이 CPU 연산 속도보다 훨씬 낮아 CPU가 데이터를 기다리며 낭비되는 현상을 가리킵니다. 이 코드는 메모리 계층(캐시) 효과를 통해 병목을 간접적으로 체감합니다. 순차 접근은 캐시가 데이터를 미리 가져와 버스 부하가 낮지만, 랜덤 접근은 캐시 미스가 빈발해 CPU가 매번 버스를 통해 메인 메모리를 기다리는 시간이 길어집니다. 캐시 미스 자체는 메모리 계층의 문제이며, 폰 노이만 병목(단일 버스 대역폭 한계)과 같은 개념은 아니지만 병목이 실제 성능에 미치는 영향을 체감하기 위한 예시입니다.
