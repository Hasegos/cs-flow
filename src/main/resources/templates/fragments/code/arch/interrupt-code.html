C — 인터럽트 벡터 테이블 등록 및 ISR 구현

// 인터럽트 서비스 루틴 (ISR) 타입 정의
typedef void (*isr_t)(void);

// x86 보호 모드에서는 IDT(Interrupt Descriptor Table)를 사용한다.
// IVT(Interrupt Vector Table)는 x86 실제 모드(Real Mode)의 구조이며,
// 보호 모드부터는 IDT가 IVT를 대체한다. 아래는 개념 설명용 단순화 구현이다.
#define IDT_SIZE 256
isr_t idt[IDT_SIZE];

// ISR 등록 함수
void register_isr(int irq_num, isr_t handler) {
    idt[irq_num] = handler;
}

// 키보드 인터럽트 핸들러 (IRQ1)
void keyboard_isr(void) {
    /* 1. 키보드 포트에서 스캔 코드 읽기 */
    uint8_t scancode = inb(0x60);

    /* 2. 스캔 코드를 ASCII로 변환해 버퍼에 저장 */
    key_buffer[buf_head] = scancode_to_ascii(scancode);
    buf_head = (buf_head + 1) % KEY_BUF_SIZE;

    /* 3. PIC에 EOI(End of Interrupt) 신호 전송 */
    outb(0x20, 0x20);

    /* IRET — ISR 종료 시 CPU가 자동으로 이전 컨텍스트 복원 */
}

// 타이머 인터럽트 핸들러 (IRQ0)
void timer_isr(void) {
    tick_count++;                   // 시스템 틱 증가
    schedule();                     // 프로세스 스케줄러 호출
    outb(0x20, 0x20);             // EOI 전송
}

// 초기화: IRQ0(타이머), IRQ1(키보드) 등록
void init_interrupts(void) {
    register_isr(0, timer_isr);    // IRQ0 → 타이머
    register_isr(1, keyboard_isr); // IRQ1 → 키보드
    sti();                          // 인터럽트 활성화 (IF 플래그 SET)
}

x86 보호 모드에서 인터럽트 디스패치는 IDT(Interrupt Descriptor Table)를 사용합니다. IRQ 번호를 인덱스로 ISR 함수 포인터(게이트 디스크립터)를 저장하며, 인터럽트 발생 시 CPU는 IDTR 레지스터가 가리키는 IDT를 참조해 해당 ISR로 점프하고, IRET 명령어로 이전 컨텍스트를 복원합니다.

C — 컨텍스트 저장 및 복원 (어셈블리 수준)

/* 인터럽트 발생 시 CPU가 자동으로 수행하는 동작 (x86 보호 모드 기준) */

// ── 인터럽트 발생 시 (하드웨어 자동) ──────────────────────
// 1. IF(Interrupt Flag) 클리어 → 추가 인터럽트 차단
// 2. 스택에 컨텍스트 PUSH:
//    - 유저 모드 → 커널 모드: SS, ESP, EFLAGS, CS, EIP 순으로 PUSH
//    - 커널 모드 → 커널 모드: 스택 전환 없음, EFLAGS, CS, EIP만 PUSH
// 3. IDT(Interrupt Descriptor Table)에서 ISR 주소 로드
// 4. ISR로 점프 (EIP = ISR 주소)

// ── ISR 진입부 (어셈블리로 작성) ──────────────────────────
__asm__(
    "keyboard_isr_entry:\n"
    "  pusha\n"          // EAX,ECX,EDX,EBX,ESP,EBP,ESI,EDI 저장
    "  call keyboard_isr\n" // C ISR 함수 호출
    "  popa\n"           // 레지스터 복원
    "  iret\n"           // EIP, CS, EFLAGS 복원 + IF 재활성화
);

// ── IRET 명령이 복원하는 것 ────────────────────────────────
typedef struct {
    uint32_t eip;     // 중단된 명령어 주소 → 여기서 재개
    uint32_t cs;      // 코드 세그먼트
    uint32_t eflags;  // 플래그 레지스터 (IF 포함)
} InterruptFrame;

// ── 인터럽트 마스킹 예시 ──────────────────────────────────
void critical_section(void) {
    cli();   // 인터럽트 비활성화 (IF = 0)

    /* 원자적으로 실행해야 하는 코드 */
    shared_resource++;

    sti();   // 인터럽트 재활성화 (IF = 1)
}

iret는 인터럽트 복귀 전용 명령어입니다. 스택에서 EIP, CS, EFLAGS를 한 번에 복원하고 인터럽트 플래그(IF)를 재활성화합니다. cli/sti로 임계 구역을 보호하면 인터럽트에 의한 데이터 경합을 방지할 수 있습니다.

Java — 소프트웨어 인터럽트 (시스템 콜 시뮬레이션)

/**
 * 소프트웨어 인터럽트(트랩) 개념 시뮬레이션
 * Java에서 직접 시스템 콜을 발생시키진 않지만,
 * JVM이 내부적으로 OS 시스템 콜을 트랩으로 처리하는 흐름을 표현
 */
public class SyscallSimulator {

    // 시스템 콜 번호 정의
    static final int SYS_READ  = 0;
    static final int SYS_WRITE = 1;
    static final int SYS_EXIT  = 60;

    // 시스템 콜 핸들러 테이블 (IVT와 동일한 개념)
    static final Runnable[] syscallTable = {
        () -> System.out.println("[KERNEL] SYS_READ 처리"),   // 0
        () -> System.out.println("[KERNEL] SYS_WRITE 처리"),  // 1
    };

    /**
     * 소프트웨어 인터럽트 발생 시뮬레이션
     * 실제 x86-32: INT 0x80 명령 → 커널 모드 전환 → 시스템 콜 처리
     * (INT 0x80은 레거시 방식. 현재 x86-64는 syscall 명령어,
     *  x86-32 빠른 경로는 sysenter 명령어를 사용한다.)
     */
    static void syscall(int callNumber) {
        System.out.println("[USER]   INT 0x80 발생 → 커널 모드 전환");
        System.out.println("[CPU]    컨텍스트 저장 (PC, 레지스터 → 스택)");

        if (callNumber >= 0 && callNumber < syscallTable.length) {
            syscallTable[callNumber].run(); // ISR 실행
        }

        System.out.println("[CPU]    컨텍스트 복원 (스택 → PC, 레지스터)");
        System.out.println("[USER]   사용자 모드 복귀\n");
    }

    public static void main(String[] args) {
        System.out.println("=== 사용자 프로그램 실행 중 ===");
        syscall(SYS_WRITE); // 소프트웨어 인터럽트 발생
        syscall(SYS_READ);
        System.out.println("=== 프로그램 계속 실행 ===");
    }
}

Java에서 System.out.println()을 호출하면 내부적으로 JVM이 OS의 write 시스템 콜을 발생시킵니다. 이 과정에서 소프트웨어 인터럽트(트랩)가 발생해 사용자 모드에서 커널 모드로 전환되고, 처리 후 다시 사용자 모드로 복귀합니다.
