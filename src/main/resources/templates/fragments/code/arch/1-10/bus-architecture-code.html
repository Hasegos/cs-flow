C — 버스 읽기/쓰기 시뮬레이션

// 버스 구조 시뮬레이션 — Address / Data / Control Bus
#include <stdint.h>
#include <stdbool.h>

// 메인 메모리 (256 바이트)
#define MEM_SIZE 256
static uint8_t memory[MEM_SIZE];

// 버스 레지스터 — 실제 하드웨어의 버스 라인을 모델링
typedef struct {
    uint8_t  address_bus;   // Address Bus — 단방향 (CPU → 장치)
    uint8_t  data_bus;      // Data Bus    — 양방향
    bool     read_signal;   // Control Bus — READ 신호
    bool     write_signal;  // Control Bus — WRITE 신호
    bool     bus_request;   // Control Bus — BR (Bus Request)
    bool     bus_grant;     // Control Bus — BG (Bus Grant)
} Bus;

static Bus bus = {0};

// 메모리 읽기: Address Bus → READ 신호 → Data Bus로 반환
uint8_t bus_read(uint8_t addr) {
    bus.address_bus  = addr;   // 1. Address Bus에 주소 전송
    bus.read_signal  = true;  // 2. Control Bus — READ 신호 활성화
    bus.data_bus     = memory[bus.address_bus]; // 3. 메모리 → Data Bus
    bus.read_signal  = false; // 4. 신호 해제
    return bus.data_bus;
}

// 메모리 쓰기: Address Bus → Data Bus에 값 적재 → WRITE 신호
void bus_write(uint8_t addr, uint8_t val) {
    bus.address_bus  = addr;   // 1. Address Bus에 주소 전송
    bus.data_bus     = val;    // 2. Data Bus에 데이터 적재
    bus.write_signal = true;  // 3. Control Bus — WRITE 신호 활성화
    memory[bus.address_bus] = bus.data_bus; // 4. 메모리에 기록
    bus.write_signal = false; // 5. 신호 해제
}

// 버스 중재 — I/O 장치가 버스 요청 후 승인받아 DMA 수행
void dma_transfer(uint8_t dst, uint8_t src_data) {
    bus.bus_request = true;  // I/O: Bus Request (BR) 신호 전송

    // Arbiter가 버스 승인 (현재 CPU 전송 완료 후)
    bus.bus_grant   = true;  // Arbiter: Bus Grant (BG) 신호 전송
    bus.bus_request = false;

    // I/O가 버스 마스터로서 메모리에 직접 쓰기 (DMA)
    bus.address_bus  = dst;
    bus.data_bus     = src_data;
    bus.write_signal = true;
    memory[dst]      = src_data;
    bus.write_signal = false;

    bus.bus_grant = false;    // 버스 반환
}

Address Bus는 단방향이므로 항상 CPU(또는 현재 Bus Master)가 주소를 올립니다. Data Bus는 양방향이며 Read 시에는 메모리→CPU, Write 시에는 CPU→메모리 방향으로 흐릅니다. DMA 전송 시에는 I/O 장치가 Arbiter로부터 버스 사용권을 받아 직접 메모리에 접근합니다.

C — 데이지 체인 버스 중재 구현

// 데이지 체인(Daisy Chain) 버스 중재
// Bus Grant 신호를 장치 체인으로 순서대로 전달
#define NUM_DEVICES 4

typedef struct {
    int  id;
    bool requesting;  // 버스 요청 중인지 여부
    bool granted;     // 버스 사용권 획득 여부
} Device;

Device devices[NUM_DEVICES] = {
    { 0, false, false },  // 우선순위 최고
    { 1, false, false },
    { 2, false, false },
    { 3, false, false },  // 우선순위 최저
};

// Arbiter: Bus Grant를 체인 순서대로 전달
// 요청 중인 첫 번째 장치가 버스를 획득
int daisy_chain_arbitrate() {
    bool grant_available = true;  // 초기에 Grant 신호 활성화

    for (int i = 0; i < NUM_DEVICES; i++) {
        if (!grant_available) break;

        if (devices[i].requesting) {
            devices[i].granted = true;   // 버스 사용권 부여
            grant_available    = false;   // Grant 신호 차단 (이후 장치는 획득 불가)
            return i;
        }
        // 요청 없으면 Grant 신호를 다음 장치로 통과
    }
    return -1;  // 요청 장치 없음
}

// 버스 사용 완료 후 반환
void release_bus(int device_id) {
    devices[device_id].granted    = false;
    devices[device_id].requesting = false;
}

데이지 체인 중재는 Bus Grant 신호를 장치 0번부터 순서대로 통과시킵니다. 요청 중인 장치가 신호를 받으면 그 자리에서 버스를 획득하고 이후 장치로의 신호를 차단합니다. 구조가 단순한 대신 앞 번호 장치가 지속적으로 요청하면 뒷 장치는 기아(Starvation) 상태가 될 수 있습니다.
