C — 메모리 계층별 접근 시간 시뮬레이션

// 메모리 계층별 접근 시간 (나노초 단위)
#define T_REGISTER  1      // 레지스터:  ~1 ns
#define T_L1        4      // L1 캐시:    4 ns
#define T_L2        12     // L2 캐시:   12 ns
#define T_L3        40     // L3 캐시:   40 ns
#define T_RAM       100    // RAM:      100 ns
#define T_SSD       100000 // SSD:  100,000 ns (100 μs)

// 히트율 기반 평균 접근 시간 계산 (표준 계층적 공식)
// AMAT = T_L1 + MissRate_L1 × (T_L2 + MissRate_L2 × (T_L3 + MissRate_L3 × T_RAM))
// L2 히트 시에도 L1 조회(미스) 비용을 함께 지불해야 하므로 누적 합산한다.
double calc_amat(double h_l1, double h_l2, double h_l3) {
    // L3 미스 시 페널티: L3 접근 + RAM 접근
    double miss_l3_penalty = T_L3 + (1.0 - h_l3) * T_RAM;

    // L2 미스 시 페널티: L2 접근 + L3 탐색 비용
    double miss_l2_penalty = T_L2 + (1.0 - h_l2) * miss_l3_penalty;

    // AMAT: L1 접근 + L1 미스 시 추가 페널티
    double amat = T_L1 + (1.0 - h_l1) * miss_l2_penalty;

    return amat;
}

// 예시: L1=90%, L2=80%, L3=90% 히트율
// AMAT = 4 + 0.1 × (12 + 0.2 × (40 + 0.1 × 100))
//      = 4 + 0.1 × (12 + 0.2 × 50)
//      = 4 + 0.1 × 22 = 6.2 ns

AMAT(Average Memory Access Time)은 계층별 히트율과 접근 시간으로 계산합니다. L1 히트율이 90%일 때 10%의 미스가 하위 계층으로 전파되므로 전체 계층의 히트율을 높이는 것이 성능 향상의 핵심입니다.

Java — 메모리 계층 접근 패턴 시뮬레이터

// 메모리 계층 시뮬레이터 — 지역성 원리 확인
public class MemoryHierarchySim {

    private static final int L1_SIZE  = 128;   // 128 KB
    private static final int L2_SIZE  = 1024;  // 1 MB
    private static final int L3_SIZE  = 8192;  // 8 MB

    private long l1Hits, l2Hits, l3Hits, ramHits;
    private long totalAccess;

    /** 주소 접근 시뮬레이션 */
    public long access(int address) {
        totalAccess++;

        // 간단한 직접 매핑 시뮬레이션
        if (inCache(address, L1_SIZE)) {
            l1Hits++;
            return 4;   // L1 접근: 4ns
        }
        if (inCache(address, L2_SIZE)) {
            l2Hits++;
            return 12;  // L2 접근: 12ns
        }
        if (inCache(address, L3_SIZE)) {
            l3Hits++;
            return 40;  // L3 접근: 40ns
        }

        ramHits++;
        return 100;   // RAM 접근: 100ns
    }

    private boolean inCache(int addr, int size) {
        return (addr % (size * 1024)) < (size * 512);
    }

    /** 히트율 및 평균 접근 시간 출력 */
    public void printStats() {
        double amat = (l1Hits*4.0 + l2Hits*12.0 +
                       l3Hits*40.0 + ramHits*100.0) / totalAccess;
        System.out.printf("L1 히트율: %.1f%%%n", l1Hits*100.0/totalAccess);
        System.out.printf("AMAT: %.2f ns%n", amat);
    }
}

시간 지역성이 높은 워크로드(같은 주소 반복 접근)는 L1 히트율이 높아 AMAT가 4ns에 근접합니다. 반면 순차 대용량 배열 접근(공간 지역성만 존재)은 캐시 미스가 많아 AMAT가 RAM 접근 시간인 100ns에 가까워집니다.
