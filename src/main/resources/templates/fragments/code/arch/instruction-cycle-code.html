C — 명령어 사이클 시뮬레이션

// 명령어 사이클: Fetch → Decode → Execute 반복

typedef struct {
    int memory[256];  // 명령어 + 데이터 동일 메모리 (폰 노이만)
    int PC;           // Program Counter  — 다음 명령어 주소
    int MAR;          // Memory Address Register
    int MDR;          // Memory Data Register
    int IR;           // Instruction Register
    int ACC;          // Accumulator
} CPU;

void run(CPU *cpu) {
    while (1) {
        /* ① FETCH — PC → MAR, MEM[MAR] → MDR → IR, PC++ */
        cpu->MAR = cpu->PC;
        cpu->MDR = cpu->memory[cpu->MAR];
        cpu->IR  = cpu->MDR;
        cpu->PC++;

        /* ② DECODE — opcode(상위 8비트) / operand(하위 8비트) 분리 */
        int opcode  = (cpu->IR >> 8) & 0xFF;
        int operand =  cpu->IR & 0xFF;

        /* ③ EXECUTE */
        if      (opcode == 0x01) cpu->ACC  = cpu->memory[operand]; // LOAD
        else if (opcode == 0x02) cpu->memory[operand] = cpu->ACC;  // STORE
        else if (opcode == 0x03) cpu->ACC += cpu->memory[operand]; // ADD
        else if (opcode == 0x04) cpu->ACC -= cpu->memory[operand]; // SUB
        else if (opcode == 0xFF) break;                            // HALT
    }
}

MAR/MDR은 CPU-메모리 간 속도 차이를 흡수하는 버퍼 레지스터입니다. PC는 Fetch마다 자동 증가하며, 점프 명령어만 이 값을 강제로 바꿉니다.

# Fetch → Decode → Execute → Interrupt Check 전체 사이클

def run(cpu):
    while True:
        # ① FETCH
        cpu['MAR'] = cpu['PC']
        cpu['MDR'] = cpu['mem'][cpu['MAR']]
        cpu['IR']  = cpu['MDR']
        cpu['PC'] += 1

        # ② DECODE
        opcode  = (cpu['IR'] >> 8) & 0xFF
        operand =  cpu['IR'] & 0xFF

        # ③ EXECUTE
        if   opcode == 0x01: cpu['ACC']  = cpu['mem'][operand]   # LOAD
        elif opcode == 0x02: cpu['mem'][operand] = cpu['ACC']   # STORE

        elif opcode == 0xFF: break                                 # HALT

인터럽트는 반드시 Execute 완료 후에 체크합니다. 명령어 실행 도중에는 처리하지 않아야 레지스터 상태의 일관성이 보장됩니다. 실제로는 PC(EIP)뿐 아니라 범용 레지스터 전체, EFLAGS 등을 스택에 저장해야 ISR 종료 후 원래 상태로 완전히 복귀할 수 있습니다. 위 코드는 흐름 설명을 위해 PC 저장만 표현한 단순화 예시입니다.
