C — CPU 구성 요소 구조체 시뮬레이션

// CPU 내부 레지스터 구조체
typedef struct {
    unsigned int PC;   // Program Counter  — 다음 명령어 주소
    unsigned int IR;   // Instruction Reg  — 현재 명령어
    unsigned int MAR;  // Memory Addr Reg  — 메모리 접근 주소
    unsigned int MBR;  // Memory Buf Reg   — 메모리 데이터 버퍼
    unsigned int ACC;  // Accumulator      — ALU 연산 결과
} Registers;

// ALU — 산술논리연산장치
unsigned int alu_execute(unsigned int op,
                          unsigned int a,
                          unsigned int b) {
    switch (op) {
        case 0x01: return a + b;   // ADD
        case 0x02: return a - b;   // SUB
        case 0x03: return a & b;   // AND
        case 0x04: return a | b;   // OR
        case 0x05: return ~a;      // NOT
        default:   return 0;
    }
}

// CU — 제어장치: 명령어 해석 후 제어 신호 생성
void cu_decode(Registers *reg, unsigned int *memory) {
    unsigned int opcode  = (reg->IR >> 24) & 0xFF;
    unsigned int operand =  reg->IR & 0xFFFFFF;

    switch (opcode) {
        case 0x01:  // LOAD — 메모리 → ACC
            reg->MAR = operand;
            reg->MBR = memory[reg->MAR];
            reg->ACC = reg->MBR;
            break;
        case 0x02:  // STORE — ACC → 메모리
            reg->MAR = operand;
            reg->MBR = reg->ACC;
            memory[reg->MAR] = reg->MBR;
            break;
        case 0x03:  // ADD — ACC + 메모리값
            reg->MAR = operand;
            reg->MBR = memory[reg->MAR];
            reg->ACC = alu_execute(0x01, reg->ACC, reg->MBR);
            break;
    }
    reg->PC++;
}

CU가 IR에 담긴 명령어를 해석해 MAR→MBR→ACC 순서로 데이터를 이동시킵니다. ALU는 CU의 제어 신호를 받아 실제 연산을 수행합니다.

Java — 레지스터 역할 시뮬레이션

// 간단한 CPU 레지스터 동작 시뮬레이터
public class SimpleCPU {

    private int PC  = 0;   // Program Counter
    private int IR  = 0;   // Instruction Register
    private int MAR = 0;   // Memory Address Register
    private int MBR = 0;   // Memory Buffer Register
    private int ACC = 0;   // Accumulator
    private int[] memory;

    public SimpleCPU(int[] program) {
        this.memory = program;
    }

    public void step() {
        // FETCH: PC → MAR → MBR → IR
        MAR = PC;
        MBR = memory[MAR];
        IR  = MBR;
        PC++;

        // DECODE & EXECUTE
        int opcode  = (IR >> 16) & 0xFF;
        int operand =  IR & 0xFFFF;

        switch (opcode) {
            case 1 -> ACC = memory[operand];         // LOAD
            case 2 -> memory[operand] = ACC;         // STORE
            case 3 -> ACC += memory[operand];        // ADD
            case 4 -> ACC -= memory[operand];        // SUB
        }
    }
}

FETCH 단계에서 PC → MAR → MBR → IR 순서로 명령어가 이동합니다. 각 레지스터의 역할이 명확히 구분되어 있으며, CU는 IR을 해석해 ALU와 메모리를 제어합니다.
