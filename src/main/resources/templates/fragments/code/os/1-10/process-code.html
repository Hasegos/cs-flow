C — fork() / exec() / wait() 프로세스 생성

#include <stdio.h>
#include <unistd.h>
#include <sys/wait.h>

int main() {
    pid_t pid = fork();  // 부모 프로세스 복제 → 자식 생성

    if (pid < 0) {
        perror("fork failed");
        return 1;

    } else if (pid == 0) {
        // ── 자식 프로세스 (fork() 반환값 = 0) ──────────────
        // 상태 전이: New → Ready → Running (스케줄러 선택 시)
        printf("Child  PID=%d, Parent PID=%d\n", getpid(), getppid());

        // exec(): 현재 프로세스 이미지를 ls 로 교체
        // 성공하면 이 아래 코드는 절대 실행되지 않음
        char *args[] = { "ls", "-l", NULL };
        execvp(args[0], args);

        // exec() 실패 시에만 도달
        perror("exec failed");
        return 1;

    } else {
        // ── 부모 프로세스 (fork() 반환값 = 자식 PID) ────────
        printf("Parent PID=%d, Child PID=%d\n", getpid(), pid);

        // wait(): 자식 종료 대기 → Waiting 상태 진입
        // 호출하지 않으면 자식이 좀비(Zombie) 프로세스가 됨
        int status;
        waitpid(pid, &status, 0);

        if (WIFEXITED(status)) {
            printf("Child exited with code %d\n", WEXITSTATUS(status));
        }
        // 부모: Waiting → Ready → Running 재진입
    }
    return 0;  // Terminated
}

fork()는 부모의 주소 공간을 복사해 자식을 생성합니다(Copy-on-Write 최적화 적용).
반환값이 0이면 자식, 양수이면 부모입니다.
exec()는 현재 프로세스 이미지를 완전히 교체하므로, fork() 직후 exec()를 호출하면 새 프로그램이 기존 PID를 유지한 채 실행됩니다.
부모가 wait() 없이 종료하면 자식은 고아(Orphan)가 되어 가장 가까운 subreaper(지정된 subreaper가 없으면 init, PID 1)에 입양되고, 자식이 종료됐는데 부모가 wait()을 호출하지 않으면 자식 PCB가 남아 좀비(Zombie)가 됩니다.

Java — ProcessBuilder로 자식 프로세스 생성

import java.io.BufferedReader;
import java.io.InputStreamReader;

public class ProcessDemo {

    public static void main(String[] args) throws Exception {

        // ProcessBuilder = fork() + exec() 조합에 해당
        ProcessBuilder pb = new ProcessBuilder("ls", "-l");
        pb.redirectErrorStream(true);  // stderr → stdout 병합

        // start() 호출 시 자식 프로세스 생성 → New → Ready
        Process child = pb.start();

        int pid = (int) child.pid();
        System.out.println("Child PID: " + pid);

        // 자식 stdout 읽기 (부모는 Waiting 상태로 I/O 대기)
        try (var reader = new BufferedReader(
                new InputStreamReader(child.getInputStream()))) {
            reader.lines().forEach(System.out::println);
        }

        // waitFor(): reaper 스레드가 수집해 둔 종료 코드를 읽음
        // (자식 reaping 자체는 JVM의 process reaper 스레드가 담당)
        int exitCode = child.waitFor();
        System.out.println("Exit code: " + exitCode);

        // ProcessHandle로 살아있는 프로세스 정보 조회 (Java 9+)
        ProcessHandle.allProcesses()
            .filter(h -> h.info().command().isPresent())
            .limit(5)
            .forEach(h -> System.out.println(
                "PID=" + h.pid() + " CMD=" + h.info().command().orElse("?")
            ));
    }
}

Java의 ProcessBuilder는 내부적으로 OS의 fork()+exec()를 호출합니다.
JVM은 ProcessBuilder.start()로 생성한 자식마다 "process reaper" 데몬 스레드를 띄워, 그 스레드가 OS 레벨에서 자식 종료를 직접 기다립니다. 따라서 사용자 코드가 child.waitFor()를 호출하지 않아도 JVM이 자식을 reap하므로 일반적으로 좀비(Zombie) 프로세스는 남지 않습니다.
child.waitFor()는 reaper가 수집해 둔 종료 코드를 읽어 오는 동작이며, 호출을 생략했을 때의 실제 위험은 좀비가 아니라 자식 출력 스트림을 읽지 않아 파이프 버퍼가 가득 차서 발생하는 데드락과 파일 디스크립터 누수입니다.
Java 9부터 ProcessHandle API로 실행 중인 모든 프로세스를 조회하고 PID·CPU 시간·시작 시각 등 상세 정보를 얻을 수 있습니다.
JVM 자체도 하나의 OS 프로세스이므로, JVM 내부의 스레드 스케줄링은 OS 프로세스 스케줄링과 별개 계층에서 동작합니다.
