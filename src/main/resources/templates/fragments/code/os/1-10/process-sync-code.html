C — Mutex로 임계 구역 보호

#include <stdio.h>
#include <pthread.h>

static int            counter = 0;
static pthread_mutex_t mtx     = PTHREAD_MUTEX_INITIALIZER;

void *worker(void *arg) {
    for (int i = 0; i < 100000; i++) {

        pthread_mutex_lock(&mtx);    // 임계 구역 진입
        counter++;                   // 읽기 → 증가 → 쓰기 (원자적 보장)
        pthread_mutex_unlock(&mtx);  // 임계 구역 탈출
    }
    return NULL;
}

int main() {
    pthread_t t1, t2;
    pthread_create(&t1, NULL, worker, NULL);
    pthread_create(&t2, NULL, worker, NULL);
    pthread_join(t1, NULL);
    pthread_join(t2, NULL);
    printf("counter = %d (expected 200000)\n", counter);
    pthread_mutex_destroy(&mtx);
    return 0;
}

counter++ 연산은 load→add→store 세 단계로 이루어져 비원자적입니다. Mutex 없이 실행하면 갱신 분실(Lost Update)이 발생해 최종값이 200000보다 작게 나옵니다.
pthread_mutex_lock()은 Mutex가 이미 잠겨 있으면 호출 스레드가 블로킹되어 일반적으로 Sleep 상태로 CPU를 양보합니다(Blocking). 다만 POSIX는 "블로킹"만 규정하므로 대기 중 CPU 점유 여부는 구현에 따라 다릅니다. glibc 기본 Mutex는 Sleep 전 짧게 Spin할 수 있고, ADAPTIVE 타입은 명시적으로 Spin합니다. 이는 바쁜 대기(Busy Waiting)로 CPU를 계속 점유하는 Spinlock과 대비됩니다.
Lock과 Unlock은 반드시 쌍으로 호출해야 합니다. 예외 발생 등으로 Unlock을 빠뜨리면 데드락이 됩니다. C에서는 cleanup handler, C++에서는 RAII(lock_guard)로 자동 해제를 보장합니다.

C — Semaphore로 생산자-소비자 문제 해결

#include <stdio.h>
#include <pthread.h>
#include <semaphore.h>

#define BUF_SIZE 5

int  buffer[BUF_SIZE];
int  in  = 0, out = 0;

sem_t empty;   // 빈 슬롯 수 (초기값: BUF_SIZE)
sem_t full;    // 찬 슬롯 수 (초기값: 0)
pthread_mutex_t mtx = PTHREAD_MUTEX_INITIALIZER;

void *producer(void *arg) {
    for (int i = 0; i < 10; i++) {
        sem_wait(&empty);            // 빈 슬롯 대기 (empty--)
        pthread_mutex_lock(&mtx);
        buffer[in] = i;
        in = (in + 1) % BUF_SIZE;
        pthread_mutex_unlock(&mtx);
        sem_post(&full);             // 찬 슬롯 증가 (full++)
    }
    return NULL;
}

void *consumer(void *arg) {
    for (int i = 0; i < 10; i++) {
        sem_wait(&full);             // 찬 슬롯 대기 (full--)
        pthread_mutex_lock(&mtx);
        int item = buffer[out];
        out = (out + 1) % BUF_SIZE;
        pthread_mutex_unlock(&mtx);
        sem_post(&empty);            // 빈 슬롯 증가 (empty++)
        printf("consumed: %d\n", item);
    }
    return NULL;
}

int main() {
    sem_init(&empty, 0, BUF_SIZE);  // 빈 슬롯 5개
    sem_init(&full,  0, 0);         // 찬 슬롯 0개
    pthread_t p, c;
    pthread_create(&p, NULL, producer, NULL);
    pthread_create(&c, NULL, consumer, NULL);
    pthread_join(p, NULL);
    pthread_join(c, NULL);
    sem_destroy(&empty);
    sem_destroy(&full);
    return 0;
}

생산자-소비자 문제는 세 개의 동기화 변수를 조합해 해결합니다. empty(빈 슬롯)와 full(찬 슬롯)은 Semaphore로 버퍼 상태를 관리하고, mtx(Mutex)는 버퍼 인덱스(in/out)에 대한 상호 배제를 보장합니다.
sem_wait()은 P 연산(Proberen, "검사/시도")으로 카운터를 감소시키며 0이면 블로킹합니다. sem_post()는 V 연산(Verhogen, "증가")으로 카운터를 증가시켜 대기 중인 스레드를 깨웁니다.
Semaphore의 wait/signal 순서가 바뀌면 데드락이 발생합니다. 반드시 Semaphore를 먼저 획득하고 Mutex를 나중에 획득하는 순서를 지켜야 합니다.

Java — synchronized / ReentrantLock / AtomicInteger

import java.util.concurrent.locks.ReentrantLock;
import java.util.concurrent.atomic.AtomicInteger;

public class SyncDemo {

    // 방법 1: synchronized — Monitor 기반, 가장 단순
    private int counter1 = 0;
    public synchronized void increment1() {
        counter1++;   // 메서드 전체가 임계 구역
    }

    // 방법 2: ReentrantLock — 세밀한 제어, tryLock / fairness 지원
    private int           counter2 = 0;
    private final ReentrantLock lock = new ReentrantLock(true); // fair lock

    public void increment2() {
        lock.lock();
        try {
            counter2++;   // 임계 구역
        } finally {
            lock.unlock();  // 예외 발생해도 반드시 해제 (RAII 대체)
        }
    }

    // 방법 3: AtomicInteger — CAS 기반 Lock-Free, 가장 고성능
    private final AtomicInteger counter3 = new AtomicInteger(0);

    public void increment3() {
        counter3.incrementAndGet();  // CAS 원자 명령어 → 락 불필요
    }

    public static void main(String[] args) throws Exception {
        SyncDemo demo = new SyncDemo();

        // 두 스레드가 각 메서드를 10만 번씩 호출
        Runnable r1 = () -> { for (int i=0;i<100_000;i++) demo.increment1(); };
        Runnable r2 = () -> { for (int i=0;i<100_000;i++) demo.increment2(); };
        Runnable r3 = () -> { for (int i=0;i<100_000;i++) demo.increment3(); };

        Thread[] ts = {
            new Thread(r1), new Thread(r1),
            new Thread(r2), new Thread(r2),
            new Thread(r3), new Thread(r3),
        };
        for (Thread t : ts) t.start();
        for (Thread t : ts) t.join();

        System.out.println("synchronized : "  + demo.counter1); // 200000
        System.out.println("ReentrantLock: "  + demo.counter2); // 200000
        System.out.println("AtomicInteger: "  + demo.counter3.get()); // 200000
    }
}

synchronized는 Java Monitor를 사용하는 가장 간단한 방법입니다. 메서드나 블록 단위로 적용할 수 있으며 예외 시 자동으로 Lock이 해제됩니다.
ReentrantLock은 tryLock(timeout), lockInterruptibly(), 공정성(fairness) 옵션을 지원합니다. 반드시 finally 블록에서 unlock()을 호출해야 합니다.
AtomicInteger는 CAS(Compare-And-Swap) 하드웨어 명령어를 사용해 OS 레벨의 Mutex 없이도 원자적 연산을 보장합니다. 단순 카운터에서는 가장 빠릅니다.
