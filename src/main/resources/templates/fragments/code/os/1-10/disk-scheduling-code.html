C — FCFS / SSTF / SCAN / C-SCAN 구현

#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_TRACK 199
#define N         8

int requests[N] = {98, 183, 37, 122, 14, 124, 65, 67};
int head = 53;

/* ── FCFS ─────────────────────────────────────── */
int fcfs() {
    int cur = head, dist = 0;
    for (int i = 0; i < N; i++) {
        dist += abs(requests[i] - cur);
        cur   = requests[i];
    }
    return dist;
}

/* ── SSTF ─────────────────────────────────────── */
int sstf() {
    int rem[N], n = N, cur = head, dist = 0;
    memcpy(rem, requests, sizeof(requests));

    while (n > 0) {
        int minD = 99999, idx = 0;
        for (int i = 0; i < n; i++) {
            int d = abs(rem[i] - cur);
            if (d < minD) { minD = d; idx = i; }
        }
        dist += minD;
        cur   = rem[idx];
        rem[idx] = rem[--n];   // 처리된 요청 제거
    }
    return dist;
}

/* ── SCAN ─────────────────────────────────────── */
int scan() {
    int sorted[N], cur = head, dist = 0;
    memcpy(sorted, requests, sizeof(requests));

    // 오름차순 정렬
    for (int i = 0; i < N-1; i++)
        for (int j = i+1; j < N; j++)
            if (sorted[j] < sorted[i]) {
                int tmp = sorted[i]; sorted[i] = sorted[j]; sorted[j] = tmp;
            }

    // 위 방향: 헤드 이상 요청을 오름차순으로 처리
    for (int i = 0; i < N; i++)
        if (sorted[i] >= head) { dist += abs(sorted[i] - cur); cur = sorted[i]; }

    // 디스크 끝(MAX_TRACK)까지 이동 후 방향 전환
    dist += abs(MAX_TRACK - cur); cur = MAX_TRACK;

    // 아래 방향: 헤드 미만 요청을 내림차순으로 처리
    for (int i = N-1; i >= 0; i--)
        if (sorted[i] < head) { dist += abs(sorted[i] - cur); cur = sorted[i]; }
    return dist;
}

int main() {
    printf("FCFS   총 이동: %d 트랙\n", fcfs());
    printf("SSTF   총 이동: %d 트랙\n", sstf());
    printf("SCAN   총 이동: %d 트랙\n", scan());
    return 0;
}

SSTF는 매 단계마다 남은 요청 중 현재 헤드에서 가장 가까운 트랙을 선택합니다. 그리디 알고리즘이라 전역 최적이 보장되지 않으며, 극단적인 경우 먼 트랙이 계속 처리되지 못하는 기아(Starvation)가 발생합니다.
SCAN은 정렬 후 현재 헤드 위치 기준으로 이상·이하를 나눠 처리합니다. 헤드가 한쪽 끝까지 이동하므로 모든 요청이 반드시 처리됩니다.
실제 Linux I/O 스케줄러는 이 알고리즘들을 기반으로 하지만, 우선순위·데드라인·공정성을 함께 고려합니다.
HDD에는 BFQ, SSD에는 none(no-op) 또는 mq-deadline이 권장됩니다.

Java — SSTF 구현 (PriorityQueue 활용)

import java.util.*;

public class DiskScheduler {

    static final int HEAD  = 53;
    static final int[] REQ = {98, 183, 37, 122, 14, 124, 65, 67};

    /** SSTF: 현재 위치 기준 가장 가까운 트랙을 우선 처리 */
    static int sstf(int head, int[] requests) {
        List<Integer> rem = new ArrayList<>();
        for (int r : requests) rem.add(r);

        int cur = head, total = 0;
        System.out.printf("HEAD: %d%n", cur);

        while (!rem.isEmpty()) {
            // 가장 가까운 트랙 탐색
            int finalCur = cur;
            rem.sort(Comparator.comparingInt(t -> Math.abs(t - finalCur)));

            int next = rem.remove(0);
            int dist = Math.abs(next - cur);
            total += dist;

            System.out.printf("%d → %d  (거리: %d, 누적: %d)%n",
                cur, next, dist, total);
            cur = next;
        }
        return total;
    }

    /** C-SCAN: 한 방향으로만, 끝에서 0으로 점프 */
    static int cscan(int head, int[] requests) {
        int[] sorted = requests.clone();
        Arrays.sort(sorted);

        List<Integer> order = new ArrayList<>();

        // head 이상: 오름차순
        for (int t : sorted) if (t >= head) order.add(t);
        // 끝(199)까지 이동 후 0으로 점프
        order.add(199);
        order.add(0);
        // head 미만: 오름차순
        for (int t : sorted) if (t < head) order.add(t);

        int cur = head, total = 0;
        for (int next : order) {
            total += Math.abs(next - cur);
            cur    = next;
        }
        return total;
    }

    public static void main(String[] args) {
        System.out.println("=== SSTF ===");
        int d1 = sstf(HEAD, REQ);
        System.out.println("총 이동: " + d1 + " 트랙\n");

        System.out.println("=== C-SCAN ===");
        int d2 = cscan(HEAD, REQ);
        System.out.println("총 이동: " + d2 + " 트랙");
    }
}

SSTF에서 매 반복마다 리스트를 현재 위치 기준으로 정렬해 첫 번째 요소가 항상 최단 거리 트랙이 되도록 합니다. Java의 Comparator.comparingInt()를 사용하면 람다로 간결하게 표현할 수 있습니다.
C-SCAN은 요청을 정렬한 뒤 현재 헤드 이상·이하로 나눠 처리합니다. 끝(199)까지 이동 후 0으로 점프하는 부분을 명시적으로 리스트에 추가합니다.
실제 HDD 드라이버는 I/O 큐를 배열이 아닌 레드-블랙 트리나 링크드 리스트로 관리해 O(log n) 삽입·삭제를 보장합니다.
