C — mmap / malloc 으로 메모리 할당 원리 확인

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <sys/mman.h>

/* ── malloc: Heap 영역 동적 할당 (내부적으로 sbrk/mmap 사용) ── */
void demo_malloc() {
    size_t size = 16 * 1024;  // 16KB (작은 할당 — 모두 128KB 미만이라 힙에 연속 배치)

    void *p1 = malloc(size);   // 연속 16KB 요청
    void *p2 = malloc(size * 2); // 연속 32KB 요청
    void *p3 = malloc(size / 2); // 연속 8KB 요청

    printf("p1: %p\np2: %p\np3: %p\n", p1, p2, p3);
    /* 주소를 보면 p1→p2→p3 순으로 연속 배치됨 */

    free(p2);   // p2 해제 → p1과 p3 사이에 '구멍' 발생 (외부 단편화)

    void *p4 = malloc(640 * 1024); // 640KB(128KB 초과): glibc는 내부적으로 mmap 사용
    // p4의 주소가 p1·p3와 전혀 다른 영역 — 페이지 단위로 분리 배치
    printf("p4: %p\n", p4);

    free(p1); free(p3); free(p4);
}

/* ── mmap: OS에 직접 페이지 단위 메모리 요청 ────────────────── */
void demo_mmap() {
    size_t page = 4096;   // 일반적인 페이지 크기 4KB

    // MAP_ANONYMOUS: 파일 없이 순수 메모리 할당 (페이징 직접 사용)
    void *mem = mmap(NULL, page * 5,
                     PROT_READ | PROT_WRITE,
                     MAP_PRIVATE | MAP_ANONYMOUS,
                     -1, 0);

    if (mem == MAP_FAILED) { perror("mmap"); return; }

    memset(mem, 0, page * 5);
    printf("mmap 주소: %p (페이지 정렬됨)\n", mem);
    // 주소가 항상 페이지 크기(4096)의 배수 → 페이징 단위로 관리

    munmap(mem, page * 5); // 해제: 프레임 반환
}

int main() {
    demo_malloc();
    demo_mmap();
    return 0;
}

malloc은 내부적으로 glibc가 Heap 영역을 관리합니다. 작은 할당은 sbrk()로 Heap을 늘리고, 큰 할당(기본 128KB 이상)은 mmap()으로 독립된 페이지를 직접 요청합니다.
free(p2) 후 p1과 p3 사이에 구멍이 생기지만, OS는 이를 모릅니다. glibc의 메모리 할당자가 free list로 관리하며 다음 malloc 시 재사용합니다. 이것이 사용자 공간의 단편화입니다.
mmap으로 얻은 주소는 항상 페이지 크기(일반적으로 4096B)의 배수입니다.
OS가 페이지 단위로만 메모리를 관리하기 때문입니다.

Java — 페이지 테이블 개념을 코드로 시뮬레이션

import java.util.*;

/**
 * 페이지 테이블 시뮬레이션
 * 실제 OS의 페이지 테이블 동작을 단순화해 구현합니다.
 */
public class PagingSimulator {

    static final int PAGE_SIZE  = 64;   // 64KB (단위 생략)
    static final int N_FRAMES   = 8;    // 물리 메모리: 8 프레임

    // 물리 프레임 테이블: -1 = 빈 프레임, 그 외 = 점유한 PID
    static int[] frames = new int[N_FRAMES];

    // 프로세스별 페이지 테이블: pid → (논리 페이지 → 물리 프레임)
    static Map<Integer, Map<Integer, Integer>> pageTables = new HashMap<>();

    static {
        Arrays.fill(frames, -1); // 초기화: 모든 프레임 비어 있음
    }

    /** 프로세스에 필요한 페이지 수만큼 프레임을 비연속 할당 */
    static boolean allocate(int pid, int sizeKB) {
        int pages = (int) Math.ceil((double) sizeKB / PAGE_SIZE);
        Map<Integer, Integer> table = new LinkedHashMap<>();

        int page = 0;
        for (int f = 0; f < N_FRAMES && page < pages; f++) {
            if (frames[f] == -1) {        // 빈 프레임 발견
                frames[f] = pid;          // 프레임 점유
                table.put(page++, f);   // 논리 페이지 → 물리 프레임 매핑
            }
        }

        if (page < pages) {               // 프레임 부족
            table.keySet().forEach(p -> frames[table.get(p)] = -1);
            return false;
        }

        pageTables.put(pid, table);
        System.out.printf("P%d 할당 (%dKB, %d 페이지): %s%n",
            pid, sizeKB, pages, table);
        return true;
    }

    /** 논리 주소 → 물리 주소 변환 (MMU 역할) */
    static int translate(int pid, int logicalAddr) {
        int pageNum  = logicalAddr / PAGE_SIZE;  // 논리 페이지 번호
        int offset   = logicalAddr % PAGE_SIZE;  // 페이지 내 오프셋
        int frameNum = pageTables.get(pid).get(pageNum); // 페이지 테이블 조회
        return frameNum * PAGE_SIZE + offset;    // 물리 주소
    }

    public static void main(String[] args) {
        allocate(1, 128); // P1: 128KB = 2 페이지 → F0, F1
        allocate(2, 256); // P2: 256KB = 4 페이지 → F2~F5
        allocate(3, 64);  // P3: 64KB  = 1 페이지 → F6

        // P2 종료 → 프레임 해제
        pageTables.get(2).values().forEach(f -> frames[f] = -1);
        pageTables.remove(2);
        System.out.println("P2 해제 후 프레임: " + Arrays.toString(frames));

        // P4: 320KB = 5 페이지 → 비연속으로 F2,F3,F4,F5,F7 할당 성공!
        allocate(4, 320);

        // 논리 주소 변환 예시
        System.out.printf("P4 논리주소 200 → 물리주소 %d%n", translate(4, 200));
    }
}

페이지 테이블은 논리 주소(프로세스 관점)를 물리 주소(실제 RAM)로 변환하는 자료구조입니다. 각 프로세스는 자신만의 독립적인 페이지 테이블을 가집니다.
논리 주소 변환: 페이지 번호 = 논리 주소 / 페이지 크기, 오프셋 = 논리 주소 % 페이지 크기, 물리 주소 = 프레임 번호 × 페이지 크기 + 오프셋.
실제 OS에서는 TLB(Translation Lookaside Buffer)가 최근 변환 결과를 캐시합니다.
TLB Hit 시 메모리 접근 없이 즉시 변환(~1 사이클), Miss 시 페이지 테이블 조회(~수십~수백 사이클)가 발생합니다.
