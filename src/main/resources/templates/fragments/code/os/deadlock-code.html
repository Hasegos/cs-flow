C — 교착 상태 발생 / Lock 순서 통일로 예방

#include <stdio.h>
#include <pthread.h>

pthread_mutex_t r1 = PTHREAD_MUTEX_INITIALIZER;
pthread_mutex_t r2 = PTHREAD_MUTEX_INITIALIZER;

/* ── 교착 상태 발생 버전 ─────────────────────────────────── */
void *thread1_bad(void *arg) {
    pthread_mutex_lock(&r1);   // R1 획득
    pthread_mutex_lock(&r2);   // R2 요청 → T2가 보유 시 블로킹
    /* 임계 구역 */
    pthread_mutex_unlock(&r2);
    pthread_mutex_unlock(&r1);
    return NULL;
}

void *thread2_bad(void *arg) {
    pthread_mutex_lock(&r2);   // R2 획득
    pthread_mutex_lock(&r1);   // R1 요청 → T1이 보유 시 블로킹 → 교착!
    /* 임계 구역 */
    pthread_mutex_unlock(&r1);
    pthread_mutex_unlock(&r2);
    return NULL;
}

/* ── 예방 버전: 항상 r1 → r2 순서로 획득 ───────────────────── */
void *thread1_safe(void *arg) {
    pthread_mutex_lock(&r1);   // 낮은 번호(r1) 먼저
    pthread_mutex_lock(&r2);   // 높은 번호(r2) 나중
    /* 임계 구역 — 순환 대기 불가 */
    pthread_mutex_unlock(&r2);
    pthread_mutex_unlock(&r1);
    return NULL;
}

void *thread2_safe(void *arg) {
    pthread_mutex_lock(&r1);   // T1·T2 모두 r1 먼저 → 순환 없음
    pthread_mutex_lock(&r2);
    /* 임계 구역 */
    pthread_mutex_unlock(&r2);
    pthread_mutex_unlock(&r1);
    return NULL;
}

int main() {
    pthread_t t1, t2;
    /* safe 버전 실행 */
    pthread_create(&t1, NULL, thread1_safe, NULL);
    pthread_create(&t2, NULL, thread2_safe, NULL);
    pthread_join(t1, NULL);
    pthread_join(t2, NULL);
    printf("완료 — 교착 상태 없음\n");
    return 0;
}

교착 상태는 Lock 획득 순서를 정반대로 할 때 발생합니다. T1이 r1→r2, T2가 r2→r1 순서로 획득하면 각자 첫 번째 Lock을 보유한 채 두 번째 Lock을 기다리는 순환 대기가 형성됩니다.
예방책: 모든 스레드에서 Lock 번호 오름차순(r1 → r2)으로만 획득합니다. 이렇게 하면 순환 대기 조건이 성립할 수 없어 교착 상태가 발생하지 않습니다.
pthread_mutex_trylock()은 Lock이 이미 잠겨 있으면 즉시 EBUSY로 실패하는 비블로킹(non-blocking) 함수입니다. 시간 제한 대기가 필요하면 pthread_mutex_timedlock()을 사용합니다. trylock 실패 시 보유한 Lock을 풀고 재시도하는 방식으로 교착을 회피할 수 있지만, 라이브락 가능성이 있으므로 재시도 간격에 지수 백오프(Exponential Backoff)를 적용해야 합니다.

Java — 교착 상태 + tryLock 타임아웃으로 회피

import java.util.concurrent.locks.ReentrantLock;
import java.util.concurrent.TimeUnit;

public class DeadlockDemo {

    private static final ReentrantLock r1 = new ReentrantLock();
    private static final ReentrantLock r2 = new ReentrantLock();

    /* ── 교착 상태 발생 가능 버전 ──────────────────────────────── */
    static void badTask1() throws InterruptedException {
        r1.lock();
        Thread.sleep(50);  // T2가 r2를 잡을 시간 부여
        r2.lock();         // T2가 r2 보유 시 블로킹 → 교착 위험
        try { /* 작업 */ } finally { r2.unlock(); r1.unlock(); }
    }

    /* ── tryLock 타임아웃으로 회피 ─────────────────────────────── */
    static boolean safeTask(ReentrantLock first, ReentrantLock second)
            throws InterruptedException {

        if (!first.tryLock(100, TimeUnit.MILLISECONDS)) {
            return false;  // 100ms 내 획득 실패 → 포기
        }
        try {
            if (!second.tryLock(100, TimeUnit.MILLISECONDS)) {
                return false;  // 두 번째 획득 실패 → first도 자동 해제(finally)
            }
            try {
                /* 두 Lock 모두 보유 — 안전하게 작업 */
                System.out.println("작업 완료");
                return true;
            } finally {
                second.unlock();
            }
        } finally {
            first.unlock();
        }
    }

    public static void main(String[] args) throws Exception {
        Thread t1 = new Thread(() -> {
            try {
                while (!safeTask(r1, r2)) {
                    Thread.sleep(10 + (long)(Math.random()*40)); // 지수 백오프 대체
                }
            } catch (InterruptedException e) { Thread.currentThread().interrupt(); }
        });

        Thread t2 = new Thread(() -> {
            try {
                // 동일 순서(r1→r2)로 요청 — 순환 대기 예방
                while (!safeTask(r1, r2)) {
                    Thread.sleep(10 + (long)(Math.random()*40));
                }
            } catch (InterruptedException e) { Thread.currentThread().interrupt(); }
        });

        t1.start(); t2.start();
        t1.join();  t2.join();
    }
}

ReentrantLock.tryLock(timeout)은 지정 시간 내에 Lock을 획득하지 못하면 false를 반환합니다. 두 번째 Lock 획득에 실패하면 첫 번째 Lock도 finally 블록에서 즉시 해제되어 교착 상태를 회피합니다.
tryLock 방식은 라이브락 가능성이 있으므로 재시도 간격에 랜덤 지연(Jitter)을 추가해야 합니다. 두 스레드가 정확히 같은 간격으로 재시도하면 계속 충돌할 수 있습니다.
가장 안전한 방법은 Lock 획득 순서를 코드 레벨에서 강제(항상 낮은 번호 → 높은 번호)하는 것입니다.
Java의 System.identityHashCode()로 객체 ID 기준 정렬 후 획득하는 패턴이 실무에서 사용됩니다.
