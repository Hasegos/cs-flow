C — pthread 생성 / 조인 / 뮤텍스

#include <stdio.h>
#include <pthread.h>

// 공유 자원 — 여러 스레드가 동시에 접근하면 Race Condition 발생
static int            counter = 0;
static pthread_mutex_t mtx     = PTHREAD_MUTEX_INITIALIZER;

void *worker(void *arg) {
    int id = *(int *)arg;
    for (int i = 0; i < 100000; i++) {
        // 임계 구역(Critical Section) 진입 전 Lock 획득
        pthread_mutex_lock(&mtx);
        counter++;  // 원자적이지 않은 read-modify-write → Lock 필수
        pthread_mutex_unlock(&mtx);
    }
    printf("Thread %d done\n", id);
    return NULL;
}

int main() {
    const int N = 4;
    pthread_t  threads[N];
    int        ids[N];

    // 스레드 생성: 각 스레드는 독립 Stack + PC 를 가지고
    // counter / mtx 를 공유함 (Code·Data·Heap 공유)
    for (int i = 0; i < N; i++) {
        ids[i] = i;
        pthread_create(&threads[i], NULL, worker, &ids[i]);
    }

    // join(): 스레드 종료 대기 + 자원 회수 (프로세스의 wait()에 해당)
    for (int i = 0; i < N; i++) {
        pthread_join(threads[i], NULL);
    }

    printf("Final counter: %d (expected %d)\n", counter, N * 100000);
    pthread_mutex_destroy(&mtx);
    return 0;
}

pthread_create()는 새 스레드를 생성합니다. 각 스레드는 독립된 Stack과 PC를 가지지만 counter·mtx 등 전역 변수는 공유합니다.
counter++ 연산은 읽기(load) → 증가 → 쓰기(store) 3단계로 이루어져 원자적이지 않습니다. Mutex 없이 실행하면 최종값이 N×100000보다 작게 나오는 Race Condition을 확인할 수 있습니다.
기본 attribute(PTHREAD_CREATE_JOINABLE)로 생성된 스레드는 pthread_join()을 호출하지 않으면 좀비 스레드로 남아 시스템 자원이 누수됩니다(누적되면 새 스레드 생성 불가).
join이 필요 없는 경우에는 pthread_detach() 또는 PTHREAD_CREATE_DETACHED 속성으로 생성해 종료 시 자동으로 회수되도록 합니다.

Java — Thread / ExecutorService / volatile

import java.util.concurrent.*;
import java.util.concurrent.atomic.AtomicInteger;

public class ThreadDemo {

    // AtomicInteger: CAS(Compare-And-Swap) 기반 락-프리 원자 연산
    private static final AtomicInteger counter = new AtomicInteger(0);

    // volatile: happens-before 관계 보장 (가시성 + 재정렬 금지)
    // 단, 복합 연산(i++)의 원자성은 보장하지 않음
    private static volatile boolean running = true;

    public static void main(String[] args) throws Exception {

        // 스레드 풀: 최대 4개 스레드를 재사용
        ExecutorService pool = Executors.newFixedThreadPool(4);

        for (int i = 0; i < 4; i++) {
            final int id = i;
            pool.submit(() -> {               // Callable / Runnable 제출
                for (int j = 0; j < 100_000; j++) {
                    counter.incrementAndGet(); // 원자적 증가
                }
                System.out.println("Worker " + id + " done");
            });
        }

        // shutdown(): 새 작업 제출 거부 + 기존 작업 완료 후 종료
        pool.shutdown();
        pool.awaitTermination(10, TimeUnit.SECONDS);

        System.out.println("Final: " + counter.get()); // 400000

        // 가상 스레드 (Java 21, Project Loom, JEP 444)
        // JVM이 carrier thread 위에 M:N으로 스케줄링 — 수백만 개 생성 가능
        try (var vtPool = Executors.newVirtualThreadPerTaskExecutor()) {
            Future<String> future = vtPool.submit(() -> {
                Thread.sleep(100);      // I/O·sleep 블로킹 시 carrier에서 unmount
                return "virtual thread result";
            });
            System.out.println(future.get());
        }
    }
}

Java의 스레드 풀(ExecutorService)은 미리 생성된 스레드를 재사용해 생성·삭제 비용을 없앱니다. 제출된 작업은 내부 큐에 쌓이고 유휴 스레드가 꺼내 실행합니다.
AtomicInteger는 CAS 명령어를 활용해 Mutex 없이도 원자적 증가를 보장합니다. 경쟁이 낮을 때는 synchronized 블록보다 오버헤드가 적지만, 경쟁이 매우 높으면 CAS 재시도 비용으로 성능이 떨어질 수 있습니다.
volatile은 메모리 배리어를 삽입해 레지스터 캐싱·명령어 재정렬을 금지하고 happens-before 관계를 보장합니다. 한 스레드의 쓰기가 이후 다른 스레드의 읽기에 즉시 보이므로 단순 플래그 변수에 적합합니다. 단, read-modify-write 복합 연산의 원자성은 보장하지 않습니다.
Java 21의 가상 스레드(Virtual Thread, JEP 444)는 OS 스레드 1:1 매핑이 아닌 JVM 스케줄러(ForkJoinPool 기반 carrier thread)에 M:N으로 마운트됩니다.
I/O·Thread.sleep 등 블로킹 시 carrier에서 unmount되어 다른 가상 스레드가 실행되므로, 수백만 개의 동시 작업을 적은 OS 스레드로 처리할 수 있습니다.
다만 Java 21~23에서는 synchronized 블록 내부에서 블로킹하면 carrier가 pinning되어 unmount되지 않으므로 이런 경우 ReentrantLock 사용이 권장됩니다(JEP 491에서 Java 24부터 해소).
가상 스레드의 스택도 힙에 저장되므로 메모리는 소모하지만 OS 스레드보다 훨씬 가볍습니다.
