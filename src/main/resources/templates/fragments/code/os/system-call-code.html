C — 시스템 콜 직접 호출 vs 라이브러리 래퍼

#include <unistd.h>
#include <fcntl.h>
#include <sys/syscall.h>

int main() {
    /* ── 방법 1: glibc 래퍼 (일반적) ──────────────────── */
    int fd1 = open("/etc/passwd", O_RDONLY);
    // glibc → syscall 명령어 → 커널 sys_open → fd 반환

    /* ── 방법 2: syscall() 함수로 직접 호출 ───────────── */
    int fd2 = syscall(SYS_open, "/etc/passwd", O_RDONLY);
    // SYS_open = 2 (x86-64 Linux)

    /* ── 방법 3: 인라인 어셈블리 (x86-64) ─────────────── */
    long fd3;
    __asm__ volatile (
        "syscall"
        : "=a"(fd3)                  // RAX = 반환값
        : "0"(2),                    // RAX = 2 (open)
          "D"("/etc/passwd"),       // RDI = 경로
          "S"((long)O_RDONLY),      // RSI = 플래그
          "d"((long)0644)           // RDX = 모드
        : "rcx", "r11", "memory"
    );

    /* ── strace 확인 명령어 ───────────────────────────── */
    // $ strace -e trace=open,openat ./a.out
    // openat(AT_FDCWD, "/etc/passwd", O_RDONLY) = 3

    close(fd1);
    return 0;
}

open()을 호출하면 glibc가 RAX=2(x86-64에서 open 번호), RDI=경로, RSI=플래그를 설정하고 syscall 명령어를 실행합니다. CPU는 Ring 3→Ring 0으로 전환하고 커널의 sys_open을 실행합니다.
syscall() 함수는 시스템 콜을 직접 호출할 때 사용합니다. 커널 버전별로 시스템 콜 번호가 다를 수 있어 일반적으로는 glibc 래퍼 사용을 권장합니다.
인라인 어셈블리로도 직접 호출 가능하지만, 이식성이 없어 커널 개발이나 저수준 실험 목적으로만 사용합니다.
대부분의 운영 코드는 glibc 래퍼를 사용합니다.

Java — JNI 없이 시스템 콜 흐름 추적

import java.io.*;
import java.nio.file.*;
import java.lang.management.*;

public class SystemCallTrace {

    public static void main(String[] args) throws Exception {
        // JVM 내부에서 발생하는 시스템 콜 확인
        // $ strace -p <PID> -e trace=openat,read,write,close

        // ── FileInputStream: open() 시스템 콜 발생 ────────────
        try (var fis = new FileInputStream("/etc/passwd")) {
            // JVM → JNI → open() → syscall → 커널 sys_openat
            byte[] buf = fis.readAllBytes(); // read() 시스템 콜
            System.out.printf("읽은 바이트: %d%n", buf.length);
        } // close() 시스템 콜 (try-with-resources)

        // ── ProcessBuilder: fork() + execve() 시스템 콜 ───────
        ProcessBuilder pb = new ProcessBuilder("ls", "-la");
        pb.redirectErrorStream(true);
        Process proc = pb.start(); // fork() + execve()

        try (var reader = new BufferedReader(
                new InputStreamReader(proc.getInputStream()))) {
            reader.lines().forEach(System.out::println);
        }
        proc.waitFor(); // wait4() 시스템 콜

        // ── JVM이 사용하는 주요 시스템 콜 목록 ────────────────
        // mmap     : 힙/스택/JIT 코드 메모리 할당
        // futex    : synchronized 블록 잠금
        // clone    : 스레드 생성 (pthread_create)
        // epoll    : NIO Selector (비동기 I/O)
        // brk      : 힙 크기 조정
        // getpid   : Thread.currentThread().getId()

        // JVM PID 확인 (getpid 시스템 콜)
        long pid = ProcessHandle.current().pid();
        System.out.println("JVM PID: " + pid);
        System.out.println("strace -p " + pid + " -e trace=all");
    }
}

Java의 파일 I/O는 JVM → JNI → POSIX API → 시스템 콜 경로로 실행됩니다. FileInputStream.read()는 내부적으로 read() 시스템 콜을 호출합니다.
ProcessBuilder.start()는 fork()와 execve() 두 개의 시스템 콜을 순서대로 발생시킵니다. fork()로 현재 프로세스를 복사하고, execve()로 새 프로그램을 실행합니다.
strace -p PID로 실행 중인 JVM의 시스템 콜을 실시간 모니터링할 수 있습니다.
성능 문제 분석 시 특정 시스템 콜이 병목인지 확인하는 데 유용합니다.
