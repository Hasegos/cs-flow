C (POSIX) — 실시간 우선순위로 RM 구현

#include <sched.h>
#include <stdio.h>
#include <stdlib.h>

/* 주기가 짧은 작업일수록 큰 우선순위 값 (RM) */
int set_rm_priority(int prio) {
    struct sched_param sp = { .sched_priority = prio };
    if (sched_setscheduler(0, SCHED_FIFO, &sp) == -1) {
        perror("sched_setscheduler");   /* 권한이 없으면 실패 */
        return -1;
    }
    return 0;
}

int main(int argc, char **argv) {
    /* ./task 60 (T1, 주기 5), ./task 50 (T2, 주기 7) */
    int prio = argc > 1 ? atoi(argv[1]) : 50;
    return set_rm_priority(prio);
}

sched_setscheduler는 대상(0이면 호출한 스레드, 단일 스레드 프로세스라면 자기 자신)의 스케줄링 정책과 우선순위를 정합니다. SCHED_FIFO는 정적 우선순위를 쓰는 실시간 정책이라, 주기가 짧은 작업에 더 큰 값을 주면 RM과 같은 우선순위 배치가 됩니다. SCHED_FIFO 우선순위는 리눅스에서 1~99이고, 실시간 정책을 쓰려면 CAP_SYS_NICE 권한(보통 root)이 있거나 RLIMIT_RTPRIO 한도 안이어야 합니다. 마감을 확인하는 기능은 이 정책에 들어 있지 않아, 우선순위만 맞게 정하는 것은 개발자의 몫입니다. 위 코드는 두 프로세스를 각각 실행하는 상황을 보이는 최소 예시입니다.

C (리눅스) — SCHED_DEADLINE으로 EDF 사용

#define _GNU_SOURCE
#include <sched.h>
#include <stdint.h>
#include <sys/syscall.h>
#include <unistd.h>

#ifndef SCHED_DEADLINE
#define SCHED_DEADLINE 6
#endif

struct sched_attr {
    uint32_t size;
    uint32_t sched_policy;
    uint64_t sched_flags;
    int32_t  sched_nice;
    uint32_t sched_priority;
    uint64_t sched_runtime;    /* 주기마다 보장받을 실행 시간(ns) */
    uint64_t sched_deadline;   /* 상대 마감(ns) */
    uint64_t sched_period;     /* 주기(ns) */
};

int main(void) {
    struct sched_attr attr = {
        .size           = sizeof(attr),
        .sched_policy   = SCHED_DEADLINE,
        .sched_runtime  = 2 * 1000 * 1000,   /* 2ms */
        .sched_deadline = 5 * 1000 * 1000,   /* 5ms */
        .sched_period   = 5 * 1000 * 1000,   /* 5ms */
    };
    return syscall(SYS_sched_setattr, 0, &attr, 0) == -1;
}

SCHED_DEADLINE은 작업마다 runtime, deadline, period를 정해 주면 커널이 EDF 방식으로 스케줄링하는 리눅스의 정책입니다. 주기(period)마다 runtime만큼의 CPU 시간을 deadline 안에 받는 것을 목표로 하며, runtime ≤ deadline ≤ period 관계가 필요합니다. 요청한 대역폭(runtime ÷ period)의 합이 커널이 허용하는 상한(기본은 CPU당 95%)을 넘으면 커널이 수용 검사에서 거절합니다. 이 검사는 필요 조건이라 통과해도 모든 마감이 보장되는 것은 아니며, 이 정책은 CAP_SYS_NICE 권한이 필요합니다. glibc 2.40 이하처럼 sched_setattr 래퍼와 sched_attr 정의가 없는 환경을 기준으로 직접 정의하고 syscall로 부르는 최소 예시입니다. glibc 2.41 이상은 sched.h가 sched_attr 구조체와 sched_setattr 함수를 제공하므로 직접 정의하지 말고 그대로 부릅니다. 실제 사용은 sched(7) 문서를 따라야 합니다.

Python — 이용률과 한계 계산

tasks = [(2, 5), (4, 7)]   # (실행 시간 C, 주기 P)

u = sum(c / p for c, p in tasks)
n = len(tasks)
rm_bound = n * (2 ** (1 / n) - 1)

print(f"U = {u:.3f}, RM 한계 = {rm_bound:.3f}")
print("RM 마감 보장:", u <= rm_bound)    # False (보장 없음)
print("EDF 마감 보장:", u <= 1)          # True

이용률 U는 작업별 실행 시간을 주기로 나눈 값의 합이고, 이 예시는 2/5 + 4/7 ≈ 0.971입니다. RM의 한계 n(2^(1/n) − 1)은 작업이 2개일 때 약 0.828이라 U가 이를 넘고, EDF의 한계는 1이라 U가 이를 넘지 않습니다. RM 한계를 넘었다고 반드시 마감을 놓치는 것은 아니고 보장이 없다는 뜻이며, 이 시각화의 작업 집합에서는 실제로 RM 실행이 마감을 놓칩니다. 이 검사는 주기와 마감이 같고 작업이 서로 독립이며 선점 가능한 단일 프로세서라는 가정에서만 성립하므로, 락이나 여러 코어가 있는 시스템에서는 더 자세한 분석이 필요합니다.
