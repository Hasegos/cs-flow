C (POSIX) — 우선순위 상속 뮤텍스

#include <pthread.h>

pthread_mutex_t lock;
pthread_mutexattr_t attr;

pthread_mutexattr_init(&attr);
pthread_mutexattr_setprotocol(&attr, PTHREAD_PRIO_INHERIT);   /* 우선순위 상속 */
pthread_mutex_init(&lock, &attr);

/* 이제 이 뮤텍스를 쥔 낮은 우선순위 스레드는,
   기다리는 높은 우선순위 스레드의 우선순위를 임시로 물려받는다 */
pthread_mutex_lock(&lock);
/* 임계 구역 */
pthread_mutex_unlock(&lock);

뮤텍스 속성의 프로토콜을 PTHREAD_PRIO_INHERIT로 정하면, 이 뮤텍스를 쥔 스레드가 그 뮤텍스를 기다리는 더 높은 우선순위 스레드의 우선순위를 임시로 물려받습니다. 뮤텍스를 풀면 원래 우선순위로 돌아갑니다. 기본 프로토콜은 PTHREAD_PRIO_NONE이라 상속이 일어나지 않습니다. 우선순위 상한 방식을 쓰려면 PTHREAD_PRIO_PROTECT와 pthread_mutexattr_setprioceiling으로 상한을 정합니다. 주로 SCHED_FIFO 같은 실시간 우선순위를 쓰는 스레드에 쓰며, 지원 여부는 시스템마다 다릅니다(POSIX 선택 옵션).

VxWorks — 패스파인더가 켠 옵션 (개념 예시)

/* 우선순위 상속을 지원하지 않는 뮤텍스 (역전 가능) */
SEM_ID m1 = semMCreate(SEM_Q_PRIORITY);

/* 우선순위 상속을 켠 뮤텍스 */
SEM_ID m2 = semMCreate(SEM_Q_PRIORITY | SEM_INVERSION_SAFE);

VxWorks의 semMCreate는 뮤텍스 세마포어를 만드는 함수이고, SEM_INVERSION_SAFE 옵션을 주면 우선순위 상속이 켜집니다. 패스파인더 사례는 이 옵션이 꺼진 뮤텍스에서 우선순위 역전이 생겼고 옵션을 켜는 패치로 해결되었다고 알려져 있습니다. SEM_INVERSION_SAFE는 큐의 순서를 우선순위로 정하는 SEM_Q_PRIORITY와 함께 쓰도록 되어 있습니다. 위 코드는 개념을 보이는 최소 예시이며, 실제 사용은 VxWorks 문서의 설명을 따라야 합니다.

리눅스 셸 — 실시간 우선순위 확인

# SCHED_FIFO 우선순위 50으로 프로그램 실행 (root 권한)
sudo chrt -f 50 ./worker

# 프로세스의 스케줄링 정책과 실시간 우선순위 확인
chrt -p 1234

# 정책(cls)과 실시간 우선순위(rtprio) 열로 목록 보기
ps -eo pid,cls,rtprio,ni,comm

chrt는 프로세스의 스케줄링 정책과 우선순위를 설정하고 조회하는 도구입니다. -f는 SCHED_FIFO 정책을 뜻하며, 이 정책의 실시간 프로세스는 같은 우선순위 안에서는 스스로 양보하거나 블록되거나 더 높은 우선순위 작업에 선점되기 전까지 CPU를 쥡니다. ps의 cls 열은 스케줄링 클래스(예: FF는 SCHED_FIFO, TS는 일반 시분할)를, rtprio 열은 실시간 우선순위를 보여 줍니다. 실시간 우선순위를 가진 작업이 락을 두고 경쟁하는 상황에서는 앞의 PTHREAD_PRIO_INHERIT 뮤텍스를 쓰는 것이 우선순위 역전을 막는 대표적인 방법입니다.
