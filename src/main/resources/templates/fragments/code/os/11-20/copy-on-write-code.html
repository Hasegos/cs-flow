C — fork 뒤 쓰기는 서로 보이지 않는다

#include <stdio.h>
#include <unistd.h>
#include <sys/wait.h>

int main(void) {
    int x = 10;
    pid_t pid = fork();

    if (pid == 0) {            /* 자식 */
        x = 99;                /* 쓰기 → 이 페이지가 복사됨 */
        printf("child  x=%d\n", x);
        return 0;
    }
    wait(NULL);
    printf("parent x=%d\n", x);   /* 10 그대로 */
}

fork 직후 부모와 자식의 x는 같은 물리 페이지를 읽기 전용으로 공유합니다. 자식이 x에 99를 쓰는 순간 페이지 폴트가 일어나 커널이 그 페이지를 복사하고, 자식은 자기 복사본에 값을 씁니다. 그래서 부모가 출력하는 x는 10 그대로입니다. COW는 구현 방식일 뿐이고 프로그램이 보는 의미는 fork 때 메모리를 전부 복사한 것과 같습니다. x가 있는 페이지가 복사되고, 건드리지 않는 페이지는 부모와 자식이 계속 공유합니다.

C — fork 직후 exec (복사가 필요 없는 패턴)

pid_t pid = fork();
if (pid == 0) {
    execlp("ls", "ls", "-l", (char *)NULL);
    _exit(127);          /* exec이 실패했을 때만 도달 */
}
waitpid(pid, NULL, 0);

/* 비슷한 일을 한 번의 호출로: */
#include <spawn.h>
extern char **environ;
char *argv[] = {"ls", "-l", NULL};
posix_spawnp(&pid, "ls", NULL, NULL, argv, environ);

자식이 fork 직후 execlp를 호출하면 자식의 메모리는 ls 프로그램으로 교체됩니다. COW 덕분에 fork 때 복제한 메모리를 실제로 복사하지 않았으므로 버려지는 복사가 없습니다. exec이 성공하면 execlp는 돌아오지 않고, 실패했을 때만 다음 줄의 _exit가 실행됩니다. 자식에서는 exit 대신 _exit를 써서, fork로 복제된 stdio 버퍼가 부모와 겹쳐 flush되지 않게 합니다. posix_spawn 계열은 새 프로그램을 시작하는 이 흔한 패턴을 하나의 호출로 제공하는 인터페이스입니다.

Python · Redis — fork와 COW를 눈으로 확인

# Python: os.fork 뒤 자식의 변경은 부모에 보이지 않음
import os
data = [1, 2, 3]
pid = os.fork()
if pid == 0:
    data.append(4)
    os._exit(0)
os.wait()
print(data)            # [1, 2, 3]

# Redis: 스냅샷 중 COW로 복사된 메모리 크기
redis-cli INFO persistence | grep cow_size

파이썬의 os.fork도 같은 의미를 줍니다. 자식이 리스트에 4를 추가해도 부모의 리스트는 [1, 2, 3] 그대로이며, 내부적으로는 쓰기가 일어난 페이지가 복사됩니다. CPython은 참조 카운트를 갱신하므로 읽기만 해도 페이지가 복사될 수 있습니다. Redis는 RDB 스냅샷과 AOF 재작성 때 자식을 fork해 운영체제의 COW를 이용하고, INFO persistence의 cow_size 항목은 그동안 COW로 복사된 메모리 크기를 알려 줍니다. 스냅샷 중 쓰기가 많을수록 복사되는 페이지가 늘어 메모리 사용량이 올라가므로, 이 값은 스냅샷 시점에 여유 메모리가 필요한 정도를 가늠하는 데 쓰입니다.
