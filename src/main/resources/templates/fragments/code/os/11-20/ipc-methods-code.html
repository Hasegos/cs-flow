C — 파이프로 부모가 자식에게 보내기

#include <stdio.h>
#include <unistd.h>
#include <sys/wait.h>

int main(void) {
    int fd[2];
    pipe(fd);                          /* fd[0]=읽기 끝, fd[1]=쓰기 끝 */

    if (fork() == 0) {                /* 자식: 읽는 쪽 */
        close(fd[1]);
        char buf[64];
        ssize_t n = read(fd[0], buf, sizeof(buf) - 1);
        if (n < 0) return 1;
        buf[n] = 0;
        printf("child got: %s\n", buf);
        return 0;
    }
    close(fd[0]);                       /* 부모: 쓰는 쪽 */
    write(fd[1], "hello", 5);
    close(fd[1]);
    wait(NULL);
}

pipe는 파일 디스크립터 두 개를 만듭니다. fd[0]은 읽기 끝, fd[1]은 쓰기 끝이며, fork하면 자식도 이 둘을 물려받아 부모와 자식이 같은 파이프를 사이에 두게 됩니다. 한 방향으로만 쓰도록 각자 쓰지 않는 끝을 닫습니다. 파이프가 비어 있는데 쓰는 끝이 열려 있으면 read는 데이터가 올 때까지 기다리고, 쓰는 끝이 모두 닫히면 0을 돌려줍니다. 양방향 통신이 필요하면 파이프를 하나 더 만들거나 소켓을 사용합니다.

C — POSIX 공유 메모리 (쓰는 쪽과 읽는 쪽)

/* 쓰는 프로세스 */
int fd = shm_open("/demo", O_CREAT | O_RDWR, 0600);
ftruncate(fd, 4096);
char *p = mmap(NULL, 4096, PROT_READ | PROT_WRITE, MAP_SHARED, fd, 0);
strcpy(p, "hello");                  /* 메모리에 쓰기만 하면 됨 */

/* 읽는 프로세스 (같은 이름으로 열어서 같은 영역을 매핑) */
int fd2 = shm_open("/demo", O_RDWR, 0);
char *q = mmap(NULL, 4096, PROT_READ, MAP_SHARED, fd2, 0);
printf("%s\n", q);

shm_unlink("/demo");                 /* 더 쓰지 않으면 이름 제거 */

shm_open이 이름 있는 공유 메모리 객체를 열고, ftruncate로 크기를 정한 뒤 mmap으로 각 프로세스의 주소 공간에 연결합니다. 연결 이후에는 일반 메모리처럼 포인터로 읽고 써서 시스템 콜 없이 데이터를 주고받습니다. 다만 쓰는 쪽이 다 쓰기 전에 읽는 쪽이 읽을 수 있으므로, 실제 코드에서는 세마포어나 프로세스 간 공유 뮤텍스로 순서를 맞춰야 합니다. 위 코드는 매핑 방법만 보여 주는 최소 예시입니다. 더 쓰지 않을 때 shm_unlink로 이름을 지웁니다. glibc 2.33 이하에서는 -lrt로 링크해야 하고, 2.34부터는 libc에 들어 있어 필요 없습니다(man 페이지에는 여전히 -lrt가 적혀 있습니다).

셸 · Python — FIFO와 multiprocessing

# 셸: 이름 있는 파이프 (관련 없는 두 프로세스가 이름으로 연결)
mkfifo /tmp/demo.fifo
cat /tmp/demo.fifo &                 # 읽는 쪽: 데이터가 올 때까지 대기
echo hello > /tmp/demo.fifo          # 쓰는 쪽

# Python: 프로세스 사이에 파이프 / 공유 메모리
from multiprocessing import Process, Pipe, shared_memory

def child(conn):
    conn.send("hi")

if __name__ == "__main__":
    parent, c = Pipe()
    p = Process(target=child, args=(c,)); p.start()
    print(parent.recv()); p.join()

    shm = shared_memory.SharedMemory(create=True, size=16)
    shm.buf[:2] = b"ok"
    shm.close(); shm.unlink()

mkfifo가 파일 시스템에 이름 있는 파이프를 만들면 cat과 echo처럼 서로 관련 없는 프로세스도 같은 이름으로 열어 통신합니다. 읽는 쪽이 열릴 때까지 쓰는 쪽이 기다리는 식으로 한쪽이 상대를 기다립니다. 파이썬의 multiprocessing.Pipe는 두 프로세스 사이에 객체를 보내는 연결을 만들고(기본은 양방향이라 운영체제 파이프와는 다른 추상화입니다), 3.8에서 추가된 shared_memory.SharedMemory는 이름 있는 공유 메모리 블록을 만들어 buf로 직접 읽고 쓰게 합니다. 공유 메모리 블록은 다 쓴 뒤 close로 자기 쪽 연결을 닫고, 만든 쪽이 unlink로 블록 자체를 없애야 시스템에 남지 않습니다.
