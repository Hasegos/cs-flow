C — 회수하지 않는 부모가 좀비를 만든다

#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <sys/wait.h>

int main(void) {
    pid_t pid = fork();
    if (pid == 0) {
        exit(0);                  /* 자식: 곧바로 종료 */
    }
    printf("child pid=%d\n", pid);
    sleep(60);                    /* 부모: wait()를 부르지 않고 60초 동안 대기 */
    /* waitpid(pid, NULL, 0);   ← 이 줄을 넣으면 좀비가 사라진다 */
}

자식은 바로 종료하지만 부모가 60초 동안 wait를 부르지 않으므로, 그동안 자식은 좀비 상태로 남습니다. 다른 터미널에서 ps로 보면 이 자식의 상태가 Z이고 defunct로 표시됩니다. 부모가 waitpid를 호출하면 커널이 종료 상태를 부모에게 넘기고 좀비 항목을 지웁니다. 부모가 sleep을 마치고 종료해도 좀비는 init에게 입양되어 회수되므로 사라집니다. kill -9로는 이 좀비를 없앨 수 없다는 점도 직접 시험해 볼 수 있습니다.

C — 좀비를 남기지 않는 두 가지 방법

/* 방법 1: SIGCHLD 핸들러에서 끝난 자식을 모두 회수 */
void on_sigchld(int sig) {
    int saved = errno;             /* waitpid가 errno를 바꿀 수 있어 보존 */
    while (waitpid(-1, NULL, WNOHANG) > 0)
        ;
    errno = saved;
}
/* main 안에서 설정 (signal.h, errno.h 필요) */
struct sigaction sa = {0};
sigemptyset(&sa.sa_mask);
sa.sa_handler = on_sigchld;
sa.sa_flags = SA_RESTART;
sigaction(SIGCHLD, &sa, NULL);

/* 방법 2: 종료 상태가 필요 없다면 SIGCHLD를 무시 */
signal(SIGCHLD, SIG_IGN);

첫 방법은 자식이 끝날 때 오는 SIGCHLD를 핸들러에서 받아 waitpid를 반복 호출합니다. 신호는 여러 개가 합쳐져 한 번만 전달될 수 있으므로 WNOHANG으로 더 회수할 자식이 없을 때까지 반복합니다. 둘째 방법은 SIGCHLD의 처리를 SIG_IGN으로 설정해, 종료한 자식이 좀비가 되지 않고 곧바로 정리되게 합니다. 대신 자식의 종료 상태는 받을 수 없습니다. 종료 상태가 필요하면 첫 방법처럼 직접 회수해야 하고, 신호 처리 방식의 세부는 플랫폼마다 다를 수 있어 서버에서는 문서를 확인합니다.

셸 · Docker · Python — 좀비 관찰과 회수

# 좀비 찾기: 상태 Z, 그 PPID가 회수하지 않는 부모
ps -eo pid,ppid,stat,cmd | awk '$3 ~ /^Z/'

# 컨테이너에서 회수를 맡는 init을 PID 1로
docker run --init myapp

# Python: 자식을 띄웠으면 wait/poll로 회수
import subprocess
p = subprocess.Popen(["sleep", "1"])
p.wait()                  # wait/poll로 회수. 부모가 호출하지 않으면 자식은 좀비로 남는 것이 일반 동작

ps의 상태 열에서 Z로 시작하는 프로세스가 좀비이며, 그 PPID가 가리키는 부모가 wait를 부르지 않는 원인입니다. 해결은 좀비를 지우는 것이 아니라 부모를 고치거나 재시작하는 것입니다. docker run의 --init 옵션은 작은 init 프로세스를 컨테이너의 PID 1로 넣어 고아와 좀비를 회수하게 합니다. 파이썬에서는 Popen으로 시작한 자식을 wait나 poll로 종료를 확인하며, 회수하지 않으면 자식은 부모가 종료할 때까지 좀비로 남는 것이 유닉스의 일반 동작입니다. awk의 작은따옴표 안 문자열은 셸에서 그대로 전달되어 3번째 열이 Z로 시작하는 줄만 출력합니다.
