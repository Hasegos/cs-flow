C — open() / read() / write() 시스템 콜

#include <fcntl.h>
#include <unistd.h>
#include <stdio.h>
#include <string.h>      // strlen
#include <sys/stat.h>    // struct stat, fstat

int main() {
    /* open(): 경로 탐색 → inode 확인 → 파일 디스크립터 반환 */
    int fd = open("/home/user/hello.txt", O_RDWR | O_CREAT, 0644);
    if (fd < 0) { perror("open"); return 1; }
    // fd: 0=stdin, 1=stdout, 2=stderr 이후 가장 작은 정수 (보통 3)

    /* write(): fd → 파일 테이블 → inode → 데이터 블록 쓰기 */
    const char *msg = "Hello, File System!\n";
    ssize_t n = write(fd, msg, strlen(msg));
    if (n < 0) { perror("write"); }

    /* lseek(): 파일 오프셋 이동 (inode의 파일 위치 포인터 변경) */
    lseek(fd, 0, SEEK_SET);   // 처음으로 돌아가기

    /* read(): 커널이 inode → 블록 포인터 → 디스크 읽기 → 버퍼 복사 */
    char buf[64];
    n = read(fd, buf, sizeof(buf) - 1);
    if (n > 0) { buf[n] = '\0'; printf("%s", buf); }

    /* stat(): inode 메타데이터 조회 (크기, 권한, 타임스탬프) */
    struct stat st;
    fstat(fd, &st);
    printf("inode: %lu, size: %ld, blocks: %ld\n",
           st.st_ino, st.st_size, st.st_blocks);

    close(fd);   // fd 해제 (inode 링크 카운트는 유지)
    return 0;
}

open()은 경로를 탐색해 inode를 찾고, 프로세스의 파일 디스크립터 테이블에 항목을 추가한 뒤 정수 fd를 반환합니다.
write()는 즉시 디스크에 쓰지 않고 커널 페이지 캐시(Page Cache)에 씁니다. fsync()를 호출해야 디스크에 확실히 반영됩니다.
fstat()로 st_ino(inode 번호), st_size(바이트 단위 크기), st_blocks(512B 단위 블록 수)를 조회할 수 있습니다. st_nlink가 하드 링크 개수입니다.

Java — NIO.2 / Files API로 inode 정보 확인

import java.nio.file.*;
import java.nio.file.attribute.*;
import java.nio.ByteBuffer;
import java.nio.channels.FileChannel;

public class FileSystemDemo {

    public static void main(String[] args) throws Exception {
        Path path = Path.of("/home/user/hello.txt");

        // ── inode 메타데이터 조회 ──────────────────────────────────
        BasicFileAttributes attr = Files.readAttributes(path, BasicFileAttributes.class);
        System.out.println("크기: "     + attr.size());
        System.out.println("생성: "     + attr.creationTime());
        System.out.println("수정: "     + attr.lastModifiedTime());
        System.out.println("디렉토리: " + attr.isDirectory());

        // Linux inode 번호 (UnixPath에서만 가용)
        if (attr instanceof PosixFileAttributes) {
            // unix:ino — 플랫폼 의존 속성, Unix/Linux에서만 가용
            Object inode = Files.getAttribute(path, "unix:ino");
            System.out.println("inode: " + inode);
        }

        // ── FileChannel: 블록 단위 직접 I/O ─────────────────────────
        try (var fc = FileChannel.open(path, StandardOpenOption.READ)) {
            ByteBuffer buf = ByteBuffer.allocate(1024);
            int n = fc.read(buf);       // 커널 → JVM 버퍼로 복사
            buf.flip();
            byte[] bytes = new byte[n];
            buf.get(bytes);
            System.out.println(new String(bytes));
        }

        // ── 하드 링크 생성 (같은 inode를 가리키는 새 경로) ────────
        Path link = Path.of("/home/user/hello_hard.txt");
        Files.createLink(link, path);     // 링크 카운트 +1

        // ── 심볼릭 링크 (경로를 가리키는 별도 inode) ──────────────
        Path sym = Path.of("/home/user/hello_sym.txt");
        Files.createSymbolicLink(sym, path);

        System.out.println("하드링크 대상: " + Files.isSameFile(path, link));
        System.out.println("심볼릭 대상: "  + Files.readSymbolicLink(sym));
    }
}

Java NIO.2의 Files.readAttributes()는 내부적으로 fstat() 시스템 콜을 호출합니다. BasicFileAttributes는 크기·타임스탬프, PosixFileAttributes는 권한·소유자까지 제공합니다.
FileChannel은 전통적인 InputStream보다 블록 단위로 효율적으로 I/O하며, 메모리 맵 파일(MappedByteBuffer)로 사용하면 디스크 블록을 메모리처럼 직접 접근할 수 있습니다.
Files.createLink()는 하드 링크(같은 inode)를 만들어 링크 카운트를 1 증가시킵니다. 원본 파일을 삭제해도 카운트가 0이 되지 않으면 데이터는 유지됩니다.
