Python — 플로이드-워셜과 경로 복원

INF = float('inf')

def floyd_warshall(n, edges):
    dist = [[INF] * n for _ in range(n)]
    nxt = [[None] * n for _ in range(n)]
    for i in range(n):
        dist[i][i] = 0
    for u, v, w in edges:
        dist[u][v] = w
        nxt[u][v] = v                           # u에서 v로 가려면 먼저 v로
    for k in range(n):                         # 경유지가 가장 바깥
        for i in range(n):
            for j in range(n):
                if dist[i][k] + dist[k][j] < dist[i][j]:
                    dist[i][j] = dist[i][k] + dist[k][j]
                    nxt[i][j] = nxt[i][k]       # k 쪽으로 먼저 간다
    return dist, nxt

def path(nxt, i, j):
    p = [i]
    while i != j:
        i = nxt[i][j]
        p.append(i)
    return p

# A=0, B=1, C=2, D=3 · (출발, 도착, 가중치)
E = [(0, 1, 3), (0, 3, 7), (1, 0, 8), (1, 2, 2),
     (2, 0, 5), (2, 3, 1), (3, 0, 2)]
dist, nxt = floyd_warshall(4, E)
for row in dist:
    print(row)
# [0, 3, 5, 6]
# [5, 0, 2, 3]
# [3, 6, 0, 1]
# [2, 5, 7, 0]
print(path(nxt, 0, 3), dist[0][3])       # [0, 1, 2, 3] 6

플로이드-워셜은 거리 행렬 dist에서 시작해, 경유지 k를 하나씩 허용하며 모든 칸 (i, j)에서 dist[i][j]와 dist[i][k] + dist[k][j] 중 작은 값을 고릅니다. 이 점화식 하나가 알고리즘의 전부이고, 경유지 k, 출발지 i, 도착지 j의 삼중 반복이므로 O(V³)입니다. 이 예시 그래프에서는 A → D가 직접 7이지만 A → B → C → D로 3 + 2 + 1 = 6이 되어 dist[0][3]이 6으로 바뀝니다. 경유지 k를 가장 바깥 반복문에 두는 것이 중요합니다. 바깥 반복이 k여야 "경유지를 k까지 허용했을 때의 최단 거리"라는 뜻이 유지되고, 순서를 바꾸면 결과가 틀릴 수 있습니다. dist를 갱신할 때 nxt[i][j]도 nxt[i][k]로 바꿔 두면, path 함수가 nxt를 따라가며 실제 경로를 복원합니다. A에서 D까지는 [0, 1, 2, 3], 곧 A → B → C → D입니다. 시각화의 거리 행렬 갱신 탭과 경로 복원 탭이 보여 주는 과정과 같습니다.

Python — 음수 사이클 탐지

neg = E + [(1, 3, -20)]                      # B → D(-20) 추가
d2, _ = floyd_warshall(4, neg)
print([d2[i][i] for i in range(4)])      # [-15, -15, -5, -30]
print(any(d2[i][i] < 0 for i in range(4)))     # True

플로이드-워셜은 음수 가중치 간선이 있어도 음수 사이클만 없으면 올바르게 동작합니다. 음수 사이클이 있는지는 계산이 끝난 뒤 대각선 dist[i][i]를 보면 알 수 있습니다. 정점에서 출발해 자기 자신으로 돌아오는 경로의 합이 0보다 작아졌다면, 그 경로가 음수 사이클을 포함하므로 그래프에 음수 사이클이 있다는 뜻입니다. 음수가 나온 정점이 꼭 그 사이클 위에 있는 것은 아닙니다. 위에서는 B → D(-20) 간선을 더해 B → D → A → B의 합이 -20 + 2 + 3 = -15인 음수 사이클을 만들었고, 대각선에 음수가 나타납니다. 음수 사이클이 있으면 최단 거리가 정의되지 않으므로 이때의 거리 값은 쓰지 않습니다.

Java — 플로이드-워셜

static final int INF = Integer.MAX_VALUE;

static int[][] floydWarshall(int n, int[][] edges) {
    int[][] d = new int[n][n];
    for (int[] row : d) Arrays.fill(row, INF);
    for (int i = 0; i < n; i++) d[i][i] = 0;
    for (int[] e : edges) d[e[0]][e[1]] = e[2];
    for (int k = 0; k < n; k++)
        for (int i = 0; i < n; i++)
            for (int j = 0; j < n; j++)
                if (d[i][k] != INF && d[k][j] != INF && d[i][k] + d[k][j] < d[i][j])
                    d[i][j] = d[i][k] + d[k][j];
    return d;
}

// 같은 그래프의 결과
// [0, 3, 5, 6]  [5, 0, 2, 3]  [3, 6, 0, 1]  [2, 5, 7, 0]

자바 버전도 같은 삼중 반복문입니다. INF를 Integer.MAX_VALUE로 두었기 때문에 두 거리를 더하기 전에 둘 다 INF가 아닌지 검사해야 합니다. 검사하지 않으면 INF에 양수를 더할 때 값이 넘쳐 음수가 되어 닿지 못하는 쌍이 가까운 것처럼 갱신됩니다. 정점 수 V에 대해 시간은 O(V³), 거리 행렬 저장에 공간은 O(V²)입니다. 모든 쌍을 한꺼번에 구해야 하고 정점 수가 크지 않을 때 간단해서 많이 씁니다. 한 출발점에서만 구한다면 다익스트라나 벨만-포드가 알맞습니다.
