Java — N-Queens 백트래킹

/* ── 한 행씩 채우다가, 막히면 되돌아감(재귀 + 원상복구) ────── */
public boolean solve(int[] cols, int row, int n) {
    if (row == n) return true;              // 모든 행에 퀸을 놓았다 — 해 발견

    for (int col = 0; col < n; col++) {
        if (isSafe(cols, row, col)) {
            cols[row] = col;                    // 일단 놓아본다
            if (solve(cols, row + 1, n)) return true;  // 다음 행 진행
            cols[row] = -1;                   // 실패했다면 원상복구(백트랙)
        }
    }
    return false;                          // 이 행에서 안전한 열이 하나도 없음 — 가지치기
}

cols[row] = -1 한 줄이 백트래킹의 핵심입니다 — 재귀 호출이 실패로 끝나면, 방금 놓은 퀸을 반드시 치워야 다음 열을 공정하게 시도할 수 있습니다.
for문이 끝까지 돌았는데도 안전한 열을 못 찾았다면 false를 반환합니다 — 이 값이 호출한 쪽(이전 행)에 "이 갈래는 안 된다"는 신호가 되어 자연스럽게 되돌아갑니다.

Java — 충돌 검사 (isSafe)

/* ── 이미 놓은 퀸들과 같은 열/대각선인지 확인 ──────────────── */
public boolean isSafe(int[] cols, int row, int col) {
    for (int r = 0; r < row; r++) {          // 이전 행들과만 비교하면 충분 (아래 행은 아직 비어 있음)
        int c = cols[r];
        if (c == col) return false;              // 같은 열
        if (Math.abs(c - col) == Math.abs(r - row)) return false;  // 같은 대각선
    }
    return true;
}

행은 이미 한 행에 퀸을 하나만 놓기로 정했으니 따로 검사할 필요가 없고, 열이 같은지와 대각선인지만 확인하면 됩니다.
두 칸이 대각선 위에 있다는 것은 |행 차이| == |열 차이|라는 뜻입니다 — 이 한 줄로 왼쪽 대각선과 오른쪽 대각선을 동시에 검사합니다.

Java — 모든 해 찾기 (되돌아가서 계속 탐색)

/* ── 해를 찾아도 멈추지 않고 기록한 뒤 계속 탐색 ──────────── */
public void solveAll(int[] cols, int row, int n, List<int[]> solutions) {
    if (row == n) {
        solutions.add(cols.clone());       // 해를 기록만 하고 return으로 끝내지 않음
        return;
    }
    for (int col = 0; col < n; col++) {
        if (isSafe(cols, row, col)) {
            cols[row] = col;
            solveAll(cols, row + 1, n, solutions);  // 성공/실패 여부와 상관없이 항상 진행
            cols[row] = -1;                   // 해를 찾았어도 원상복구하고 다음 열 계속 시도
        }
    }
}

첫 번째 코드는 해를 하나만 찾으면 true를 반환하며 즉시 멈추지만, 이 버전은 해를 찾아도 계속 진행해 4-Queens의 두 해([1,3,0,2]와 [2,0,3,1])를 모두 찾습니다.
이 시각화의 "탐색 트리" 탭이 정확히 이 버전으로 동작합니다 — 첫 해를 찾은 뒤에도 트리 탐색이 끝까지 계속되는 것을 확인할 수 있습니다.
