Java — 크루스칼 (유니온-파인드로 사이클 검사)

/* ── 가중치 오름차순 정렬 후, 사이클을 안 만들면 추가 ──────── */
public int kruskal(int n, int[][] edges) {  // edges[i] = {u, v, weight}
    Arrays.sort(edges, (a, b) -> a[2] - b[2]);   // 가중치 기준 정렬
    int[] parent = new int[n];
    for (int i = 0; i < n; i++) parent[i] = i;

    int totalWeight = 0, edgeCount = 0;
    for (int[] e : edges) {
        int ru = find(parent, e[0]), rv = find(parent, e[1]);
        if (ru != rv) {                     // 서로 다른 그룹 → 연결해도 사이클 없음
            parent[ru] = rv;
            totalWeight += e[2];
            edgeCount++;
            if (edgeCount == n - 1) break;  // 간선을 n-1개 모으면 완성
        }
    }
    return totalWeight;
}

private int find(int[] parent, int x) {
    while (parent[x] != x) x = parent[x];   // x가 속한 그룹의 대표 노드 찾기
    return x;
}

find(parent, u) != find(parent, v) 한 줄이 "사이클이 생기는지" 판단의 전부입니다 — 두 노드가 이미 같은 그룹(같은 대표 노드)이면 연결 시 반드시 사이클이 생깁니다.
정렬에 O(E log E), 유니온-파인드 연산은 사실상 O(1)에 가까워 전체는 O(E log E)입니다.
algo-07(그리디)의 활동 선택처럼 "정렬 후 한 번 훑기"와 같은 패턴입니다.

Java — 프림 (algo-05 다익스트라와 거의 동일한 골격)

/* ── "누적 거리" 대신 "트리에 연결하는 간선 하나의 가중치"만 비교 ── */
public int prim(int n, List<List<int[]>> adj, int start) {
    boolean[] inTree = new boolean[n];
    int[] minEdge = new int[n];
    Arrays.fill(minEdge, Integer.MAX_VALUE);
    minEdge[start] = 0;

    int totalWeight = 0;
    for (int count = 0; count < n; count++) {
        int u = -1;
        for (int i = 0; i < n; i++) {          // 트리 밖에서 연결 비용이 가장 싼 노드 탐색
            if (!inTree[i] && (u == -1 || minEdge[i] < minEdge[u])) u = i;
        }
        inTree[u] = true;
        totalWeight += minEdge[u];

        for (int[] edge : adj.get(u)) {          // u의 인접 간선으로 연결 비용 갱신
            int v = edge[0], w = edge[1];
            if (!inTree[v] && w < minEdge[v]) minEdge[v] = w;
        }
    }
    return totalWeight;
}

다익스트라 코드와 나란히 놓고 비교해보면 구조가 거의 동일합니다 — 차이는 minEdge[v] = w 딱 한 줄입니다. 다익스트라는 dist[u] + w(누적 거리)와 비교하지만, 프림은 w(간선 하나의 가중치)만 비교합니다.
"지금까지 온 총 거리"가 중요한가, "트리에 새로 연결하는 이 간선 하나"가 중요한가 — 이 관점 차이가 최단 경로 문제와 MST 문제를 가릅니다.

Java — 유니온-파인드 경로 압축 (최적화)

/* ── find할 때마다 트리를 납작하게 눌러 다음 find를 더 빠르게 ── */
private int findCompressed(int[] parent, int x) {
    if (parent[x] != x) {
        parent[x] = findCompressed(parent, parent[x]);  // 재귀로 대표 노드를 찾은 뒤
    }
    return parent[x];                    // x를 대표 노드에 직접 연결(경로 압축)
}

기본 find는 그룹이 길게 이어져 있으면 대표 노드까지 거슬러 올라가는 데 시간이 걸립니다.
경로 압축은 한 번 찾은 뒤 중간 노드들을 전부 대표 노드에 직접 연결해, 다음번 find는 거의 O(1)에 끝나게 만듭니다.
이 시각화의 크루스칼은 간단한 버전(경로 압축 없음)으로 동작하지만, 그래프가 커지면 이 최적화가 성능에 큰 차이를 만듭니다.
