C — 5단계 파이프라인 레지스터 시뮬레이션

// 파이프라인 단계 간 데이터를 보관하는 파이프라인 레지스터
typedef struct { unsigned int pc, instruction; }           IF_ID;
typedef struct { unsigned int pc, opcode, rs, rt, rd, imm; } ID_EX;
typedef struct { unsigned int alu_result, rt_val, rd; }        EX_MEM;
typedef struct { unsigned int result, rd; }                    MEM_WB;

// 파이프라인 레지스터 전역 선언
IF_ID  if_id   = {0};
ID_EX  id_ex   = {0};
EX_MEM ex_mem  = {0};
MEM_WB mem_wb  = {0};

// 각 단계 함수 — 한 클럭에 동시 실행됨
void stage_IF(unsigned int *pc, unsigned int *mem) {
    if_id.instruction = mem[*pc];  // 명령어 인출
    if_id.pc = (*pc)++;            // PC 증가
}

void stage_ID() {
    id_ex.opcode = (if_id.instruction >> 26) & 0x3F;
    id_ex.rs     = (if_id.instruction >> 21) & 0x1F;
    id_ex.rt     = (if_id.instruction >> 16) & 0x1F;
    id_ex.rd     = (if_id.instruction >> 11) & 0x1F;
    id_ex.imm    =  if_id.instruction        & 0xFFFF;
    id_ex.pc     =  if_id.pc;
}

void stage_EX(unsigned int *regs) {
    ex_mem.rd     = id_ex.rd;
    ex_mem.rt_val = regs[id_ex.rt];
    switch (id_ex.opcode) {
        case 0x00: ex_mem.alu_result = regs[id_ex.rs] + regs[id_ex.rt]; break; // ADD
        case 0x23: ex_mem.alu_result = regs[id_ex.rs] + id_ex.imm;    break; // LOAD
        case 0x2B: ex_mem.alu_result = regs[id_ex.rs] + id_ex.imm;    break; // STORE
    }
}

void stage_MEM(unsigned int *data_mem) {
    mem_wb.rd = ex_mem.rd;
    if (id_ex.opcode == 0x23)       // LOAD
        mem_wb.result = data_mem[ex_mem.alu_result];
    else if (id_ex.opcode == 0x2B)  // STORE
        data_mem[ex_mem.alu_result] = ex_mem.rt_val;
    else
        mem_wb.result = ex_mem.alu_result;
}

void stage_WB(unsigned int *regs) {
    regs[mem_wb.rd] = mem_wb.result;  // 결과를 레지스터에 기록
}

// 한 클럭 사이클 실행 — 5단계가 동시에 작동
void clock_cycle(unsigned int *pc,
                  unsigned int *instr_mem,
                  unsigned int *data_mem,
                  unsigned int *regs) {
    stage_WB(regs);        // WB 먼저 — 포워딩 충돌 방지
    stage_MEM(data_mem);
    stage_EX(regs);
    stage_ID();
    stage_IF(pc, instr_mem);
}

파이프라인 레지스터(IF/ID, ID/EX, EX/MEM, MEM/WB)가 각 단계 사이에서 데이터를 래치(Latch)합니다. 매 클럭에 5단계가 동시에 실행되며, WB를 먼저 수행해 같은 클럭 내 포워딩 충돌을 방지합니다.

Java — 파이프라인 해저드 시뮬레이션

// 데이터 해저드 감지 및 스톨 삽입 시뮬레이터
public class PipelineSimulator {

    private int[] registers = new int[32];
    private int   stallCount = 0;

    // 파이프라인 레지스터 (단계 간 데이터 보관)
    private int ifId_instr, idEx_rd, exMem_rd, memWb_rd;
    private boolean idEx_regWrite, exMem_regWrite;

    /**
     * 데이터 해저드(RAW) 감지
     * — ID 단계 명령어의 소스 레지스터가
     *   앞선 명령어의 목적지 레지스터와 겹치는지 확인
     */
    private boolean detectDataHazard(int rs, int rt) {
        // EX/MEM 해저드: 바로 앞 명령어가 아직 WB 전
        if (exMem_regWrite && exMem_rd != 0 &&
            (exMem_rd == rs || exMem_rd == rt)) {
            return true;
        }
        // MEM/WB 해저드: 두 단계 앞 명령어
        if (idEx_regWrite && idEx_rd != 0 &&
            (idEx_rd == rs || idEx_rd == rt)) {
            return true;
        }
        return false;
    }

    /**
     * 스톨(Bubble) 삽입
     * — 해저드 감지 시 NOP을 끼워 파이프라인을 1클럭 지연
     */
    private void insertStall() {
        // ID/EX 레지스터를 NOP(버블)으로 초기화
        idEx_rd       = 0;
        idEx_regWrite = false;
        stallCount++;
        System.out.printf("[STALL] 버블 삽입 — 총 스톨 횟수: %d%n", stallCount);
    }

    /**
     * 포워딩(Forwarding) — 스톨 없이 해저드 해소
     * EX/MEM 또는 MEM/WB 레지스터에서 직접 값을 가져옴
     */
    private int forwardValue(int regNum) {
        if (exMem_regWrite && exMem_rd == regNum) {
            System.out.println("[FORWARD] EX/MEM → EX");
            return registers[exMem_rd]; // EX/MEM 포워딩
        }
        if (idEx_regWrite && idEx_rd == regNum) {
            System.out.println("[FORWARD] MEM/WB → EX");
            return registers[idEx_rd];  // MEM/WB 포워딩
        }
        return registers[regNum];
    }
}

RAW(Read After Write) 해저드는 앞선 명령어의 결과가 아직 레지스터에 기록되기 전에 뒤 명령어가 같은 레지스터를 읽을 때 발생합니다. 스톨(버블 삽입)은 성능을 낮추지만 정확성을 보장하고, 포워딩은 파이프라인 레지스터에서 직접 값을 전달해 스톨을 최소화합니다.
