Java — DNS 조회 및 레코드 분석

import java.net.*;
import javax.naming.*;
import javax.naming.directory.*;

public class DnsLookup {

    public static void main(String[] args) throws Exception {

        // ── 기본 A 레코드 조회 (InetAddress) ─────────────────
        // OS의 /etc/resolv.conf에 설정된 DNS 리졸버 사용
        InetAddress[] addrs = InetAddress.getAllByName("example.com");
        for (InetAddress addr : addrs) {
            System.out.println("A: " + addr.getHostAddress()); // 93.184.216.34
        }

        // 역방향 조회 (PTR 레코드)
        InetAddress ip = InetAddress.getByName("8.8.8.8");
        System.out.println("PTR: " + ip.getHostName()); // dns.google

        // ── JNDI로 다양한 DNS 레코드 조회 ─────────────────────
        var env = new java.util.Hashtable<String, String>();
        env.put(Context.INITIAL_CONTEXT_FACTORY,
               "com.sun.jndi.dns.DnsContextFactory");
        env.put(Context.PROVIDER_URL, "dns://8.8.8.8"); // Google DNS 직접 지정

        DirContext ctx = new InitialDirContext(env);

        // MX 레코드 조회 (메일 서버)
        Attributes mx = ctx.getAttributes("gmail.com", new String[]{"MX"});
        NamingEnumeration<?> mxVals = mx.get("MX").getAll();
        while (mxVals.hasMore()) {
            System.out.println("MX: " + mxVals.next());
            // "10 alt1.gmail-smtp-in.l.google.com."
        }

        // TXT 레코드 조회 (SPF, DKIM 등)
        Attributes txt = ctx.getAttributes("example.com", new String[]{"TXT"});
        System.out.println("TXT: " + txt.get("TXT"));

        // NS 레코드 조회 (권한 네임서버)
        Attributes ns = ctx.getAttributes("example.com", new String[]{"NS"});
        System.out.println("NS: " + ns.get("NS"));
        // "a.iana-servers.net., b.iana-servers.net."

        ctx.close();
    }
}

InetAddress.getAllByName()은 JVM이 OS의 DNS 리졸버를 통해 A 레코드를 조회합니다.
결과는 JVM 내부에 캐시되며, networkaddress.cache.ttl 시스템 프로퍼티로 TTL을 제어합니다.
JNDI DnsContextFactory를 사용하면 A 레코드 외에 MX·NS·TXT·SOA 등 다양한 레코드를 조회할 수 있습니다.
DNS 서버 주소를 직접 지정해 특정 리졸버로 우회할 수도 있습니다.
MX 레코드의 숫자 접두사는 우선순위(낮을수록 높음)입니다. 메일 서버는 우선순위가 낮은 MX부터 시도합니다.
Spring 환경에서는 JVM DNS 캐시가 영구 캐싱되어 문제가 될 수 있습니다.
networkaddress.cache.ttl=60으로 설정하면 60초마다 재조회합니다.

Python — DNS 질의 패킷 직접 분석 (dnspython)

import dns.resolver
import dns.message
import dns.query
import dns.rdatatype

# ── 기본 A 레코드 조회 ──────────────────────────────────
resolver = dns.resolver.Resolver()
resolver.nameservers = ['8.8.8.8', '1.1.1.1']  # Google / Cloudflare DNS

answers = resolver.resolve('example.com', 'A')
for rdata in answers:
    print(f"A: {rdata.address}")          # 93.184.216.34
    print(f"TTL: {answers.ttl}초")        # 3600

# ── 재귀 질의 직접 추적 (iterative) ─────────────────────
def trace_dns(domain: str):
    # 루트 서버부터 시작
    nameservers = ['198.41.0.4']   # a.root-servers.net

    for level in ['루트', 'TLD', '권한']:
        qname = dns.name.from_text(domain)
        req   = dns.message.make_query(qname, dns.rdatatype.A)

        try:
            resp = dns.query.udp(req, nameservers[0], timeout=3)
        except Exception as e:
            print(f"{level} 질의 실패: {e}"); break

        # 최종 답변이 있으면 종료
        if resp.answer:
            print(f"{level} → 최종 답변: {resp.answer[0]}")
            break

        # 위임 정보(NS)로 다음 서버 추출
        next_ns = []
        for rrset in resp.additional:
            if rrset.rdtype == dns.rdatatype.A:
                next_ns.append(str(rrset[0]))
        print(f"{level} → 다음 NS: {next_ns[:2]}")
        if next_ns: nameservers = next_ns

trace_dns('example.com')

# ── 캐시 확인 (로컬 시스템) ─────────────────────────────
# macOS:  sudo dscacheutil -cachedump -entries Host
# Linux:  systemd-resolve --statistics
# Windows: ipconfig /displaydns

dnspython은 DNS 프로토콜을 직접 다루는 라이브러리입니다. 특정 DNS 서버를 지정해 질의하거나, UDP 패킷을 직접 구성해 전송할 수 있습니다.
trace_dns 함수는 브라우저 없이 DNS 재귀 질의 과정을 직접 추적합니다. 루트→TLD→권한 서버를 단계별로 질의하며 위임 체인을 따라갑니다.
resp.additional 섹션에는 위임된 NS 서버의 IP(글루 레코드)가 포함됩니다. 이를 통해 닭과 달걀 문제를 해결합니다.
실제 운영 환경에서는 시스템 DNS 캐시를 주기적으로 플러시해야 도메인 변경이 즉시 반영됩니다.
TTL 만료 전에는 캐시된 이전 IP로 연결될 수 있습니다.
