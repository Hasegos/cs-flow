Java — Spring WebSocket + STOMP 서버 구현

import org.springframework.web.socket.*;
import org.springframework.messaging.handler.annotation.*;
import org.springframework.messaging.simp.SimpMessagingTemplate;

/* ── WebSocket 설정 ──────────────────────────────────────── */
@Configuration
@EnableWebSocketMessageBroker
public class WebSocketConfig implements WebSocketMessageBrokerConfigurer {

    @Override
    public void configureMessageBroker(MessageBrokerRegistry registry) {
        // 인메모리 메시지 브로커 — /topic(브로드캐스트), /queue(1:1)
        registry.enableSimpleBroker("/topic", "/queue");
        // 클라이언트 → 서버 메시지 prefix
        registry.setApplicationDestinationPrefixes("/app");
    }

    @Override
    public void registerStompEndpoints(StompEndpointRegistry registry) {
        registry.addEndpoint("/ws")
                .setAllowedOriginPatterns("*")
                .withSockJS(); // WebSocket 미지원 환경 폴백
    }
}

/* ── 채팅 컨트롤러 ──────────────────────────────────────── */
@Controller
public class ChatController {

    private final SimpMessagingTemplate messagingTemplate;

    public ChatController(SimpMessagingTemplate messagingTemplate) {
        this.messagingTemplate = messagingTemplate;
    }

    // 클라이언트 → /app/chat.send → 브로드캐스트
    @MessageMapping("/chat.send")
    @SendTo("/topic/chat")
    public ChatMessage handleMessage(ChatMessage message) {
        message.setTimestamp(Instant.now());
        return message; // /topic/chat 구독자 전체에 전달
    }

    // 서버에서 특정 사용자에게 1:1 메시지 전송
    public void sendToUser(String userId, String message) {
        messagingTemplate.convertAndSendToUser(
            userId, "/queue/notification",
            new Notification(message)
        );
    }

    // 연결/해제 이벤트 처리
    @EventListener
    public void handleConnect(SessionConnectedEvent event) {
        log.info("WebSocket 연결: {}", event.getUser());
    }

    @EventListener
    public void handleDisconnect(SessionDisconnectEvent event) {
        log.info("WebSocket 해제: {}", event.getSessionId());
    }
}

@EnableWebSocketMessageBroker로 STOMP 메시지 브로커를 활성화합니다.
enableSimpleBroker는 인메모리 브로커로, 운영 환경에서는 Redis를 브로커로 사용해 여러 서버 인스턴스 간 메시지를 공유합니다.
@MessageMapping은 /app/chat.send로 오는 메시지를 처리하고, @SendTo는 /topic/chat 구독자 전체에 결과를 브로드캐스트합니다.
withSockJS()는 WebSocket을 지원하지 않는 환경에서 롱폴링 등으로 자동 폴백합니다.
현대 브라우저에서는 대부분 불필요하지만 기업 네트워크 프록시 환경에서 유용합니다.
setAllowedOriginPatterns("*")는 개발 환경용입니다. 운영 환경에서는 실제 도메인만 허용해야 합니다.

JavaScript — WebSocket 클라이언트 + 재연결 로직

// ── 기본 WebSocket API ───────────────────────────────────
const ws = new WebSocket('wss://example.com/ws');

ws.onopen = () => {
    console.log('연결 수립', ws.readyState); // OPEN = 1
    ws.send('안녕하세요!');               // 텍스트 전송
    ws.send(new Uint8Array([1,2,3]));    // 바이너리 전송
};

ws.onmessage = (event) => {
    if (typeof event.data === 'string') {
        const msg = JSON.parse(event.data);
        console.log('수신:', msg);
    }
};

ws.onclose = (event) => {
    // event.code: 1000(정상), 1006(비정상), 1011(서버오류)
    console.log(`종료: ${event.code} ${event.reason}`);
};

ws.onerror = (error) => console.error('오류:', error);

// ── 자동 재연결 래퍼 ─────────────────────────────────────
class ReconnectingWebSocket {
    constructor(url, options = {}) {
        this.url        = url;
        this.maxRetries = options.maxRetries || 5;
        this.retryDelay = options.retryDelay || 1000;
        this.retries    = 0;
        this.connect();
    }

    connect() {
        this.ws = new WebSocket(this.url);

        this.ws.onopen = () => {
            this.retries = 0; // 성공 시 카운터 리셋
            this.onopen?.();
        };

        this.ws.onmessage = (e) => this.onmessage?.(e);

        this.ws.onclose = (e) => {
            if (e.code !== 1000 && this.retries < this.maxRetries) {
                const delay = this.retryDelay * Math.pow(2, this.retries);
                setTimeout(() => this.connect(), delay); // 지수 백오프
                this.retries++;
            }
        };
    }

    send(data) {
        if (this.ws.readyState === WebSocket.OPEN)
            this.ws.send(JSON.stringify(data));
    }
}

WebSocket.readyState는 CONNECTING(0), OPEN(1), CLOSING(2), CLOSED(3) 네 가지 상태를 가집니다.
send() 호출 전 OPEN 상태인지 확인해야 합니다.
ws://는 평문, wss://는 TLS 암호화 WebSocket입니다. 운영 환경에서는 반드시 wss://를 사용합니다.
자동 재연결 시 지수 백오프(exponential backoff)를 적용합니다.
재연결 실패 시 점점 늘어나는 대기 시간으로 서버 부하를 줄입니다.
close 코드 1000은 정상 종료이며 재연결하지 않습니다. 1006은 비정상 종료(네트워크 단절 등)로 재연결을 시도합니다.
