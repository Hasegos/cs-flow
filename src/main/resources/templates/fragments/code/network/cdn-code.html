Java — Spring Boot Cache-Control 헤더 설정

import org.springframework.http.CacheControl;
import org.springframework.web.servlet.config.annotation.*;

/* ── 정적 리소스 Cache-Control 설정 ─────────────────────── */
@Configuration
public class WebConfig implements WebMvcConfigurer {

    @Override
    public void addResourceHandlers(ResourceHandlerRegistry registry) {
        // JS/CSS: 콘텐츠 해시 파일명 사용 → 1년 캐싱 + immutable
        registry.addResourceHandler("/js/**", "/css/**")
                .addResourceLocations("classpath:/static/js/",
                                       "classpath:/static/css/")
                .setCacheControl(CacheControl
                    .maxAge(365, TimeUnit.DAYS)
                    .cachePublic()
                    .immutable());

        // 이미지: 1시간 캐싱
        registry.addResourceHandler("/images/**")
                .addResourceLocations("classpath:/static/images/")
                .setCacheControl(CacheControl
                    .maxAge(1, TimeUnit.HOURS)
                    .cachePublic());
    }
}

/* ── HTML 응답: 캐싱 방지 ────────────────────────────────── */
@Controller
public class TopicController {

    @GetMapping("/{subject}/{topic}")
    public ResponseEntity<String> topic(
            @PathVariable String subject,
            @PathVariable String topic,
            Model model) {

        // HTML은 캐시 금지 — 내용이 바뀔 수 있음
        HttpHeaders headers = new HttpHeaders();
        headers.setCacheControl(
            CacheControl.noStore().getHeaderValue()
        );
        return ResponseEntity
            .ok()
            .headers(headers)
            .body("topics/" + subject + "/" + topic);
    }
}

/* ── Cloudflare 캐시 퍼지 (배포 시 자동 실행) ──────────── */
public class CloudflarePurgeService {

    private final String zoneId = System.getenv("CF_ZONE_ID");
    private final String apiKey = System.getenv("CF_API_KEY");

    public void purgeAll() {
        var client = HttpClient.newHttpClient();
        var request = HttpRequest.newBuilder()
            .uri(URI.create(
                "https://api.cloudflare.com/client/v4/zones/"
                + zoneId + "/purge_cache"))
            .header("Authorization", "Bearer " + apiKey)
            .header("Content-Type", "application/json")
            .POST(HttpRequest.BodyPublishers.ofString(
                "{\"purge_everything\": true}"))
            .build();

        client.sendAsync(request, HttpResponse.BodyHandlers.ofString())
              .thenAccept(res -> log.info("CF purge: {}", res.body()));
    }
}

정적 리소스(JS, CSS)는 콘텐츠 해시가 파일명에 포함되므로 파일이 변경되면 URL 자체가 바뀝니다.
따라서 max-age=1년 + immutable로 설정해도 안전합니다. Spring은 ResourceHandlerRegistry에서 이를 자동으로 처리합니다.
HTML은 항상 최신 버전을 보여야 하므로 no-store로 캐싱을 금지합니다. Cloudflare는 이 헤더를 존중해 HTML을 캐싱하지 않습니다.
배포 시 Cloudflare API로 캐시를 퍼지합니다. API 키는 반드시 환경 변수에서 읽어야 합니다.
GitHub Actions에서 배포 완료 후 purge_cache API를 호출하는 스텝을 추가하면 됩니다.

nginx — CDN 오리진 서버 설정

# nginx.conf — Cloudflare가 앞단에서 CDN 역할

server {
    listen 80;
    server_name csflow.kr;

    # Cloudflare Real IP 복원
    real_ip_header CF-Connecting-IP;

    # HTML — CDN 캐싱 금지
    location / {
        proxy_pass http://csflow_backend;
        add_header Cache-Control "no-store";
        add_header Vary "Accept-Encoding";
    }

    # 정적 리소스 — CDN이 1년 캐싱
    location ~* \.(css|js|png|jpg|svg|ico|woff|woff2)$ {
        proxy_pass http://csflow_backend;
        proxy_set_header Host $host;
        proxy_http_version 1.1;
        proxy_set_header Connection "";

        # Cloudflare가 이 헤더를 보고 엣지에 캐싱
        expires 1y;
        add_header Cache-Control "public, max-age=31536000, immutable";
    }

    # API — 캐싱 금지
    location /api/ {
        proxy_pass http://csflow_backend;
        add_header Cache-Control "no-store, no-cache";
        add_header Pragma "no-cache";
    }

    # Actuator — 외부 차단
    location /actuator {
        deny all;
        return 403;
    }
}

Cloudflare가 앞단에서 CDN 역할을 할 때 nginx는 오리진 서버로 동작합니다. Cache-Control 헤더를 통해 Cloudflare가 어떤 콘텐츠를 캐싱할지 결정합니다.
정적 리소스에 max-age=31536000 immutable을 설정하면 Cloudflare 엣지 서버가 1년간 캐싱합니다. 이후 해당 파일 요청은 오리진까지 오지 않습니다.
API 응답은 no-store로 캐싱을 막습니다. 캐시된 API 응답이 다른 사용자에게 전달되면 보안 문제가 발생할 수 있습니다.
