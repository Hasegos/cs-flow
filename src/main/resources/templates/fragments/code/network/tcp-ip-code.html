Java — TCP 서버 / 클라이언트 소켓 (3-way handshake 흐름)

import java.net.*;
import java.io.*;

/* ── 서버: LISTEN → SYN_RECEIVED → ESTABLISHED ────── */
public class TcpServer {
    public static void main(String[] args) throws Exception {

        // ServerSocket.bind() → 서버 LISTEN 상태
        ServerSocket server = new ServerSocket(8080);
        System.out.println("서버 LISTEN — 포트 8080");

        // accept()는 블로킹 — 클라이언트 SYN을 기다림
        // SYN 수신 → OS가 SYN+ACK 자동 전송 → ACK 수신
        // → 3-way handshake 완료 → ESTABLISHED
        Socket client = server.accept(); // 여기서 handshake 완료
        System.out.println("ESTABLISHED: " + client.getRemoteSocketAddress());

        try (var out = new PrintWriter(client.getOutputStream(), true);
             var in  = new BufferedReader(
                           new InputStreamReader(client.getInputStream()))) {
            String msg = in.readLine();
            out.println("Echo: " + msg);
        }
        // try-with-resources → close() → 4-way termination 자동 수행
        server.close();
    }
}

/* ── 클라이언트: CLOSED → SYN_SENT → ESTABLISHED ───── */
public class TcpClient {
    public static void main(String[] args) throws Exception {

        // new Socket() → SYN 전송 → SYN+ACK 수신 → ACK 전송
        // → ESTABLISHED (3-way handshake를 생성자에서 완료)
        try (Socket sock = new Socket("localhost", 8080)) {
            System.out.println("ESTABLISHED");

            // TCP 옵션 설정 예시
            sock.setTcpNoDelay(true);          // Nagle 알고리즘 비활성
            sock.setSoTimeout(5000);          // read 타임아웃 5초
            sock.setKeepAlive(true);          // TCP Keep-Alive 활성

            var out = new PrintWriter(sock.getOutputStream(), true);
            var in  = new BufferedReader(
                          new InputStreamReader(sock.getInputStream()));
            out.println("Hello TCP");
            System.out.println(in.readLine());
        }
        // 소켓 닫힘 → FIN 전송 → 4-way termination
    }
}

Java에서 new Socket("host", port)를 호출하는 순간 OS가 TCP 3-way handshake를 수행합니다.
개발자는 생성자가 반환되면 이미 ESTABLISHED 상태임을 알 수 있습니다.
ServerSocket.accept()는 OS의 백로그 큐에서 완료된 연결을 꺼냅니다. SYN_RECEIVED → ESTABLISHED 과정은 OS 커널이 자동으로 처리합니다.
Socket.close()를 호출하면 OS가 4-way termination을 수행합니다.
try-with-resources를 사용하면 예외 발생 시에도 FIN이 전송됩니다.
setTcpNoDelay(true)는 Nagle 알고리즘을 끄는 옵션으로, 소량의 패킷을 즉시 전송해야 하는 실시간 애플리케이션에 사용합니다.

Python — TCP 상태 확인 및 Raw 시퀀스 번호 분석

import socket
import struct
import subprocess

# ── TCP 상태 확인 (ss 명령어 래핑) ─────────────────────
def show_tcp_states():
    result = subprocess.run(
        ['ss', '-tn', 'state', 'established'],
        capture_output=True, text=True
    )
    print(result.stdout)
    # Recv-Q Send-Q LocalAddr:Port PeerAddr:Port
    # Recv-Q: 수신 버퍼 미처리 데이터 (0이어야 정상)
    # Send-Q: 미전송 데이터 (0이면 전송 완료)

# ── TCP 헤더 파싱 (raw socket, root 권한 필요) ─────────
def parse_tcp_header(raw: bytes):
    # IP 헤더 이후 TCP 헤더 파싱
    ip_ihl = (raw[0] & 0x0F) * 4          # IP 헤더 길이
    tcp    = raw[ip_ihl:]

    # !HHLLBBHHH: big-endian, 각 필드 언팩
    src_port, dst_port, seq, ack_num, \
    offset_flags, flags, window, checksum, urg = \
        struct.unpack('!HHLLBBHHH', tcp[:20])

    syn = (flags & 0x02) != 0
    ack = (flags & 0x10) != 0
    fin = (flags & 0x01) != 0
    rst = (flags & 0x04) != 0

    print(f"포트: {src_port} → {dst_port}")
    print(f"seq={seq}, ack={ack_num}")
    print(f"SYN={syn} ACK={ack} FIN={fin} RST={rst}")
    print(f"window={window}, checksum=0x{checksum:04X}")

    if syn and not ack:
        print("→ 3-way handshake Step 1: SYN")
    elif syn and ack:
        print("→ 3-way handshake Step 2: SYN+ACK")
    elif fin:
        print("→ 4-way termination: FIN")

# ── TIME_WAIT 확인 ──────────────────────────────────────
# $ ss -tn state time-wait
# TIME_WAIT는 2MSL(~60초) 후 자동 해제
# SO_REUSEADDR 옵션으로 TIME_WAIT 중 포트 재사용 가능

sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
sock.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
sock.bind(('', 8080))

ss -tn 명령어로 현재 시스템의 TCP 연결 상태를 확인할 수 있습니다. Recv-Q와 Send-Q가 지속적으로 0이 아니라면 버퍼 병목을 의심해야 합니다.
raw socket으로 TCP 헤더를 파싱하면 SYN·ACK·FIN 플래그를 비트 마스킹으로 추출할 수 있습니다. Wireshark·tcpdump가 이 방식으로 동작합니다.
SO_REUSEADDR 옵션을 서버 소켓에 설정하면 TIME_WAIT 상태의 포트를 즉시 재사용할 수 있습니다.
개발 서버 재시작 시 "Address already in use" 오류를 방지합니다.
TIME_WAIT는 OS가 관리하며 약 60초(2MSL) 후 자동 해제됩니다.
대용량 단기 연결이 많은 서버에서는 TIME_WAIT 소켓이 누적되어 포트 고갈이 발생할 수 있습니다.
