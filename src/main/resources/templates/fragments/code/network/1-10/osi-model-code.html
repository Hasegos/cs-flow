Python — 소켓 통신으로 보는 OSI 계층 흐름

import socket
import ssl

# ── 송신 측 (클라이언트) ──────────────────────
# Layer 7 — Application: HTTP 요청 메시지 생성
request = (
    "GET / HTTP/1.1\r\n"
    "Host: example.com\r\n"
    "Connection: close\r\n\r\n"
)

# Layer 4 — Transport: TCP 소켓 생성 (OS가 처리)
# socket.SOCK_STREAM = TCP, SOCK_DGRAM = UDP
raw_sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)

# Layer 6 — Presentation: TLS 래핑 (암호화)
context = ssl.create_default_context()
sock = context.wrap_socket(raw_sock, server_hostname="example.com")

# Layer 3 — Network: IP 주소로 목적지 지정
# OS가 DNS 조회 → IP 확보 → 라우팅 테이블 참조
sock.connect(("example.com", 443))
# ↑ connect() 시점에 TCP 3-way handshake 발생
#   Layer 2 — Data Link: Ethernet 프레임 생성 (NIC/OS)
#   Layer 1 — Physical:  전기 신호로 변환 (NIC 하드웨어)

# Layer 7: 데이터 전송 (하위 계층들이 캡슐화 처리)
sock.sendall(request.encode())

# ── 수신 측 (서버) — 역캡슐화 ──────────────────────
# Physical → Data Link → Network → Transport → Application
# 각 계층이 헤더를 벗겨내고 상위 계층에 전달
response = b""
while chunk := sock.recv(4096):
    response += chunk

sock.close()

# 애플리케이션 계층에 도착한 순수 HTTP 응답
print(response.decode()[:200])

Python socket API는 Transport 계층(L4) 위에서 동작합니다.
소켓을 생성하면 OS가 TCP 헤더를 붙이고(L4), IP 헤더를 붙이고(L3), Ethernet 프레임으로 감싸(L2), NIC가 전기 신호로 변환(L1)합니다.
개발자는 Application 계층의 데이터(HTTP 메시지)만 다루며, L1~L4의 캡슐화/역캡슐화는 OS와 하드웨어가 자동으로 처리합니다.
ssl.wrap_socket()은 L6(Presentation) 역할을 합니다. TLS 핸드셰이크로 암호화 키를 교환하고 이후 데이터를 암호화합니다.
connect()를 호출하는 순간 3-way handshake(SYN→SYN-ACK→ACK)가 발생하며 이는 L4 TCP의 연결 수립 과정입니다.

Java — Wireshark 없이 패킷 헤더 직접 파싱

import java.net.*;
import java.nio.*;

public class OsiLayerDemo {

    /**
     * 각 OSI 계층의 헤더 구조를 상수로 표현
     * 실제 패킷을 파싱할 때 이 오프셋으로 필드를 추출
     */

    // Layer 2 — Ethernet 프레임 헤더 (14바이트)
    static final int ETH_DST_MAC  = 0;  // 6바이트: 목적지 MAC
    static final int ETH_SRC_MAC  = 6;  // 6바이트: 출발지 MAC
    static final int ETH_TYPE     = 12; // 2바이트: 0x0800=IPv4

    // Layer 3 — IPv4 헤더 (최소 20바이트)
    static final int IP_VERSION   = 14; // 상위 4비트: 버전(4)
    static final int IP_TTL       = 22; // 1바이트: TTL
    static final int IP_PROTOCOL  = 23; // 1바이트: 6=TCP, 17=UDP
    static final int IP_SRC       = 26; // 4바이트: 출발지 IP
    static final int IP_DST       = 30; // 4바이트: 목적지 IP

    // Layer 4 — TCP 헤더 (최소 20바이트, IP 헤더 이후)
    static final int TCP_SRC_PORT = 34; // 2바이트: 출발지 포트
    static final int TCP_DST_PORT = 36; // 2바이트: 목적지 포트 (80/443)
    static final int TCP_SEQ      = 38; // 4바이트: 시퀀스 번호
    static final int TCP_FLAGS    = 47; // TCP 시작(34)+13: SYN=0x02, ACK=0x10, FIN=0x01 (오프셋 46은 Data Offset 바이트)

    // Layer 7 — HTTP 데이터 시작 위치 (TCP 옵션 없을 때)
    static final int HTTP_PAYLOAD = 54; // ETH(14)+IP(20)+TCP(20)

    static void parsePacket(byte[] raw) {
        ByteBuffer buf = ByteBuffer.wrap(raw);

        // L2 파싱
        short ethType = buf.getShort(ETH_TYPE);
        boolean isIPv4 = (ethType & 0xFFFF) == 0x0800;

        // L3 파싱
        byte protocol = buf.get(IP_PROTOCOL);
        boolean isTCP = protocol == 6;

        // L4 파싱
        int dstPort = buf.getShort(TCP_DST_PORT) & 0xFFFF;
        byte flags   = buf.get(TCP_FLAGS);
        boolean isSyn = (flags & 0x02) != 0;

        // L7: HTTP 페이로드 (54바이트 이후)
        if (raw.length > HTTP_PAYLOAD) {
            String payload = new String(raw, HTTP_PAYLOAD, raw.length - HTTP_PAYLOAD);
            // "GET / HTTP/1.1\r\nHost: ..."
        }
    }
}

실제 이더넷 프레임의 바이트 오프셋을 직접 정의했습니다. 패킷 캡처 도구(Wireshark, tcpdump)는 이 구조로 raw 바이트를 파싱해 계층별 필드를 표시합니다.
HTTP 페이로드는 Ethernet(14) + IP(20) + TCP(20) = 54바이트 이후에 위치합니다. TCP 옵션이 있으면 오프셋이 더 뒤로 밀립니다.
TCP flags 바이트에서 비트 마스킹으로 SYN·ACK·FIN을 구분합니다. 3-way handshake는 SYN → SYN+ACK → ACK 순서로 flags가 변합니다.
Java에서 실제 raw 소켓에 접근하려면 JNI 또는 pcap4j 같은 라이브러리가 필요합니다.
일반 socket API는 L4(TCP/UDP)부터만 접근 가능합니다.
