Java — Spring Boot HTTP 요청/응답 및 상태 코드 처리

import org.springframework.web.bind.annotation.*;
import org.springframework.http.*;
import org.springframework.web.client.RestTemplate;

/* ── REST 컨트롤러 (서버 측) ──────────────────────────── */
@RestController
@RequestMapping("/api")
public class HttpDemoController {

    // GET — 200 OK / 404 Not Found
    @GetMapping("/users/{id}")
    public ResponseEntity<User> getUser(@PathVariable Long id) {
        return userRepository.findById(id)
            .map(ResponseEntity::ok)              // 200 OK
            .orElse(ResponseEntity.notFound().build()); // 404
    }

    // POST — 201 Created + Location 헤더
    @PostMapping("/users")
    public ResponseEntity<User> createUser(@RequestBody UserDto dto) {
        User saved = userService.create(dto);
        URI location = URI.create("/api/users/" + saved.getId());
        return ResponseEntity.created(location).body(saved); // 201
    }

    // 캐시 제어 헤더 설정
    @GetMapping("/static/{file}")
    public ResponseEntity<Resource> getStatic(@PathVariable String file) {
        return ResponseEntity.ok()
            .cacheControl(CacheControl.maxAge(1, TimeUnit.HOURS)
                                      .cachePublic())   // Cache-Control: public, max-age=3600
            .eTag(computeETag(file))                   // ETag 헤더
            .body(loadResource(file));
    }

    // CORS 직접 설정 (전역은 WebMvcConfigurer 사용)
    @CrossOrigin(origins = "https://frontend.example.com")
    @GetMapping("/public")
    public ResponseEntity<String> publicApi() {
        return ResponseEntity.ok("OK"); // Access-Control-Allow-Origin 헤더 포함
    }
}

/* ── HTTP 클라이언트 (RestTemplate) ───────────────────── */
@Service
public class HttpClientService {

    private final RestTemplate rest = new RestTemplate();

    public void demo() {
        // GET 요청 — 응답 바디를 객체로 역직렬화
        ResponseEntity<User> res = rest.getForEntity(
            "https://api.example.com/users/1", User.class
        );
        System.out.println(res.getStatusCode()); // 200 OK
        System.out.println(res.getHeaders().getContentType());

        // POST 요청 — 헤더 직접 설정
        HttpHeaders headers = new HttpHeaders();
        headers.setContentType(MediaType.APPLICATION_JSON);
        headers.setBearerAuth("eyJhbGci..."); // Authorization: Bearer ...

        HttpEntity<UserDto> req = new HttpEntity<>(dto, headers);
        ResponseEntity<User> created = rest.postForEntity(
            "https://api.example.com/users", req, User.class
        );
        // 201 Created → Location 헤더에서 새 리소스 URI 확인
        URI newUri = created.getHeaders().getLocation();
    }
}

Spring의 ResponseEntity를 사용하면 HTTP 상태 코드와 헤더를 명시적으로 제어할 수 있습니다.
orElse(notFound().build())처럼 Optional과 조합하면 404 처리가 간결해집니다.
POST 생성 시 201 Created와 Location 헤더를 반환하는 것이 REST 표준입니다.
클라이언트는 Location을 보고 새 리소스 URI를 알 수 있습니다.
CacheControl 빌더로 Cache-Control 헤더를 설정하면 브라우저·CDN 캐싱을 제어합니다.
ETag와 함께 사용하면 조건부 요청(304)으로 불필요한 데이터 전송을 줄입니다.
실무에서는 RestTemplate보다 WebClient(비동기) 또는 OpenFeign(선언적)을 더 많이 사용합니다.

Python — HTTP 요청 분석 및 HTTP/2 비교

import httpx   # HTTP/1.1 + HTTP/2 모두 지원
import time

# ── HTTP/1.1 순차 요청 ──────────────────────────────────
def http1_sequential():
    urls = [
        "https://example.com/index.html",
        "https://example.com/style.css",
        "https://example.com/script.js",
    ]
    start = time.time()
    with httpx.Client(http2=False) as client:
        for url in urls:
            res = client.get(url)
            print(f"{res.status_code} {url}")
            # HOL 블로킹: 앞 요청 완료 후 다음 전송
    print(f"HTTP/1.1 총 시간: {time.time()-start:.2f}s")

# ── HTTP/2 멀티플렉싱 ────────────────────────────────────
import asyncio

async def http2_multiplex():
    urls = [
        "https://example.com/index.html",
        "https://example.com/style.css",
        "https://example.com/script.js",
    ]
    start = time.time()
    async with httpx.AsyncClient(http2=True) as client:
        # 3개 요청 동시 전송 (멀티플렉싱)
        tasks = [client.get(url) for url in urls]
        responses = await asyncio.gather(*tasks)
        for res in responses:
            print(f"{res.status_code} HTTP/{res.http_version}")
    print(f"HTTP/2 총 시간: {time.time()-start:.2f}s")

# ── 응답 헤더 분석 ──────────────────────────────────────
def inspect_headers():
    with httpx.Client() as client:
        res = client.get("https://httpbin.org/get")

        print(f"상태 코드: {res.status_code}")              # 200
        print(f"Content-Type: {res.headers['content-type']}")
        print(f"X-RateLimit-Remaining: {res.headers.get('x-ratelimit-remaining', 'N/A')}")

        # 조건부 요청 (ETag 활용)
        etag = res.headers.get('etag')
        if etag:
            res2 = client.get(
                "https://httpbin.org/get",
                headers={"If-None-Match": etag}
            )
            print(f"조건부 요청: {res2.status_code}")  # 304 Not Modified

# asyncio.run(http2_multiplex())

httpx 라이브러리는 HTTP/1.1과 HTTP/2를 모두 지원합니다. http2=True 옵션만으로 멀티플렉싱을 활성화할 수 있습니다.
HTTP/1.1 순차 요청과 HTTP/2 멀티플렉싱을 같은 URL로 비교하면 RTT 차이를 직접 확인할 수 있습니다. 리소스가 많을수록 HTTP/2의 이점이 커집니다.
ETag를 활용한 조건부 요청은 변경이 없으면 304 Not Modified를 반환해 바디 전송을 생략합니다. 정적 리소스 캐싱에 매우 효과적입니다.
실제 HTTP/2 멀티플렉싱 효과는 네트워크 지연(RTT)이 클수록 두드러집니다. 로컬 환경에서는 차이가 미미할 수 있습니다.
