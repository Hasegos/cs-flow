Python — ipaddress로 네트워크 정보 구하기

import ipaddress

net = ipaddress.ip_network("192.168.1.128/26")
print(net.network_address)     # 192.168.1.128
print(net.netmask)             # 255.255.255.192
print(net.broadcast_address)   # 192.168.1.191
print(net.num_addresses)       # 64

hosts = list(net.hosts())
print(hosts[0], hosts[-1], len(hosts))   # 192.168.1.129 192.168.1.190 62

print(ipaddress.ip_address("192.168.1.130") in net)   # True

파이썬 표준 라이브러리 ipaddress의 ip_network는 주소와 접두사 길이로 네트워크 객체를 만들고, 네트워크 주소, 넷마스크, 브로드캐스트 주소, 주소 개수를 속성으로 줍니다. hosts()는 보통 네트워크 주소와 브로드캐스트 주소를 뺀 호스트 주소(/31, /32는 예외)를 차례로 돌려주므로 /26에서는 62개입니다. in 연산자로 어떤 IP가 그 네트워크에 속하는지도 확인합니다. 접두사 뒤의 호스트 비트가 0이 아닌 주소(192.168.1.130/26)는 기본적으로 오류가 나며, strict=False를 주면 네트워크 주소로 맞춰서 받아 줍니다.

Python — 서브넷 나누기와 CIDR 요약

import ipaddress

net = ipaddress.ip_network("192.168.1.0/24")
for sub in net.subnets(new_prefix=26):
    print(sub, sub.broadcast_address)
# 192.168.1.0/26 192.168.1.63 ... 192.168.1.192/26 192.168.1.255

a = ipaddress.ip_network("192.168.0.0/24")
b = ipaddress.ip_network("192.168.1.0/24")
c = ipaddress.ip_network("192.168.2.0/24")

print(list(ipaddress.collapse_addresses([a, b])))   # 192.168.0.0/23 하나
print(list(ipaddress.collapse_addresses([b, c])))   # 합쳐지지 않고 둘 그대로

subnets(new_prefix=26)은 /24를 /26 네 개로 나눠 하나씩 돌려줍니다. 시각화의 서브넷 나누기 탭과 같은 결과입니다. collapse_addresses는 합칠 수 있는 네트워크를 가능한 한 적은 수의 네트워크로 합쳐 줍니다. 192.168.0.0/24와 192.168.1.0/24는 /23 하나가 되지만, 192.168.1.0/24와 192.168.2.0/24는 정렬이 맞지 않아 둘 그대로 남습니다. 이 함수는 정확히 같은 주소 범위만 합치므로 시각화에서 본 "원래 없던 네트워크까지 포함하는" 요약은 만들지 않습니다.

C — 비트 연산으로 네트워크 주소 구하기

#include <stdint.h>
#include <stdio.h>

int main(void) {
    uint32_t ip = (192u << 24) | (168u << 16) | (1u << 8) | 130u;
    int prefix = 26;

    uint32_t mask = prefix == 0 ? 0 : 0xFFFFFFFFu << (32 - prefix);
    uint32_t network = ip & mask;
    uint32_t broadcast = network | ~mask;

    printf("network %u.%u.%u.%u\n", network >> 24, (network >> 16) & 255,
           (network >> 8) & 255, network & 255);   /* 192.168.1.128 */
    return broadcast == network + 63 ? 0 : 1;
}

IPv4 주소를 32비트 정수로 보면, 마스크는 앞쪽 prefix개의 비트만 1인 값이고 네트워크 주소는 IP와 마스크의 AND, 브로드캐스트 주소는 네트워크 주소에 마스크를 뒤집은 값(~mask)을 OR한 것입니다. prefix가 0일 때는 32비트 이동이 정의되지 않아 따로 처리해야 하고, 이동하는 값은 부호 없는 정수(u)로 써야 합니다. 이 계산은 시각화의 비트로 계산 탭과 같으며, 192.168.1.130/26에서 네트워크 주소 192.168.1.128과 브로드캐스트 192.168.1.191(= 128 + 63)을 얻습니다.
