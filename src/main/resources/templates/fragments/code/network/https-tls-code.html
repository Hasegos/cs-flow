Java — Spring Boot HTTPS 설정 및 TLS 인증서 적용

## application.yml — TLS 설정
##
## server:
##   port: 443
##   ssl:
##     enabled: true
##     key-store: classpath:keystore.p12
##     key-store-password: ${SSL_KEY_STORE_PASSWORD}
##     key-store-type: PKCS12
##     protocol: TLS
##     enabled-protocols: TLSv1.3,TLSv1.2

import org.springframework.boot.web.embedded.tomcat.*;
import org.springframework.context.annotation.*;
import javax.net.ssl.*;
import java.net.http.*;
import java.net.URI;

/* ── HTTPS 리다이렉트 설정 ─────────────────────────────── */
@Configuration
public class HttpsConfig {

    // HTTP(8080) → HTTPS(443) 자동 리다이렉트
    @Bean
    public TomcatServletWebServerFactory servletContainer() {
        var factory = new TomcatServletWebServerFactory();
        factory.addAdditionalTomcatConnectors(httpConnector());
        return factory;
    }

    private Connector httpConnector() {
        var connector = new Connector(TomcatServletWebServerFactory.DEFAULT_PROTOCOL);
        connector.setScheme("http");
        connector.setPort(8080);
        connector.setSecure(false);
        connector.setRedirectPort(443); // HTTP → HTTPS 리다이렉트
        return connector;
    }
}

/* ── HSTS 헤더 설정 ────────────────────────────────────── */
@Configuration
@EnableWebSecurity
public class SecurityConfig {

    @Bean
    public SecurityFilterChain filterChain(HttpSecurity http) throws Exception {
        http
            .requiresChannel(ch -> ch.anyRequest().requiresSecure())
            .headers(headers -> headers
                .httpStrictTransportSecurity(hsts -> hsts
                    .maxAgeInSeconds(31536000) // HSTS 1년
                    .includeSubDomains(true)
                    .preload(true)
                )
            );
        return http.build();
    }
}

/* ── Java HTTP Client로 HTTPS 요청 ─────────────────────── */
public class HttpsClient {
    public static void main(String[] args) throws Exception {
        var client = HttpClient.newBuilder()
            .version(HttpClient.Version.HTTP_2)
            // TLS 1.3 자동 협상, 인증서 검증 포함
            .build();

        var request = HttpRequest.newBuilder()
            .uri(URI.create("https://example.com"))
            .GET()
            .build();

        var response = client.send(request,
            HttpResponse.BodyHandlers.ofString());

        System.out.println("상태: " + response.statusCode());
        System.out.println("버전: " + response.version());
    }
}

Spring Boot에서 HTTPS를 설정하려면 PKCS12 형식의 키스토어를 준비하고 application.yml에 경로를 설정합니다.
실제 운영에서는 Let's Encrypt로 발급한 인증서를 certbot이 자동 갱신합니다.
HSTS 헤더는 브라우저에 "이 도메인은 항상 HTTPS만 사용하라"고 지시합니다.
max-age=31536000은 1년간 강제 HTTPS를 의미합니다. preload=true로 설정하면 브라우저에 사전 등록할 수 있습니다.
Java 11+의 HttpClient는 TLS 1.3을 자동으로 협상하며 인증서 검증을 기본으로 수행합니다.
개발 환경에서 자체 서명 인증서를 사용할 때만 검증을 비활성화합니다.
키스토어 비밀번호는 절대 소스 코드나 설정 파일에 하드코딩하지 않고 환경 변수나 시크릿 매니저를 사용합니다.

Python — TLS 핸드셰이크 상세 정보 확인

import ssl
import socket
import json

# ── TLS 연결 정보 상세 확인 ─────────────────────────────
def inspect_tls(host: str, port: int = 443):
    context = ssl.create_default_context()

    with socket.create_connection((host, port)) as sock:
        with context.wrap_socket(sock, server_hostname=host) as ssock:

            # TLS 버전 확인
            print(f"TLS 버전: {ssock.version()}")        # TLSv1.3

            # 협상된 암호 스위트
            cipher = ssock.cipher()
            print(f"암호 스위트: {cipher[0]}")         # TLS_AES_256_GCM_SHA384
            print(f"프로토콜: {cipher[1]}")            # TLSv1.3
            print(f"키 길이: {cipher[2]} bits")       # 256

            # 서버 인증서 정보
            cert = ssock.getpeercert()
            subject = dict(x[0] for x in cert['subject'])
            issuer  = dict(x[0] for x in cert['issuer'])

            print(f"도메인: {subject.get('commonName')}")
            print(f"발급자: {issuer.get('organizationName')}")
            print(f"만료일: {cert['notAfter']}")

            # SAN (Subject Alternative Names)
            sans = cert.get('subjectAltName', [])
            print(f"SAN: {[s[1] for s in sans]}")

# ── 인증서 체인 확인 ────────────────────────────────────
def check_cert_chain(host: str):
    context = ssl.create_default_context()
    context.check_hostname = True
    context.verify_mode    = ssl.CERT_REQUIRED

    with socket.create_connection((host, 443)) as sock:
        with context.wrap_socket(sock, server_hostname=host) as ssock:
            # DER 형식 인증서 체인
            der_certs = ssock.getpeercert(binary_form=True)
            print(f"인증서 체인 크기: {len(der_certs)} bytes")

# ── 실행
inspect_tls("example.com")
# $ openssl s_client -connect example.com:443 -tls1_3 로도 확인 가능

ssl.create_default_context()는 시스템의 CA 번들을 사용해 인증서를 검증합니다.
check_hostname=True와 verify_mode=CERT_REQUIRED가 기본값으로, 인증서 검증을 건너뛰지 않는 것이 중요합니다.
cipher() 메서드로 실제 협상된 암호 스위트를 확인할 수 있습니다.
TLS 1.3에서는 TLS_AES_256_GCM_SHA384 또는 TLS_CHACHA20_POLY1305_SHA256 중 하나가 선택됩니다.
SAN(Subject Alternative Names)은 하나의 인증서가 여러 도메인을 커버하는 데 사용됩니다.
와일드카드(*.example.com)도 SAN에 포함됩니다.
openssl s_client 명령은 TLS 핸드셰이크 전체 과정을 터미널에서 디버깅할 때 가장 유용한 도구입니다.
