Java — Spring Cloud LoadBalancer 구현 (Round Robin / Random)

// ── Spring Cloud LoadBalancer 의존성 ──────────────────────
// implementation 'org.springframework.cloud:spring-cloud-starter-loadbalancer'

import org.springframework.cloud.client.loadbalancer.*;
import org.springframework.cloud.loadbalancer.core.*;
import reactor.core.publisher.Mono;

/* ── 커스텀 Round Robin 로드 밸런서 ─────────────────────── */
public class RoundRobinLoadBalancer implements ReactorServiceInstanceLoadBalancer {

    private final ObjectProvider<ServiceInstanceListSupplier> supplier;
    private final AtomicInteger position = new AtomicInteger(0);
    private final String serviceId;

    @Override
    public Mono<Response<ServiceInstance>> choose(Request request) {
        return supplier.getIfAvailable(NoopServiceInstanceListSupplier::new)
            .get(request)
            .next()
            .map(instances -> processInstanceResponse(instances));
    }

    private Response<ServiceInstance> processInstanceResponse(
            ServiceInstanceResponse instances) {
        var list = instances.getInstances();
        if (list.isEmpty()) return new EmptyResponse<>();

        // Round Robin: AtomicInteger로 스레드 안전하게 순환
        int pos  = Math.abs(position.getAndIncrement());
        var inst = list.get(pos % list.size());
        return new DefaultResponse<>(inst);
    }
}

/* ── Least Connection 로드 밸런서 ────────────────────────── */
public class LeastConnectionLoadBalancer implements ReactorServiceInstanceLoadBalancer {

    // 서버별 활성 연결 수 추적
    private final ConcurrentHashMap<String, AtomicInteger> connectionCounts
        = new ConcurrentHashMap<>();

    public ServiceInstance selectInstance(List<ServiceInstance> instances) {
        return instances.stream()
            .min(Comparator.comparingInt(inst ->
                connectionCounts
                    .computeIfAbsent(inst.getInstanceId(),
                                      k -> new AtomicInteger(0))
                    .get()
            ))
            .orElseThrow();
    }

    public void incrementConn(String instanceId) {
        connectionCounts.computeIfAbsent(instanceId, k -> new AtomicInteger(0))
                        .incrementAndGet();
    }

    public void decrementConn(String instanceId) {
        connectionCounts.computeIfPresent(instanceId,
            (k, v) -> { v.decrementAndGet(); return v; });
    }
}

/* ── RestTemplate에 로드 밸런서 적용 ────────────────────── */
@Configuration
public class LoadBalancerConfig {

    @Bean
    @LoadBalanced  // 서비스명으로 자동 로드 밸런싱
    public RestTemplate restTemplate() {
        return new RestTemplate();
    }
}

// 사용: 서비스 이름으로 호출 → LB가 자동으로 인스턴스 선택
// restTemplate.getForObject("http://order-service/api/orders", ...)

Spring Cloud LoadBalancer는 서비스 디스커버리(Eureka, Consul)와 연동해 동적으로 인스턴스를 관리합니다.
@LoadBalanced 어노테이션 하나로 RestTemplate에 로드 밸런싱이 적용됩니다.
Least Connection 구현에서 AtomicInteger로 동시성을 처리합니다.
요청 완료 시 반드시 decrementConn을 호출해야 하므로, 실제로는 try-finally나 인터셉터를 사용합니다.
Round Robin에서 position.getAndIncrement()의 오버플로우를 방지하기 위해 Math.abs()를 사용합니다.
Integer.MIN_VALUE의 abs()는 여전히 음수이므로 주의가 필요합니다.

nginx — L7 로드 밸런서 설정

# /etc/nginx/conf.d/load-balancer.conf

# ── upstream 서버 풀 정의 ──────────────────────────────────
upstream backend {
    # Round Robin (기본값)
    server 10.0.0.1:8080 weight=3;  # 가중치 3 (더 많은 요청)
    server 10.0.0.2:8080 weight=2;
    server 10.0.0.3:8080 weight=1;

    # Least Connection 방식
    # least_conn;

    # IP Hash (세션 유지)
    # ip_hash;

    # 헬스 체크 설정
    keepalive 32;  # 연결 재사용
}

upstream api_backend {
    least_conn;  # API는 처리 시간 다양 → Least Connection 적합
    server 10.0.0.4:8081;
    server 10.0.0.5:8081 backup;  # 나머지 전부 다운 시에만 사용
    server 10.0.0.6:8081 down;    # 일시적으로 비활성화
}

server {
    listen 80;

    # L7 콘텐츠 기반 라우팅
    location /api/ {
        proxy_pass http://api_backend;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    }

    location / {
        proxy_pass http://backend;
        proxy_set_header Host $host;
        proxy_http_version 1.1;
        proxy_set_header Connection "";  # keepalive 활성화

        # 헬스 체크 (nginx plus 기능)
        # health_check interval=5 fails=3 passes=2 uri=/health;
    }
}

nginx는 upstream 블록으로 서버 풀을 정의합니다. 기본값은 Round Robin이며, least_conn이나 ip_hash 디렉티브로 방식을 변경합니다.
weight 파라미터로 Weighted Round Robin을 구현합니다. weight=3인 서버는 전체 요청의 3/6=50%를 처리합니다.
backup 서버는 일반 서버가 모두 응답 불가일 때만 활성화됩니다. down은 임시 점검 시 서버를 풀에서 제외합니다.
proxy_http_version 1.1과 Connection="" 설정으로 upstream과의 keepalive 연결을 유지해 연결 오버헤드를 줄입니다.
